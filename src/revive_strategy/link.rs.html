1use std::{collections::BTreeMap, fs, path::Path, process::Command};
2
3const ELF_MAGIC: [u8; 4] = [0x7f, b'E', b'L', b'F'];
4
5fn is_elf(bytes: &[u8]) -> bool {
6    bytes.len() >= 4 && bytes[..4] == ELF_MAGIC
7}
8
9pub fn resolc_link(
10    resolc_path: &Path,
11    solc_path: Option<&Path>,
12    bytecodes: &BTreeMap<String, Vec<u8>>,
13    libraries: &[String],
14) -> eyre::Result<BTreeMap<String, Vec<u8>>> {
15    let temp_dir = tempfile::tempdir()?;
16
17    let mut relative_paths = Vec::new();
18    let mut path_to_id: BTreeMap<std::path::PathBuf, String> = BTreeMap::new();
19    for (contract_id, bytecode) in bytecodes {
20        let relative = std::path::PathBuf::from(format!("{contract_id}.pvm"));
21        let absolute = temp_dir.path().join(&relative);
22        if let Some(parent) = absolute.parent() {
23            fs::create_dir_all(parent)?;
24        }
25        fs::write(&absolute, bytecode)?;
26        path_to_id.insert(relative.clone(), contract_id.clone());
27        relative_paths.push(relative);
28    }
29
30    let mut cmd = Command::new(resolc_path);
31    cmd.arg("--link");
32    cmd.current_dir(temp_dir.path());
33    if let Some(solc) = solc_path {
34        cmd.arg("--solc").arg(solc);
35    }
36    for lib in libraries {
37        cmd.arg("--libraries").arg(lib);
38    }
39    for path in &relative_paths {
40        cmd.arg(path);
41    }
42
43    let output = cmd.output()?;
44    if !output.status.success() {
45        let stderr = String::from_utf8_lossy(&output.stderr);
46        eyre::bail!("resolc --link failed (exit code {:?}): {stderr}", output.status.code());
47    }
48
49    let mut linked = BTreeMap::new();
50    let mut still_unlinked = Vec::new();
51    for relative in &relative_paths {
52        let contract_id = &path_to_id[relative];
53        let original = &bytecodes[contract_id];
54        let absolute = temp_dir.path().join(relative);
55        let updated = fs::read(&absolute)?;
56
57        if is_elf(original) && !is_elf(&updated) {
58            linked.insert(contract_id.clone(), updated);
59        } else if is_elf(original) && is_elf(&updated) {
60            still_unlinked.push(contract_id.clone());
61        }
62    }
63
64    if !still_unlinked.is_empty() {
65        tracing::warn!(
66            target: "forge",
67            "resolc --link: {} contract(s) remain unlinked: {}",
68            still_unlinked.len(),
69            still_unlinked.join(", ")
70        );
71    }
72
73    Ok(linked)
74}
