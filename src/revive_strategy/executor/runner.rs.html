21/// Defines the [ExecutorStrategyRunner] strategy for Revive.

31impl ExecutorStrategyRunner for ReviveExecutorStrategyRunner {
32    fn new_backend_strategy(&self, _ctx: &dyn ExecutorStrategyContext) -> BackendStrategy {
33        BackendStrategy::new_revive()
34    }
35
36    fn new_cheatcodes_strategy(
37        &self,
38        ctx: &dyn ExecutorStrategyContext,
39    ) -> foundry_cheatcodes::CheatcodesStrategy {
40        let ctx = get_context_ref(ctx);
41        CheatcodeInspectorStrategy::new_pvm(
42            ctx.dual_compiled_contracts.clone(),
43            ctx.runtime_mode,
44            ctx.externalties.shallow_clone(),
45        )
46    }
47
48    /// Sets the balance of an account.
49    ///
50    /// Amount should be in the range of [0, u128::MAX] despite the type
51    /// because Ethereum balances are u256 while Polkadot balances are u128.
52    fn set_balance(
53        &self,
54        executor: &mut foundry_evm::executors::Executor,
55        address: Address,
56        amount: U256,
57    ) -> foundry_evm::backend::BackendResult<()> {
58        EvmExecutorStrategyRunner.set_balance(executor, address, amount)?;
59
60        let ctx = get_context_ref_mut(executor.strategy.context.as_mut());
61
62        ctx.externalties.set_balance(address, amount);
63        Ok(())
64    }
65
66    fn get_balance(
67        &self,
68        executor: &mut foundry_evm::executors::Executor,
69        address: Address,
70    ) -> foundry_evm::backend::BackendResult<U256> {
71        let evm_balance = EvmExecutorStrategyRunner.get_balance(executor, address)?;
72        let ctx = get_context_ref_mut(executor.strategy.context.as_mut());
73
74        let revive_balance = ctx.externalties.get_balance(address);
75        assert_eq!(evm_balance, revive_balance);
76        Ok(evm_balance)
77    }
78
79    fn set_nonce(
80        &self,
81        executor: &mut foundry_evm::executors::Executor,
82        address: Address,
83        nonce: u64,
84    ) -> foundry_evm::backend::BackendResult<()> {
85        EvmExecutorStrategyRunner.set_nonce(executor, address, nonce)?;
86        let ctx = get_context_ref_mut(executor.strategy.context.as_mut());
87        ctx.externalties.set_nonce(address, nonce);
88        Ok(())
89    }
90
91    fn get_nonce(
92        &self,
93        executor: &mut foundry_evm::executors::Executor,
94        address: Address,
95    ) -> foundry_evm::backend::BackendResult<u64> {
96        let evm_nonce = EvmExecutorStrategyRunner.get_nonce(executor, address)?;
97        let ctx = get_context_ref_mut(executor.strategy.context.as_mut());
98
99        let revive_nonce = ctx.externalties.get_nonce(address);
100
101        assert_eq!(evm_nonce, revive_nonce as u64);
102        Ok(evm_nonce)
103    }

128fn get_context_ref(ctx: &dyn ExecutorStrategyContext) -> &ReviveExecutorStrategyContext {

138impl ExecutorStrategyExt for ReviveExecutorStrategyRunner {

146        config: &foundry_config::Config,

162    fn revive_set_dual_compiled_contracts(
163        &self,
164        ctx: &mut dyn ExecutorStrategyContext,
165        dual_compiled_contracts: DualCompiledContracts,
166    ) {
167        let ctx = get_context_ref_mut(ctx);
168        *ctx.dual_compiled_contracts.lock().unwrap() = dual_compiled_contracts;
169    }
170
171    fn revive_set_compilation_output(
172        &self,
173        ctx: &mut dyn ExecutorStrategyContext,
174        output: std::sync::Arc<ProjectCompileOutput>,

177        ctx.compilation_output.replace(output);

179
180    fn revive_set_resolc_output(
181        &self,

183        output: std::sync::Arc<ProjectCompileOutput>,
184    ) {
185        let ctx = get_context_ref_mut(ctx);
186        ctx.resolc_output.replace(output);
187    }
188
189    fn start_transaction(&self, ctx: &dyn ExecutorStrategyContext) {
190        let ctx = get_context_ref(ctx);
191        let mut state = ctx.externalties.0.lock().unwrap();
192        TestEnv::start_transaction(&mut state);
193    }
194
195    fn rollback_transaction(&self, ctx: &dyn ExecutorStrategyContext) {
196        let ctx = get_context_ref(ctx);
197        let mut state = ctx.externalties.0.lock().unwrap();
198        TestEnv::revert_transaction(&mut state);
199    }
