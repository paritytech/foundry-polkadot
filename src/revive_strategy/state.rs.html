86    pub fn start_snapshotting(&mut self, snapshot_id: U256) {
87        let mut state = self.0.lock().unwrap();
88        let current_depth = state.depth;
89        state.snapshot_depths.insert(snapshot_id, current_depth);
90        Self::start_transaction(&mut state);
91        state.depth += 1;
92    }

94    pub fn revert(&mut self, snapshot_id: U256) {
95        let mut state = self.0.lock().unwrap();
96
97        let target_depth = match state.snapshot_depths.get(&snapshot_id) {
98            Some(&depth) => depth,
99            None => {
100                // Unknown snapshot - reset pallet-revive completely.
101                // This can happen with cross-function snapshots (Clone committed transactions)
102                // in setUp or test contract constructor call
103                tracing::warn!(
104                    snapshot_id = ?snapshot_id,
105                    current_depth = state.depth,
106                    "snapshot not found, resetting pallet-revive to sync with REVM"
107                );
108                while state.depth > 0 {
109                    Self::revert_transaction(&mut state);
110                    state.depth -= 1;
111                }
112                state.snapshot_depths.clear();
113                return;
114            }
115        };
116
117        let rollbacks_needed = state.depth.saturating_sub(target_depth);
118        for _ in 0..rollbacks_needed {
119            if state.depth > 0 {
120                Self::revert_transaction(&mut state);
121                state.depth -= 1;
122            }
123        }
124
125        // Remove snapshots that are now invalid (taken after the target snapshot)
126        state.snapshot_depths.retain(|_, &mut depth| depth <= target_depth);
127
128        Self::start_transaction(&mut state);
129        state.depth = target_depth + 1;
130    }

146    pub fn execute_with_transient_storage<
147        R,
148        F: FnOnce(RefCell<TransientStorage<Runtime>>) -> (R, RefCell<TransientStorage<Runtime>>),
149    >(
150        &mut self,
151        f: F,
152    ) -> R {
153        let mut data = self.0.lock().unwrap();
154        let mut temp = TransientStorage::new(u32::MAX);
155        std::mem::swap(&mut data.transient_storage, &mut temp);
156        let transient_storage = RefCell::new(temp);
157        let (result, transient_storage) = data.externalities.execute_with(|| f(transient_storage));
158        data.transient_storage = transient_storage.into_inner();
159        result
160    }

191    pub fn set_block_number(
192        &mut self,
193        new_height: U256,
194        prev_new_height_hash: B256,
195        new_height_hash: B256,
196    ) -> U256 {
197        // Set block number in pallet-revive runtime.
198        self.0.lock().unwrap().externalities.execute_with(|| {
199            let u64_max = U256::from(u64::MAX);
200            let clamped_height = if new_height > u64_max {
201                tracing::warn!(
202                    block_number = ?new_height,
203                    max = ?u64_max,
204                    "Block number exceeds u64::MAX. Clamping to u64::MAX."
205                );
206                u64_max
207            } else {
208                new_height
209            };
210
211            let new_block_number: u64 = clamped_height.to();
212            let digest = System::digest();
213            if System::block_hash(new_block_number) == H256::zero() {
214                // First initialize and finalize the parent block to set up correct hashes.
215                if new_block_number > 0 {
216                    System::set_block_number(new_block_number - 1);
217                    let current_hash = H256::from_slice(prev_new_height_hash.0.as_slice());
218                    System::initialize(&new_block_number, &current_hash, &digest);
219                }
220
221                // Now finalize the new block to set up its hash.
222                if new_block_number < u64::MAX {
223                    System::set_block_number(new_block_number);
224                    let current_hash = H256::from_slice(new_height_hash.0.as_slice());
225                    System::initialize(&(new_block_number + 1), &current_hash, &digest);
226                }
227            }
228            System::set_block_number(new_block_number);
229            clamped_height
230        })
231    }

238    pub fn roll<DB: revm::Database + ?Sized>(
239        &mut self,
240        new_height: U256,
241        database: &mut DB,
242    ) -> U256 {
243        let block_num_u64 = new_height.saturating_to::<u64>();
244        let prev_block_hash =
245            database.block_hash(block_num_u64.saturating_sub(1)).unwrap_or_default();
246        let current_block_hash = database.block_hash(block_num_u64).unwrap_or_default();
247
248        self.set_block_number(new_height, prev_block_hash, current_block_hash)
249    }

251    pub fn set_timestamp(&mut self, new_timestamp: U256) -> U256 {
252        // Set timestamp in pallet-revive runtime (milliseconds).
253        // We clamp to u64::MAX / 1000 to prevent overflow when converting to milliseconds.
254        self.0.lock().unwrap().externalities.execute_with(|| {
255            let max_timestamp_seconds = U256::from(u64::MAX / 1000);
256            let clamped_timestamp = if new_timestamp > max_timestamp_seconds {
257                tracing::warn!(
258                    timestamp = ?new_timestamp,
259                    max = ?max_timestamp_seconds,
260                    "Timestamp exceeds maximum. Clamping to u64::MAX / 1000."
261                );
262                max_timestamp_seconds
263            } else {
264                new_timestamp
265            };
266
267            let timestamp_ms = clamped_timestamp.saturating_to::<u64>() * 1000;
268            Timestamp::set_timestamp(timestamp_ms);
269            clamped_timestamp
270        })
271    }

273    fn set_base_deposit_hold(
274        target_address: &H160,
275        target_account: &AccountId32,
276        contract_info: &mut ContractInfo<Runtime>,
277        code_deposit: u128,
278    ) -> foundry_cheatcodes::Result {
279        contract_info.update_base_deposit(code_deposit);
280
281        let base_deposit: u128 = contract_info.storage_base_deposit();
282        let hold_reason: revive_env::RuntimeHoldReason = HoldReason::StorageDepositReserve.into();
283
284        // Release any existing hold
285        let current_held = Balances::balance_on_hold(&hold_reason, target_account);
286        if current_held > 0 {
287            Balances::release(&hold_reason, target_account, current_held, Precision::BestEffort)
288                .map_err(|_| <&str as Into<Error>>::into("Could not release old hold"))?;
289
290            // Decrease EVM balance by released amount (hold became free, so visible balance would
291            // increase)
292            let current_evm_balance = Pallet::<Runtime>::evm_balance(target_address);
293            let release_wei = sp_core::U256::from(current_held)
294                .saturating_mul(sp_core::U256::from(NativeToEthRatio::get() as u128));
295            let adjusted_balance = current_evm_balance.saturating_sub(release_wei);
296            Pallet::<Runtime>::set_evm_balance(target_address, adjusted_balance).map_err(|_| {
297                <&str as Into<Error>>::into("Could not adjust balance after release")
298            })?;
299        }
300
301        // Create new hold with correct amount
302        if base_deposit > 0 {
303            let current_evm_balance = Pallet::<Runtime>::evm_balance(target_address);
304            let hold_wei = sp_core::U256::from(base_deposit)
305                .saturating_mul(sp_core::U256::from(NativeToEthRatio::get() as u128));
306            let new_evm_balance = current_evm_balance.saturating_add(hold_wei);
307
308            Pallet::<Runtime>::set_evm_balance(target_address, new_evm_balance)
309                .map_err(|_| <&str as Into<Error>>::into("Could not set balance for new hold"))?;
310
311            Balances::hold(&hold_reason, target_account, base_deposit)
312                .map_err(|_| <&str as Into<Error>>::into("Could not create new hold"))?;
313        }
314
315        Ok(Default::default())
316    }

318    pub fn etch_call(&mut self, target: &Address, new_runtime_code: &Bytes) -> Result {
319        self.0.lock().unwrap().externalities.execute_with(|| {
320            let target_address = H160::from_slice(target.as_slice());
321            let target_account =
322                AccountId32Mapper::<Runtime>::to_fallback_account_id(&target_address);
323
324            let code = new_runtime_code.to_vec();
325            let code_type =
326                if code.starts_with(b"PVM\0") { BytecodeType::Pvm } else { BytecodeType::Evm };
327            let contract_blob = Pallet::<Runtime>::try_upload_code(
328                Pallet::<Runtime>::account_id(),
329                code,
330                code_type,
331                &mut ResourceMeter::new(pallet_revive::TransactionLimits::WeightAndDeposit {
332                    weight_limit: Weight::from_parts(10_000_000_000_000, 100_000_000),
333                    deposit_limit: { 100_000_000_000_000 },
334                })
335                .unwrap(),
336                &ExecConfig::new_substrate_tx(),
337            )
338            .map_err(|_| <&str as Into<Error>>::into("Could not upload PVM code"))?;
339
340            let code_deposit = contract_blob.code_info().deposit();
341            let code_hash = *contract_blob.code_hash();
342
343            let mut contract_info = if let Some(contract_info) =
344                AccountInfo::<Runtime>::load_contract(&target_address)
345            {
346                contract_info
347            } else {
348                let contract_info = ContractInfo::<Runtime>::new(
349                    &target_address,
350                    System::account_nonce(&target_account),
351                    code_hash,
352                )
353                .map_err(|err| {
354                    tracing::error!("Could not create contract info: {:?}", err);
355                    <&str as Into<Error>>::into("Could not create contract info")
356                })?;
357                System::inc_account_nonce(AccountId32Mapper::<Runtime>::to_fallback_account_id(
358                    &target_address,
359                ));
360                contract_info
361            };
362
363            contract_info.code_hash = code_hash;
364
365            // Update base deposit hold for both new and existing contracts
366            // Note: Code upload deposits are already held on the pallet account by try_upload_code
367            Self::set_base_deposit_hold(
368                &target_address,
369                &target_account,
370                &mut contract_info,
371                code_deposit,
372            )?;
373
374            AccountInfo::<Runtime>::insert_contract(
375                &H160::from_slice(target.as_slice()),
376                contract_info.clone(),
377            );
378
379            Ok::<(), Error>(())
380        })?;
381        Ok(Default::default())
382    }

422    pub fn set_balance(&mut self, address: Address, amount: U256) -> U256 {
423        let u128_max = U256::from(u128::MAX);
424        let clamped_amount = if amount > u128_max {
425            tracing::warn!(
426                address = ?address,
427                requested = ?amount,
428                actual = ?u128_max,
429                "Balance exceeds u128::MAX, clamping to u128::MAX. \
430                 pallet-revive uses u128 for balances."
431            );
432            u128_max
433        } else {
434            amount
435        };
436
437        let amount_pvm = sp_core::U256::from_little_endian(&clamped_amount.as_le_bytes());
438
439        self.0.lock().unwrap().externalities.execute_with(|| {
440            let h160_addr = H160::from_slice(address.as_slice());
441            pallet_revive::Pallet::<Runtime>::set_evm_balance(&h160_addr, amount_pvm)
442                .expect("failed to set evm balance");
443        });
444
445        clamped_amount
446    }
