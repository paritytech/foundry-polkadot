68impl<T: AsRef<Path>> FoundryPathExt for T {
69    fn is_sol_test(&self) -> bool {
70        self.as_ref()
71            .file_name()
72            .and_then(|s| s.to_str())
73            .map(|s| s.ends_with(".t.sol"))
74            .unwrap_or_default()
75    }
76
77    fn is_sol(&self) -> bool {
78        self.as_ref().extension() == Some(std::ffi::OsStr::new("sol"))
79    }
80
81    fn is_yul(&self) -> bool {
82        self.as_ref().extension() == Some(std::ffi::OsStr::new("yul"))
83    }

109/// Returns a [ProviderBuilder] instantiated using [Config] values.
110///
111/// Defaults to `http://localhost:8545` and `Mainnet`.
112pub fn get_provider_builder(config: &Config) -> Result<ProviderBuilder> {
113    let url = config.get_rpc_url_or_localhost_http()?;
114    let mut builder = ProviderBuilder::new(url.as_ref());
115
116    builder = builder.accept_invalid_certs(config.eth_rpc_accept_invalid_certs);
117
118    if let Ok(chain) = config.chain.unwrap_or_default().try_into() {
119        builder = builder.chain(chain);
120    }
121
122    if let Some(jwt) = config.get_rpc_jwt_secret()? {
123        builder = builder.jwt(jwt.as_ref());
124    }
125
126    if let Some(rpc_timeout) = config.eth_rpc_timeout {
127        builder = builder.timeout(Duration::from_secs(rpc_timeout));
128    }
129
130    if let Some(rpc_headers) = config.eth_rpc_headers.clone() {
131        builder = builder.headers(rpc_headers);
132    }
133
134    Ok(builder)
135}

137/// Return an [ExecutorStrategy] via the config.
138pub fn get_executor_strategy(config: &Config) -> ExecutorStrategy {
139    use foundry_config::revive::PolkadotMode;
140    use revive_strategy::{ReviveExecutorStrategyBuilder, ReviveRuntimeMode};
141
142    match config.polkadot.polkadot {
143        // No --polkadot flag: Standard Foundry EVM
144        None => {
145            info!("using standard EVM strategy");
146            ExecutorStrategy::new_evm()
147        }
148        // --polkadot or --polkadot=evm: Polkadot EVM backend
149        Some(PolkadotMode::Evm) => {
150            info!("using revive strategy with EVM backend on pallet-revive");
151            ExecutorStrategy::new_revive(ReviveRuntimeMode::Evm)
152        }
153        // --polkadot=pvm: Polkadot PVM backend
154        Some(PolkadotMode::Pvm) => {
155            info!("using revive strategy with PVM backend");
156            ExecutorStrategy::new_revive(ReviveRuntimeMode::Pvm)
157        }
158    }
159}

171/// Parses an ether value from a string.
172///
173/// The amount can be tagged with a unit, e.g. "1ether".
174///
175/// If the string represents an untagged amount (e.g. "100") then
176/// it is interpreted as wei.
177pub fn parse_ether_value(value: &str) -> Result<U256> {
178    Ok(if value.starts_with("0x") {
179        U256::from_str_radix(value, 16)?
180    } else {
181        alloy_dyn_abi::DynSolType::coerce_str(&alloy_dyn_abi::DynSolType::Uint(256), value)?
182            .as_uint()
183            .wrap_err("Could not parse ether value from string")?
184            .0
185    })
186}

193/// Parses a `Duration` from a &str
194pub fn parse_delay(delay: &str) -> Result<Duration> {
195    let delay = if delay.ends_with("ms") {
196        let d: u64 = delay.trim_end_matches("ms").parse()?;
197        Duration::from_millis(d)
198    } else {
199        let d: f64 = delay.parse()?;
200        let delay = (d * 1000.0).round();
201        if delay.is_infinite() || delay.is_nan() || delay.is_sign_negative() {
202            eyre::bail!("delay must be finite and non-negative");
203        }
204
205        Duration::from_millis(delay as u64)
206    };
207    Ok(delay)
208}

223/// Loads a dotenv file, from the cwd and the project root, ignoring potential failure.
224///
225/// We could use `warn!` here, but that would imply that the dotenv file can't configure
226/// the logging behavior of Foundry.
227///
228/// Similarly, we could just use `eprintln!`, but colors are off limits otherwise dotenv is implied
229/// to not be able to configure the colors. It would also mess up the JSON output.
230pub fn load_dotenv() {
231    let load = |p: &Path| {
232        dotenvy::from_path(p.join(".env")).ok();
233    };
234
235    // we only want the .env file of the cwd and project root
236    // `find_project_root` calls `current_dir` internally so both paths are either both `Ok` or
237    // both `Err`
238    if let (Ok(cwd), Ok(prj_root)) = (std::env::current_dir(), find_project_root(None)) {
239        load(&prj_root);
240        if cwd != prj_root {
241            // prj root and cwd can be identical
242            load(&cwd);
243        }
244    };
245}

282/// Useful extensions to [`std::process::Command`].
283pub trait CommandUtils {
284    /// Returns the command's output if execution is successful, otherwise, throws an error.
285    fn exec(&mut self) -> Result<Output>;
286
287    /// Returns the command's stdout if execution is successful, otherwise, throws an error.
288    fn get_stdout_lossy(&mut self) -> Result<String>;
289}
290
291impl CommandUtils for Command {
292    #[track_caller]
293    fn exec(&mut self) -> Result<Output> {
294        trace!(command=?self, "executing");
295
296        let output = self.output()?;
297
298        trace!(code=?output.status.code(), ?output);
299
300        if output.status.success() {
301            Ok(output)
302        } else {
303            let stdout = String::from_utf8_lossy(&output.stdout);
304            let stdout = stdout.trim();
305            let stderr = String::from_utf8_lossy(&output.stderr);
306            let stderr = stderr.trim();
307            let msg = if stdout.is_empty() {
308                stderr.to_string()
309            } else if stderr.is_empty() {
310                stdout.to_string()
311            } else {
312                format!("stdout:\n{stdout}\n\nstderr:\n{stderr}")
313            };
314
315            let mut name = self.get_program().to_string_lossy();
316            if let Some(arg) = self.get_args().next() {
317                let arg = arg.to_string_lossy();
318                if !arg.starts_with('-') {
319                    let name = name.to_mut();
320                    name.push(' ');
321                    name.push_str(&arg);
322                }
323            }
324
325            let mut err = match output.status.code() {
326                Some(code) => format!("{name} exited with code {code}"),
327                None => format!("{name} terminated by a signal"),
328            };
329            if !msg.is_empty() {
330                err.push(':');
331                err.push(if msg.lines().count() == 0 { ' ' } else { '\n' });
332                err.push_str(&msg);
333            }
334            Err(eyre::eyre!(err))
335        }
336    }
337
338    #[track_caller]
339    fn get_stdout_lossy(&mut self) -> Result<String> {
340        let output = self.exec()?;
341        let stdout = String::from_utf8_lossy(&output.stdout);
342        Ok(stdout.trim().into())
343    }
344}

498    pub fn commit(self, msg: &str) -> Result<()> {
499        let output = self
500            .cmd()
501            .args(["commit", "-m", msg])
502            .args(cfg!(any(test, debug_assertions)).then_some("--no-gpg-sign"))
503            .output()?;
504        if !output.status.success() {
505            let stdout = String::from_utf8_lossy(&output.stdout);
506            let stderr = String::from_utf8_lossy(&output.stderr);
507            // ignore "nothing to commit" error
508            let msg = "nothing to commit, working tree clean";
509            if !(stdout.contains(msg) || stderr.contains(msg)) {
510                return Err(eyre::eyre!(
511                    "failed to commit (code={:?}, stdout={:?}, stderr={:?})",
512                    output.status.code(),
513                    stdout.trim(),
514                    stderr.trim()
515                ));
516            }
517        }
518        Ok(())

604    /// Returns a list of tuples of submodule paths and their respective branches.
605    ///
606    /// This function reads the `.gitmodules` file and returns the paths of all submodules that have
607    /// a branch. The paths are relative to the Git::root_of(git.root) and not lib/ directory.
608    ///
609    /// `at` is the dir in which the `.gitmodules` file is located, this is the git root.
610    /// `lib` is name of the directory where the submodules are located.
611    pub fn read_submodules_with_branch(
612        self,
613        at: &Path,
614        lib: &OsStr,
615    ) -> Result<HashMap<PathBuf, String>> {
616        // Read the .gitmodules file
617        let gitmodules = foundry_common::fs::read_to_string(at.join(".gitmodules"))?;
618
619        let paths = SUBMODULE_BRANCH_REGEX
620            .captures_iter(&gitmodules)
621            .map(|cap| {
622                let path_str = cap.get(1).unwrap().as_str();
623                let path = PathBuf::from_str(path_str).unwrap();
624                trace!(path = %path.display(), "unstripped path");
625
626                // Keep only the components that come after the lib directory.
627                // This needs to be done because the lockfile uses paths relative foundry project
628                // root whereas .gitmodules use paths relative to the git root which may not be the
629                // project root. e.g monorepo.
630                // Hence, if path is lib/solady, then `lib/solady` is kept. if path is
631                // packages/contract-bedrock/lib/solady, then `lib/solady` is kept.
632                let lib_pos = path.components().find_position(|c| c.as_os_str() == lib);
633                let path = path
634                    .components()
635                    .skip(lib_pos.map(|(i, _)| i).unwrap_or(0))
636                    .collect::<PathBuf>();
637
638                let branch = cap.get(2).unwrap().as_str().to_string();
639                (path, branch)
640            })
641            .collect::<HashMap<_, _>>();
642
643        Ok(paths)
644    }

778/// Deserialized `git submodule status lib/dep` output.
779#[derive(Debug, Clone, serde::Serialize, serde::Deserialize, PartialEq, Eq)]
780pub struct Submodule {
781    /// Current commit hash the submodule is checked out at.
782    rev: String,
783    /// Relative path to the submodule.
784    path: PathBuf,
785}
786
787impl Submodule {
788    pub fn new(rev: String, path: PathBuf) -> Self {
789        Self { rev, path }
790    }
791
792    pub fn rev(&self) -> &str {
793        &self.rev
794    }
795
796    pub fn path(&self) -> &PathBuf {
797        &self.path
798    }
799}
800
801impl FromStr for Submodule {
802    type Err = eyre::Report;
803
804    fn from_str(s: &str) -> Result<Self> {
805        let caps = SUBMODULE_STATUS_REGEX
806            .captures(s)
807            .ok_or_else(|| eyre::eyre!("Invalid submodule status format"))?;
808
809        Ok(Self {
810            rev: caps.get(1).unwrap().as_str().to_string(),
811            path: PathBuf::from(caps.get(2).unwrap().as_str()),
812        })
813    }
814}

816/// Deserialized `git submodule status` output.
817#[derive(Debug, Clone, PartialEq, Eq)]
818pub struct Submodules(pub Vec<Submodule>);
819
820impl Submodules {
821    pub fn len(&self) -> usize {
822        self.0.len()
823    }
824
825    pub fn is_empty(&self) -> bool {
826        self.0.is_empty()
827    }
828}
829
830impl FromStr for Submodules {
831    type Err = eyre::Report;
832
833    fn from_str(s: &str) -> Result<Self> {
834        let subs = s.lines().map(str::parse).collect::<Result<Vec<Submodule>>>()?;
835        Ok(Self(subs))
836    }
837}

839impl<'a> IntoIterator for &'a Submodules {
840    type Item = &'a Submodule;
841    type IntoIter = std::slice::Iter<'a, Submodule>;
842
843    fn into_iter(self) -> Self::IntoIter {
844        self.0.iter()
845    }
