57/// Represents an executed transaction (transacted on the DB)
58#[derive(Debug)]
59pub struct ExecutedTransaction {
60    transaction: Arc<PoolTransaction>,
61    exit_reason: InstructionResult,
62    out: Option<Output>,
63    gas_used: u64,
64    logs: Vec<Log>,
65    traces: Vec<CallTraceNode>,
66    nonce: u64,
67}

71impl ExecutedTransaction {
72    /// Creates the receipt for the transaction
73    fn create_receipt(&self, cumulative_gas_used: &mut u64) -> TypedReceipt {
74        let logs = self.logs.clone();
75        *cumulative_gas_used = cumulative_gas_used.saturating_add(self.gas_used);
76
77        // successful return see [Return]
78        let status_code = u8::from(self.exit_reason as u8 <= InstructionResult::SelfDestruct as u8);
79        let receipt_with_bloom: ReceiptWithBloom = Receipt {
80            status: (status_code == 1).into(),
81            cumulative_gas_used: *cumulative_gas_used,
82            logs,
83        }
84        .into();
85
86        match &self.transaction.pending_transaction.transaction.transaction {
87            TypedTransaction::Legacy(_) => TypedReceipt::Legacy(receipt_with_bloom),
88            TypedTransaction::EIP2930(_) => TypedReceipt::EIP2930(receipt_with_bloom),
89            TypedTransaction::EIP1559(_) => TypedReceipt::EIP1559(receipt_with_bloom),
90            TypedTransaction::EIP4844(_) => TypedReceipt::EIP4844(receipt_with_bloom),
91            TypedTransaction::EIP7702(_) => TypedReceipt::EIP7702(receipt_with_bloom),
92            TypedTransaction::Deposit(_tx) => {
93                TypedReceipt::Deposit(op_alloy_consensus::OpDepositReceiptWithBloom {
94                    receipt: op_alloy_consensus::OpDepositReceipt {
95                        inner: receipt_with_bloom.receipt,
96                        deposit_nonce: Some(0),
97                        deposit_receipt_version: Some(1),
98                    },
99                    logs_bloom: receipt_with_bloom.logs_bloom,
100                })
101            }
102        }
103    }
104}

146impl<DB: Db + ?Sized, V: TransactionValidator> TransactionExecutor<'_, DB, V> {
147    /// Executes all transactions and puts them in a new block with the provided `timestamp`
148    pub fn execute(mut self) -> ExecutedTransactions {
149        let mut transactions = Vec::new();
150        let mut transaction_infos = Vec::new();
151        let mut receipts = Vec::new();
152        let mut bloom = Bloom::default();
153        let mut cumulative_gas_used = 0u64;
154        let mut invalid = Vec::new();
155        let mut included = Vec::new();
156        let gas_limit = self.block_env.gas_limit;
157        let parent_hash = self.parent_hash;
158        let block_number = self.block_env.number;
159        let difficulty = self.block_env.difficulty;
160        let mix_hash = self.block_env.prevrandao;
161        let beneficiary = self.block_env.beneficiary;
162        let timestamp = self.block_env.timestamp;
163        let base_fee = if self.cfg_env.spec.is_enabled_in(SpecId::LONDON) {
164            Some(self.block_env.basefee)
165        } else {
166            None
167        };
168
169        let is_shanghai = self.cfg_env.spec >= SpecId::SHANGHAI;
170        let is_cancun = self.cfg_env.spec >= SpecId::CANCUN;
171        let is_prague = self.cfg_env.spec >= SpecId::PRAGUE;
172        let excess_blob_gas = if is_cancun { self.block_env.blob_excess_gas() } else { None };
173        let mut cumulative_blob_gas_used = if is_cancun { Some(0u64) } else { None };
174
175        for tx in self.into_iter() {
176            let tx = match tx {
177                TransactionExecutionOutcome::Executed(tx) => {
178                    included.push(tx.transaction.clone());
179                    tx
180                }
181                TransactionExecutionOutcome::BlockGasExhausted(tx) => {
182                    trace!(target: "backend",  tx_gas_limit = %tx.pending_transaction.transaction.gas_limit(), ?tx,  "block gas limit exhausting, skipping transaction");
183                    continue;
184                }
185                TransactionExecutionOutcome::BlobGasExhausted(tx) => {
186                    trace!(target: "backend",  blob_gas = %tx.pending_transaction.transaction.blob_gas().unwrap_or_default(), ?tx,  "block blob gas limit exhausting, skipping transaction");
187                    continue;
188                }
189                TransactionExecutionOutcome::TransactionGasExhausted(tx) => {
190                    trace!(target: "backend",  tx_gas_limit = %tx.pending_transaction.transaction.gas_limit(), ?tx,  "transaction gas limit exhausting, skipping transaction");
191                    continue;
192                }
193                TransactionExecutionOutcome::Invalid(tx, _) => {
194                    trace!(target: "backend", ?tx,  "skipping invalid transaction");
195                    invalid.push(tx);
196                    continue;
197                }
198                TransactionExecutionOutcome::DatabaseError(_, err) => {
199                    // Note: this is only possible in forking mode, if for example a rpc request
200                    // failed
201                    trace!(target: "backend", ?err,  "Failed to execute transaction due to database error");
202                    continue;
203                }
204            };
205            if is_cancun {
206                let tx_blob_gas = tx
207                    .transaction
208                    .pending_transaction
209                    .transaction
210                    .transaction
211                    .blob_gas()
212                    .unwrap_or(0);
213                cumulative_blob_gas_used =
214                    Some(cumulative_blob_gas_used.unwrap_or(0u64).saturating_add(tx_blob_gas));
215            }
216            let receipt = tx.create_receipt(&mut cumulative_gas_used);
217
218            let ExecutedTransaction { transaction, logs, out, traces, exit_reason: exit, .. } = tx;
219            build_logs_bloom(&logs, &mut bloom);
220
221            let contract_address = out.as_ref().and_then(|out| {
222                if let Output::Create(_, contract_address) = out {
223                    trace!(target: "backend", "New contract deployed: at {:?}", contract_address);
224                    *contract_address
225                } else {
226                    None
227                }
228            });
229
230            let transaction_index = transaction_infos.len() as u64;
231            let info = TransactionInfo {
232                transaction_hash: transaction.hash(),
233                transaction_index,
234                from: *transaction.pending_transaction.sender(),
235                to: transaction.pending_transaction.transaction.to(),
236                contract_address,
237                traces,
238                exit,
239                out: out.map(Output::into_data),
240                nonce: tx.nonce,
241                gas_used: tx.gas_used,
242            };
243
244            transaction_infos.push(info);
245            receipts.push(receipt);
246            transactions.push(transaction.pending_transaction.transaction.clone());
247        }
248
249        let receipts_root = calculate_receipt_root(&receipts);
250
251        let header = Header {
252            parent_hash,
253            ommers_hash: Default::default(),
254            beneficiary,
255            state_root: self.db.maybe_state_root().unwrap_or_default(),
256            transactions_root: Default::default(), // Will be computed by create_block
257            receipts_root,
258            logs_bloom: bloom,
259            difficulty,
260            number: block_number.saturating_to(),
261            gas_limit,
262            gas_used: cumulative_gas_used,
263            timestamp: timestamp.saturating_to(),
264            extra_data: Default::default(),
265            mix_hash: mix_hash.unwrap_or_default(),
266            nonce: Default::default(),
267            base_fee_per_gas: base_fee,
268            parent_beacon_block_root: is_cancun.then_some(Default::default()),
269            blob_gas_used: cumulative_blob_gas_used,
270            excess_blob_gas,
271            withdrawals_root: is_shanghai.then_some(EMPTY_WITHDRAWALS),
272            requests_hash: is_prague.then_some(EMPTY_REQUESTS_HASH),
273        };
274
275        let block = create_block(header, transactions);
276        let block = BlockInfo { block, transactions: transaction_infos, receipts };
277        ExecutedTransactions { block, included, invalid }
278    }

280    fn env_for(&self, tx: &PendingTransaction) -> Env {
281        let mut tx_env: OpTransaction<TxEnv> =
282            FromRecoveredTx::from_recovered_tx(&tx.transaction.transaction, *tx.sender());
283
284        if let TypedTransaction::EIP7702(tx_7702) = &tx.transaction.transaction
285            && self.cheats.has_recover_overrides()
286        {
287            // Override invalid recovered authorizations with signature overrides from cheat manager
288            let cheated_auths = tx_7702
289                .tx()
290                .authorization_list
291                .iter()
292                .zip(tx_env.base.authorization_list)
293                .map(|(signed_auth, either_auth)| {
294                    either_auth.right_and_then(|recovered_auth| {
295                        if recovered_auth.authority().is_none()
296                            && let Ok(signature) = signed_auth.signature()
297                            && let Some(override_addr) =
298                                self.cheats.get_recover_override(&signature.as_bytes().into())
299                        {
300                            Either::Right(RecoveredAuthorization::new_unchecked(
301                                recovered_auth.into_parts().0,
302                                RecoveredAuthority::Valid(override_addr),
303                            ))
304                        } else {
305                            Either::Right(recovered_auth)
306                        }
307                    })
308                })
309                .collect();
310            tx_env.base.authorization_list = cheated_auths;
311        }
312
313        if self.networks.is_optimism() {
314            tx_env.enveloped_tx = Some(alloy_rlp::encode(&tx.transaction.transaction).into());
315        }
316
317        Env::new(self.cfg_env.clone(), self.block_env.clone(), tx_env, self.networks)
318    }
319}

321/// Represents the result of a single transaction execution attempt
322#[derive(Debug)]
323pub enum TransactionExecutionOutcome {
324    /// Transaction successfully executed
325    Executed(ExecutedTransaction),
326    /// Invalid transaction not executed
327    Invalid(Arc<PoolTransaction>, InvalidTransactionError),
328    /// Execution skipped because could exceed block gas limit
329    BlockGasExhausted(Arc<PoolTransaction>),
330    /// Execution skipped because it exceeded the blob gas limit
331    BlobGasExhausted(Arc<PoolTransaction>),
332    /// Execution skipped because it exceeded the transaction gas limit
333    TransactionGasExhausted(Arc<PoolTransaction>),
334    /// When an error occurred during execution
335    DatabaseError(Arc<PoolTransaction>, DatabaseError),
336}
337
338impl<DB: Db + ?Sized, V: TransactionValidator> Iterator for &mut TransactionExecutor<'_, DB, V> {
339    type Item = TransactionExecutionOutcome;
340
341    fn next(&mut self) -> Option<Self::Item> {
342        let transaction = self.pending.next()?;
343        let sender = *transaction.pending_transaction.sender();
344        let account = match self.db.basic(sender).map(|acc| acc.unwrap_or_default()) {
345            Ok(account) => account,
346            Err(err) => return Some(TransactionExecutionOutcome::DatabaseError(transaction, err)),
347        };
348        let env = self.env_for(&transaction.pending_transaction);
349
350        // check that we comply with the block's gas limit, if not disabled
351        let max_block_gas = self.gas_used.saturating_add(env.tx.base.gas_limit);
352        if !env.evm_env.cfg_env.disable_block_gas_limit
353            && max_block_gas > env.evm_env.block_env.gas_limit
354        {
355            return Some(TransactionExecutionOutcome::BlockGasExhausted(transaction));
356        }
357
358        // check that we comply with the transaction's gas limit as imposed by Osaka (EIP-7825)
359        if env.evm_env.cfg_env.tx_gas_limit_cap.is_none()
360            && transaction.pending_transaction.transaction.gas_limit()
361                > env.evm_env.cfg_env().tx_gas_limit_cap()
362        {
363            return Some(TransactionExecutionOutcome::TransactionGasExhausted(transaction));
364        }
365
366        // check that we comply with the block's blob gas limit
367        let max_blob_gas = self.blob_gas_used.saturating_add(
368            transaction.pending_transaction.transaction.transaction.blob_gas().unwrap_or(0),
369        );
370        if max_blob_gas > self.blob_params.max_blob_gas_per_block() {
371            return Some(TransactionExecutionOutcome::BlobGasExhausted(transaction));
372        }
373
374        // validate before executing
375        if let Err(err) = self.validator.validate_pool_transaction_for(
376            &transaction.pending_transaction,
377            &account,
378            &env,
379        ) {
380            warn!(target: "backend", "Skipping invalid tx execution [{:?}] {}", transaction.hash(), err);
381            return Some(TransactionExecutionOutcome::Invalid(transaction, err));
382        }
383
384        let nonce = account.nonce;
385
386        let mut inspector = AnvilInspector::default().with_tracing();
387        if self.enable_steps_tracing {
388            inspector = inspector.with_steps_tracing();
389        }
390        if self.print_logs {
391            inspector = inspector.with_log_collector();
392        }
393        if self.print_traces {
394            inspector = inspector.with_trace_printer();
395        }
396
397        let exec_result = {
398            let mut evm = new_evm_with_inspector(&mut *self.db, &env, &mut inspector);
399            self.networks.inject_precompiles(evm.precompiles_mut());
400
401            if let Some(factory) = &self.precompile_factory {
402                evm.precompiles_mut().extend_precompiles(factory.precompiles());
403            }
404
405            let cheats = Arc::new(self.cheats.clone());
406            if cheats.has_recover_overrides() {
407                let cheat_ecrecover = CheatEcrecover::new(Arc::clone(&cheats));
408                evm.precompiles_mut().apply_precompile(&EC_RECOVER, move |_| {
409                    Some(DynPrecompile::new_stateful(
410                        cheat_ecrecover.precompile_id().clone(),
411                        move |input| cheat_ecrecover.call(input),
412                    ))
413                });
414            }
415
416            trace!(target: "backend", "[{:?}] executing", transaction.hash());
417            // transact and commit the transaction
418            match evm.transact_commit(env.tx) {
419                Ok(exec_result) => exec_result,
420                Err(err) => {
421                    warn!(target: "backend", "[{:?}] failed to execute: {:?}", transaction.hash(), err);
422                    match err {
423                        EVMError::Database(err) => {
424                            return Some(TransactionExecutionOutcome::DatabaseError(
425                                transaction,
426                                err,
427                            ));
428                        }
429                        EVMError::Transaction(err) => {
430                            return Some(TransactionExecutionOutcome::Invalid(
431                                transaction,
432                                err.into(),
433                            ));
434                        }
435                        // This will correspond to prevrandao not set, and it should never happen.
436                        // If it does, it's a bug.
437                        e => panic!("failed to execute transaction: {e}"),
438                    }
439                }
440            }
441        };
442
443        if self.print_traces {
444            inspector.print_traces(self.call_trace_decoder.clone());
445        }
446        inspector.print_logs();
447
448        let (exit_reason, gas_used, out, logs) = match exec_result {
449            ExecutionResult::Success { reason, gas_used, logs, output, .. } => {
450                (reason.into(), gas_used, Some(output), Some(logs))
451            }
452            ExecutionResult::Revert { gas_used, output } => {
453                (InstructionResult::Revert, gas_used, Some(Output::Call(output)), None)
454            }
455            ExecutionResult::Halt { reason, gas_used } => {
456                (op_haltreason_to_instruction_result(reason), gas_used, None, None)
457            }
458        };
459
460        if exit_reason == InstructionResult::OutOfGas {
461            // this currently useful for debugging estimations
462            warn!(target: "backend", "[{:?}] executed with out of gas", transaction.hash())
463        }
464
465        trace!(target: "backend", ?exit_reason, ?gas_used, "[{:?}] executed with out={:?}", transaction.hash(), out);
466
467        // Track the total gas used for total gas per block checks
468        self.gas_used = self.gas_used.saturating_add(gas_used);
469
470        // Track the total blob gas used for total blob gas per blob checks
471        if let Some(blob_gas) = transaction.pending_transaction.transaction.transaction.blob_gas() {
472            self.blob_gas_used = self.blob_gas_used.saturating_add(blob_gas);
473        }
474
475        trace!(target: "backend::executor", "transacted [{:?}], result: {:?} gas {}", transaction.hash(), exit_reason, gas_used);
476
477        let tx = ExecutedTransaction {
478            transaction,
479            exit_reason,
480            out,
481            gas_used,
482            logs: logs.unwrap_or_default(),
483            traces: inspector.tracer.map(|t| t.into_traces().into_nodes()).unwrap_or_default(),
484            nonce,
485        };
486
487        Some(TransactionExecutionOutcome::Executed(tx))
488    }
489}

491/// Inserts all logs into the bloom
492fn build_logs_bloom(logs: &[Log], bloom: &mut Bloom) {
493    for log in logs {
494        bloom.accrue(BloomInput::Raw(&log.address[..]));
495        for topic in log.topics() {
496            bloom.accrue(BloomInput::Raw(&topic[..]));
497        }
498    }
499}

501/// Creates a database with given database and inspector.
502pub fn new_evm_with_inspector<DB, I>(
503    db: DB,
504    env: &Env,
505    inspector: I,
506) -> EitherEvm<DB, I, PrecompilesMap>
507where
508    DB: Database<Error = DatabaseError> + Debug,
509    I: Inspector<EthEvmContext<DB>> + Inspector<OpContext<DB>>,
510{
511    if env.networks.is_optimism() {
512        let op_cfg = env.evm_env.cfg_env.clone().with_spec(op_revm::OpSpecId::ISTHMUS);
513        let op_context = OpContext {
514            journaled_state: {
515                let mut journal = Journal::new(db);
516                // Converting SpecId into OpSpecId
517                journal.set_spec_id(env.evm_env.cfg_env.spec);
518                journal
519            },
520            block: env.evm_env.block_env.clone(),
521            cfg: op_cfg.clone(),
522            tx: env.tx.clone(),
523            chain: L1BlockInfo::default(),
524            local: LocalContext::default(),
525            error: Ok(()),
526        };
527
528        let op_precompiles = OpPrecompiles::new_with_spec(op_cfg.spec).precompiles();
529        let op_evm = op_revm::OpEvm(RevmEvm::new_with_inspector(
530            op_context,
531            inspector,
532            EthInstructions::default(),
533            PrecompilesMap::from_static(op_precompiles),
534        ));
535
536        let op = OpEvm::new(op_evm, true);
537
538        EitherEvm::Op(op)
539    } else {
540        let spec = env.evm_env.cfg_env.spec;
541        let eth_context = EthEvmContext {
542            journaled_state: {
543                let mut journal = Journal::new(db);
544                journal.set_spec_id(spec);
545                journal
546            },
547            block: env.evm_env.block_env.clone(),
548            cfg: env.evm_env.cfg_env.clone(),
549            tx: env.tx.base.clone(),
550            chain: (),
551            local: LocalContext::default(),
552            error: Ok(()),
553        };
554
555        let eth_precompiles = EthPrecompiles {
556            precompiles: Precompiles::new(PrecompileSpecId::from_spec_id(spec)),
557            spec,
558        }
559        .precompiles;
560        let eth_evm = RevmEvm::new_with_inspector(
561            eth_context,
562            inspector,
563            EthInstructions::default(),
564            PrecompilesMap::from_static(eth_precompiles),
565        );
566
567        let eth = EthEvm::new(eth_evm, true);
568
569        EitherEvm::Eth(eth)
570    }
571}
