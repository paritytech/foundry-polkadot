47/// An extension trait that allows us to add additional hooks to Inspector for later use in
48/// handlers.
49#[auto_impl(&mut, Box)]
50pub trait InspectorExt: for<'a> Inspector<EthEvmContext<&'a mut dyn DatabaseExt>> {
51    /// Determines whether the `DEFAULT_CREATE2_DEPLOYER` should be used for a CREATE2 frame.
52    ///
53    /// If this function returns true, we'll replace CREATE2 frame with a CALL frame to CREATE2
54    /// factory.
55    fn should_use_create2_factory(
56        &mut self,
57        _context: &mut EthEvmContext<&mut dyn DatabaseExt>,
58        _inputs: &CreateInputs,
59    ) -> bool {
60        false
61    }
62
63    /// Simulates `console.log` invocation.
64    fn console_log(&mut self, msg: &str) {
65        let _ = msg;
66    }
67
68    /// Returns configured networks.
69    fn get_networks(&self) -> NetworkConfigs {
70        NetworkConfigs::default()
71    }
72
73    /// Returns the CREATE2 deployer address.
74    fn create2_deployer(&self) -> Address {
75        DEFAULT_CREATE2_DEPLOYER
76    }
77
78    fn trace_revive(
79        &mut self,
80        _context: Ecx<'_, '_, '_>,
81        _call_traces: Box<dyn std::any::Any>,
82        _record_top_call: bool,
83    ) {
84    }
85}
86
87impl InspectorExt for NoOpInspector {}

89impl InspectorExt for AccessListInspector {}
