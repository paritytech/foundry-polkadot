118    /// let alloy_provider = ProviderBuilder::<_,_, AnyNetwork>::default().connect("http://localhost:8545").await?;;
119    /// let to = Address::from_str("0xB3C95ff08316fb2F2e3E52Ee82F8e7b605Aa1304")?;
120    /// let greeting = greetingCall { i: U256::from(5) }.abi_encode();
121    /// let bytes = Bytes::from_iter(greeting.iter());
122    /// let tx = TransactionRequest::default().to(to).input(bytes.into());
123    /// let tx = WithOtherFields::new(tx);
124    ///
125    /// // Create state overrides
126    /// let mut state_override = StateOverride::default();
127    /// let mut account_override = AccountOverride::default();
128    /// account_override.balance = Some(U256::from(1000));
129    /// state_override.insert(to, account_override);
130    /// let state_override_object = StateOverridesBuilder::default().build();
131    /// let block_override_object = BlockOverrides::default();
132    ///
133    /// let cast = Cast::new(alloy_provider);
134    /// let data = cast.call(&tx, None, None, Some(state_override_object), Some(block_override_object)).await?;
135    /// println!("{}", data);
136    /// # Ok(())
137    /// # }
138    /// ```
139    pub async fn call(
140        &self,
141        req: &WithOtherFields<TransactionRequest>,
142        func: Option<&Function>,
143        block: Option<BlockId>,
144        state_override: Option<StateOverride>,
145        block_override: Option<BlockOverrides>,
146    ) -> Result<String> {
147        let mut call = self
148            .provider
149            .call(req.clone())
150            .block(block.unwrap_or_default())
151            .with_block_overrides_opt(block_override);
152        if let Some(state_override) = state_override {
153            call = call.overrides(state_override)
154        }
155
156        let res = call.await?;
157        let mut decoded = vec![];
158
159        if let Some(func) = func {
160            // decode args into tokens
161            decoded = match func.abi_decode_output(res.as_ref()) {
162                Ok(decoded) => decoded,
163                Err(err) => {
164                    // ensure the address is a contract
165                    if res.is_empty() {
166                        // check that the recipient is a contract that can be called
167                        if let Some(TxKind::Call(addr)) = req.to {
168                            if let Ok(code) = self
169                                .provider
170                                .get_code_at(addr)
171                                .block_id(block.unwrap_or_default())
172                                .await
173                                && code.is_empty()
174                            {
175                                eyre::bail!("contract {addr:?} does not have any code")
176                            }
177                        } else if Some(TxKind::Create) == req.to {
178                            eyre::bail!("tx req is a contract deployment");
179                        } else {
180                            eyre::bail!("recipient is None");
181                        }
182                    }
183                    return Err(err).wrap_err(
184                        "could not decode output; did you specify the wrong function return data type?"
185                    );
186                }
187            };
188        }
189
190        // handle case when return type is not specified
191        Ok(if decoded.is_empty() {
192            res.to_string()
193        } else if shell::is_json() {
194            let tokens = decoded
195                .into_iter()
196                .map(|value| serialize_value_as_json(value, None))
197                .collect::<eyre::Result<Vec<_>>>()?;
198            serde_json::to_string_pretty(&tokens).unwrap()
199        } else {
200            // seth compatible user-friendly return type conversions
201            decoded.iter().map(format_token).collect::<Vec<_>>().join("\n")
202        })
203    }

235    pub async fn access_list(
236        &self,
237        req: &WithOtherFields<TransactionRequest>,
238        block: Option<BlockId>,
239    ) -> Result<String> {
240        let access_list =
241            self.provider.create_access_list(req).block_id(block.unwrap_or_default()).await?;
242        let res = if shell::is_json() {
243            serde_json::to_string(&access_list)?
244        } else {
245            let mut s =
246                vec![format!("gas used: {}", access_list.gas_used), "access list:".to_string()];
247            for al in access_list.access_list.0 {
248                s.push(format!("- address: {}", &al.address.to_checksum(None)));
249                if !al.storage_keys.is_empty() {
250                    s.push("  keys:".to_string());
251                    for key in al.storage_keys {
252                        s.push(format!("    {key:?}"));
253                    }
254                }
255            }
256            s.join("\n")
257        };
258
259        Ok(res)
260    }

376    pub async fn block<B: Into<BlockId>>(
377        &self,
378        block: B,
379        full: bool,
380        fields: Vec<String>,
381        raw: bool,
382    ) -> Result<String> {
383        let block = block.into();
384        if fields.contains(&"transactions".into()) && !full {
385            eyre::bail!("use --full to view transactions")
386        }
387
388        let block = self
389            .provider
390            .get_block(block)
391            .kind(full.into())
392            .await?
393            .ok_or_else(|| eyre::eyre!("block {:?} not found", block))?;
394
395        Ok(if raw {
396            let header: Header = block.into_inner().header.inner.try_into_header()?;
397            format!("0x{}", hex::encode(alloy_rlp::encode(&header)))
398        } else if !fields.is_empty() {
399            let mut result = String::new();
400            for field in fields {
401                result.push_str(
402                    &get_pretty_block_attr(&block, &field)
403                        .unwrap_or_else(|| format!("{field} is not a valid block field")),
404                );
405
406                result.push('\n');
407            }
408            result.trim_end().to_string()
409        } else if shell::is_json() {
410            serde_json::to_value(&block).unwrap().to_string()
411        } else {
412            block.pretty()
413        })
414    }

809        nonce: Option<u64>,
810        field: Option<String>,
811        raw: bool,
812        to_request: bool,
813    ) -> Result<String> {
814        let tx = if let Some(tx_hash) = tx_hash {
815            let tx_hash = TxHash::from_str(&tx_hash).wrap_err("invalid tx hash")?;
816            self.provider
817                .get_transaction_by_hash(tx_hash)
818                .await?
819                .ok_or_else(|| eyre::eyre!("tx not found: {:?}", tx_hash))?
820        } else if let Some(from) = from {
821            // If nonce is not provided, uses 0.
822            let nonce = U64::from(nonce.unwrap_or_default());
823            let from = from.resolve(self.provider.root()).await?;
824
825            self.provider
826                .raw_request::<_, Option<AnyRpcTransaction>>(
827                    "eth_getTransactionBySenderAndNonce".into(),
828                    (from, nonce),
829                )
830                .await?
831                .ok_or_else(|| {
832                    eyre::eyre!("tx not found for sender {from} and nonce {:?}", nonce.to::<u64>())
833                })?
834        } else {
835            eyre::bail!("tx hash or from address is required")
836        };
837
838        Ok(if raw {
839            // also consider opstack deposit transactions
840            let either_tx = tx.try_into_either::<OpTxEnvelope>()?;
841            let encoded = either_tx.encoded_2718();
842            format!("0x{}", hex::encode(encoded))
843        } else if let Some(field) = field {
844            get_pretty_tx_attr(&tx.inner, field.as_str())
845                .ok_or_else(|| eyre::eyre!("invalid tx field: {}", field.to_string()))?
846        } else if shell::is_json() {
847            // to_value first to sort json object keys
848            serde_json::to_value(&tx)?.to_string()
849        } else if to_request {
850            serde_json::to_string_pretty(&TransactionRequest::from_recovered_transaction(
851                tx.into(),
852            ))?
853        } else {
854            tx.pretty()
855        })
856    }

874    pub async fn receipt(
875        &self,
876        tx_hash: String,
877        field: Option<String>,
878        confs: u64,
879        timeout: Option<u64>,
880        cast_async: bool,
881    ) -> Result<String> {
882        let tx_hash = TxHash::from_str(&tx_hash).wrap_err("invalid tx hash")?;
883
884        let mut receipt: TransactionReceiptWithRevertReason =
885            match self.provider.get_transaction_receipt(tx_hash).await? {
886                Some(r) => r,
887                None => {
888                    // if the async flag is provided, immediately exit if no tx is found, otherwise
889                    // try to poll for it
890                    if cast_async {
891                        eyre::bail!("tx not found: {:?}", tx_hash)
892                    } else {
893                        PendingTransactionBuilder::new(self.provider.root().clone(), tx_hash)
894                            .with_required_confirmations(confs)
895                            .with_timeout(timeout.map(Duration::from_secs))
896                            .get_receipt()
897                            .await?
898                    }
899                }
900            }
901            .into();
902
903        // Allow to fail silently
904        let _ = receipt.update_revert_reason(&self.provider).await;
905
906        self.format_receipt(receipt, field)
907    }

1047    /// Sets up a subscription to the given filter and writes the logs to the given output.
1048    ///
1049    /// # Example
1050    ///
1051    /// ```
1052    /// use alloy_primitives::Address;
1053    /// use alloy_provider::{ProviderBuilder, RootProvider, network::AnyNetwork};
1054    /// use alloy_rpc_types::Filter;
1055    /// use alloy_transport::BoxTransport;
1056    /// use cast::Cast;
1057    /// use std::{io, str::FromStr};
1058    ///
1059    /// # async fn foo() -> eyre::Result<()> {
1060    /// let provider =
1061    ///     ProviderBuilder::<_, _, AnyNetwork>::default().connect("wss://localhost:8545").await?;
1062    /// let cast = Cast::new(provider);
1063    ///
1064    /// let filter =
1065    ///     Filter::new().address(Address::from_str("0x00000000006c3852cbEf3e08E8dF289169EdE581")?);
1066    /// let mut output = io::stdout();
1067    /// cast.subscribe(filter, &mut output).await?;
1068    /// # Ok(())
1069    /// # }
1070    /// ```
1071    pub async fn subscribe(&self, filter: Filter, output: &mut dyn io::Write) -> Result<()> {
1072        // Initialize the subscription stream for logs
1073        let mut subscription = self.provider.subscribe_logs(&filter).await?.into_stream();
1074
1075        // Check if a to_block is specified, if so, subscribe to blocks
1076        let mut block_subscription = if filter.get_to_block().is_some() {
1077            Some(self.provider.subscribe_blocks().await?.into_stream())
1078        } else {
1079            None
1080        };
1081
1082        let format_json = shell::is_json();
1083        let to_block_number = filter.get_to_block();
1084
1085        // If output should be JSON, start with an opening bracket
1086        if format_json {
1087            write!(output, "[")?;
1088        }
1089
1090        let mut first = true;
1091
1092        loop {
1093            tokio::select! {
1094                // If block subscription is present, listen to it to avoid blocking indefinitely past the desired to_block
1095                block = if let Some(bs) = &mut block_subscription {
1096                    Either::Left(bs.next().fuse())
1097                } else {
1098                    Either::Right(futures::future::pending())
1099                } => {
1100                    if let (Some(block), Some(to_block)) = (block, to_block_number)
1101                        && block.number  > to_block {
1102                            break;
1103                        }
1104                },
1105                // Process incoming log
1106                log = subscription.next() => {
1107                    if format_json {
1108                        if !first {
1109                            write!(output, ",")?;
1110                        }
1111                        first = false;
1112                        let log_str = serde_json::to_string(&log).unwrap();
1113                        write!(output, "{log_str}")?;
1114                    } else {
1115                        let log_str = log.pretty()
1116                            .replacen('\n', "- ", 1)  // Remove empty first line
1117                            .replace('\n', "\n  ");  // Indent
1118                        writeln!(output, "{log_str}")?;
1119                    }
1120                },
1121                // Break on cancel signal, to allow for closing JSON bracket
1122                _ = ctrl_c() => {
1123                    break;
1124                },
1125                else => break,
1126            }
1127        }
1128
1129        // If output was JSON, end with a closing bracket
1130        if format_json {
1131            write!(output, "]")?;
1132        }
1133
1134        Ok(())
1135    }

1296            let sign = if number.is_nonnegative() { "" } else { "-" };
1297            let value = format!("{number:#}");

1860    /// Performs ABI encoding of an event to produce the topics and data.
1861    ///
1862    /// # Example
1863    ///
1864    /// ```
1865    /// use alloy_primitives::hex;
1866    /// use cast::SimpleCast as Cast;
1867    ///
1868    /// let log_data = Cast::abi_encode_event(
1869    ///     "Transfer(address indexed from, address indexed to, uint256 value)",
1870    ///     &[
1871    ///         "0x1234567890123456789012345678901234567890",
1872    ///         "0xabcdefabcdefabcdefabcdefabcdefabcdefabcd",
1873    ///         "1000",
1874    ///     ],
1875    /// )
1876    /// .unwrap();
1877    ///
1878    /// // topic0 is the event selector
1879    /// assert_eq!(log_data.topics().len(), 3);
1880    /// assert_eq!(
1881    ///     log_data.topics()[0].to_string(),
1882    ///     "0xddf252ad1be2c89b69c2b068fc378daa952ba7f163c4a11628f55a4df523b3ef"
1883    /// );
1884    /// assert_eq!(
1885    ///     log_data.topics()[1].to_string(),
1886    ///     "0x0000000000000000000000001234567890123456789012345678901234567890"
1887    /// );
1888    /// assert_eq!(
1889    ///     log_data.topics()[2].to_string(),
1890    ///     "0x000000000000000000000000abcdefabcdefabcdefabcdefabcdefabcdefabcd"
1891    /// );
1892    /// assert_eq!(
1893    ///     hex::encode_prefixed(log_data.data),
1894    ///     "0x00000000000000000000000000000000000000000000000000000000000003e8"
1895    /// );
1896    /// # Ok::<_, eyre::Report>(())
1897    /// ```
1898    pub fn abi_encode_event(sig: &str, args: &[impl AsRef<str>]) -> Result<LogData> {
1899        let event = get_event(sig)?;
1900        let tokens = std::iter::zip(&event.inputs, args)
1901            .map(|(input, arg)| coerce_value(&input.ty, arg.as_ref()))
1902            .collect::<Result<Vec<_>>>()?;
1903
1904        let mut topics = vec![event.selector()];
1905        let mut data_tokens: Vec<u8> = Vec::new();
1906
1907        for (input, token) in event.inputs.iter().zip(tokens) {
1908            if input.indexed {
1909                let ty = DynSolType::parse(&input.ty)?;
1910                if matches!(
1911                    ty,
1912                    DynSolType::String
1913                        | DynSolType::Bytes
1914                        | DynSolType::Array(_)
1915                        | DynSolType::Tuple(_)
1916                ) {
1917                    // For dynamic types, hash the encoded value
1918                    let encoded = token.abi_encode();
1919                    let hash = keccak256(encoded);
1920                    topics.push(hash);
1921                } else {
1922                    // For fixed-size types, encode directly to 32 bytes
1923                    let mut encoded = [0u8; 32];
1924                    let token_encoded = token.abi_encode();
1925                    if token_encoded.len() <= 32 {
1926                        let start = 32 - token_encoded.len();
1927                        encoded[start..].copy_from_slice(&token_encoded);
1928                    }
1929                    topics.push(B256::from(encoded));
1930                }
1931            } else {
1932                // Non-indexed parameters go into data
1933                data_tokens.extend_from_slice(&token.abi_encode());
1934            }
1935        }
1936
1937        Ok(LogData::new_unchecked(topics, data_tokens.into()))
1938    }

1959    /// Returns the slot number for a given mapping key and slot.
1960    ///
1961    /// Given `mapping(k => v) m`, for a key `k` the slot number of its associated `v` is
1962    /// `keccak256(concat(h(k), p))`, where `h` is the padding function for `k`'s type, and `p`
1963    /// is slot number of the mapping `m`.
1964    ///
1965    /// See [the Solidity documentation](https://docs.soliditylang.org/en/latest/internals/layout_in_storage.html#mappings-and-dynamic-arrays)
1966    /// for more details.
1967    ///
1968    /// # Example
1969    ///
1970    /// ```
1971    /// # use cast::SimpleCast as Cast;
1972    ///
1973    /// // Value types.
1974    /// assert_eq!(
1975    ///     Cast::index("address", "0xD0074F4E6490ae3f888d1d4f7E3E43326bD3f0f5", "2").unwrap().as_str(),
1976    ///     "0x9525a448a9000053a4d151336329d6563b7e80b24f8e628e95527f218e8ab5fb"
1977    /// );
1978    /// assert_eq!(
1979    ///     Cast::index("uint256", "42", "6").unwrap().as_str(),
1980    ///     "0xfc808b0f31a1e6b9cf25ff6289feae9b51017b392cc8e25620a94a38dcdafcc1"
1981    /// );
1982    ///
1983    /// // Strings and byte arrays.
1984    /// assert_eq!(
1985    ///     Cast::index("string", "hello", "1").unwrap().as_str(),
1986    ///     "0x8404bb4d805e9ca2bd5dd5c43a107e935c8ec393caa7851b353b3192cd5379ae"
1987    /// );
1988    /// # Ok::<_, eyre::Report>(())
1989    /// ```
1990    pub fn index(key_type: &str, key: &str, slot_number: &str) -> Result<String> {
1991        let mut hasher = Keccak256::new();
1992
1993        let k_ty = DynSolType::parse(key_type).wrap_err("Could not parse type")?;
1994        let k = k_ty.coerce_str(key).wrap_err("Could not parse value")?;
1995        match k_ty {
1996            // For value types, `h` pads the value to 32 bytes in the same way as when storing the
1997            // value in memory.
1998            DynSolType::Bool
1999            | DynSolType::Int(_)
2000            | DynSolType::Uint(_)
2001            | DynSolType::FixedBytes(_)
2002            | DynSolType::Address
2003            | DynSolType::Function => hasher.update(k.as_word().unwrap()),
2004
2005            // For strings and byte arrays, `h(k)` is just the unpadded data.
2006            DynSolType::String | DynSolType::Bytes => hasher.update(k.as_packed_seq().unwrap()),
2007
2008            DynSolType::Array(..)
2009            | DynSolType::FixedArray(..)
2010            | DynSolType::Tuple(..)
2011            | DynSolType::CustomStruct { .. } => {
2012                eyre::bail!("Type `{k_ty}` is not supported as a mapping key")
2013            }
2014        }
2015
2016        let p = DynSolType::Uint(256)
2017            .coerce_str(slot_number)
2018            .wrap_err("Could not parse slot number")?;
2019        let p = p.as_word().unwrap();
2020        hasher.update(p);
2021
2022        let location = hasher.finalize();
2023        Ok(location.to_string())
2024    }

2246    /// Gets the selector for a given function signature
2247    /// Optimizes if the `optimize` parameter is set to a number of leading zeroes
2248    ///
2249    /// # Example
2250    ///
2251    /// ```
2252    /// use cast::SimpleCast as Cast;
2253    ///
2254    /// assert_eq!(Cast::get_selector("foo(address,uint256)", 0)?.0, String::from("0xbd0d639f"));
2255    /// # Ok::<(), eyre::Error>(())
2256    /// ```
2257    pub fn get_selector(signature: &str, optimize: usize) -> Result<(String, String)> {
2258        if optimize > 4 {
2259            eyre::bail!("number of leading zeroes must not be greater than 4");
2260        }
2261        if optimize == 0 {
2262            let selector = get_func(signature)?.selector();
2263            return Ok((selector.to_string(), String::from(signature)));
2264        }
2265        let Some((name, params)) = signature.split_once('(') else {
2266            eyre::bail!("invalid function signature");
2267        };
2268
2269        let num_threads = std::thread::available_parallelism().map_or(1, |n| n.get());
2270        let found = AtomicBool::new(false);
2271
2272        let result: Option<(u32, String, String)> =
2273            (0..num_threads).into_par_iter().find_map_any(|i| {
2274                let nonce_start = i as u32;
2275                let nonce_step = num_threads as u32;
2276
2277                let mut nonce = nonce_start;
2278                while nonce < u32::MAX && !found.load(Ordering::Relaxed) {
2279                    let input = format!("{name}{nonce}({params}");
2280                    let hash = keccak256(input.as_bytes());
2281                    let selector = &hash[..4];
2282
2283                    if selector.iter().take_while(|&&byte| byte == 0).count() == optimize {
2284                        found.store(true, Ordering::Relaxed);
2285                        return Some((nonce, hex::encode_prefixed(selector), input));
2286                    }
2287
2288                    nonce += nonce_step;
2289                }
2290                None
2291            });
2292
2293        match result {
2294            Some((_nonce, selector, signature)) => Ok((selector, signature)),
2295            None => eyre::bail!("No selector found"),
2296        }
2297    }
