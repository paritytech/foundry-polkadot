23/// CLI arguments for `cast send`.
24#[derive(Debug, Parser)]
25pub struct SendTxArgs {
26    /// The destination of the transaction.
27    ///
28    /// If not provided, you must use cast send --create.
29    #[arg(value_parser = NameOrAddress::from_str)]
30    to: Option<NameOrAddress>,
31
32    /// The signature of the function to call.
33    sig: Option<String>,
34
35    /// The arguments of the function to call.
36    #[arg(allow_negative_numbers = true)]
37    args: Vec<String>,
38
39    /// Only print the transaction hash and exit immediately.
40    #[arg(id = "async", long = "async", alias = "cast-async", env = "CAST_ASYNC")]
41    cast_async: bool,
42
43    /// Wait for transaction receipt synchronously instead of polling.
44    /// Note: uses `eth_sendTransactionSync` which may not be supported by all clients.
45    #[arg(long, conflicts_with = "async")]
46    sync: bool,
47
48    /// The number of confirmations until the receipt is fetched.
49    #[arg(long, default_value = "1")]
50    confirmations: u64,
51
52    /// Polling interval for transaction receipts (in seconds).
53    #[arg(long, alias = "poll-interval", env = "ETH_POLL_INTERVAL")]
54    poll_interval: Option<u64>,
55
56    #[command(subcommand)]
57    command: Option<SendTxSubcommands>,
58
59    /// Send via `eth_sendTransaction` using the `--from` argument or $ETH_FROM as sender
60    #[arg(long, requires = "from")]
61    unlocked: bool,
62
63    /// Timeout for sending the transaction.
64    #[arg(long, env = "ETH_TIMEOUT")]
65    pub timeout: Option<u64>,
66
67    #[command(flatten)]
68    tx: TransactionOpts,
69
70    #[command(flatten)]
71    eth: EthereumOpts,
72
73    /// The path of blob data to be sent.
74    #[arg(
75        long,
76        value_name = "BLOB_DATA_PATH",
77        conflicts_with = "legacy",
78        requires = "blob",
79        help_heading = "Transaction options"
80    )]
81    path: Option<PathBuf>,
82}

101impl SendTxArgs {
102    pub async fn run(self) -> eyre::Result<()> {
103        let Self {
104            eth,
105            to,
106            mut sig,
107            cast_async,
108            sync,
109            mut args,
110            tx,
111            confirmations,
112            command,
113            unlocked,
114            path,
115            timeout,
116            poll_interval,
117        } = self;
118
119        let blob_data = if let Some(path) = path { Some(std::fs::read(path)?) } else { None };
120
121        let code = if let Some(SendTxSubcommands::Create {
122            code,
123            sig: constructor_sig,
124            args: constructor_args,
125        }) = command
126        {
127            // ensure we don't violate settings for transactions that can't be CREATE: 7702 and 4844
128            // which require mandatory target
129            if to.is_none() && tx.auth.is_some() {
130                return Err(eyre!(
131                    "EIP-7702 transactions can't be CREATE transactions and require a destination address"
132                ));
133            }
134            // ensure we don't violate settings for transactions that can't be CREATE: 7702 and 4844
135            // which require mandatory target
136            if to.is_none() && blob_data.is_some() {
137                return Err(eyre!(
138                    "EIP-4844 transactions can't be CREATE transactions and require a destination address"
139                ));
140            }
141
142            sig = constructor_sig;
143            args = constructor_args;
144            Some(code)
145        } else {
146            None
147        };
148
149        let config = eth.load_config()?;
150        let provider = utils::get_provider(&config)?;
151
152        if let Some(interval) = poll_interval {
153            provider.client().set_poll_interval(Duration::from_secs(interval))
154        }
155
156        let builder = CastTxBuilder::new(&provider, tx, &config)
157            .await?
158            .with_to(to)
159            .await?
160            .with_code_sig_and_args(code, sig, args)
161            .await?
162            .with_blob_data(blob_data)?;
163
164        let timeout = timeout.unwrap_or(config.transaction_timeout);
165
166        // Case 1:
167        // Default to sending via eth_sendTransaction if the --unlocked flag is passed.
168        // This should be the only way this RPC method is used as it requires a local node
169        // or remote RPC with unlocked accounts.
170        if unlocked && !eth.wallet.browser {
171            // only check current chain id if it was specified in the config
172            if let Some(config_chain) = config.chain {
173                let current_chain_id = provider.get_chain_id().await?;
174                let config_chain_id = config_chain.id();
175                // switch chain if current chain id is not the same as the one specified in the
176                // config
177                if config_chain_id != current_chain_id {
178                    sh_warn!("Switching to chain {}", config_chain)?;
179                    provider
180                        .raw_request::<_, ()>(
181                            "wallet_switchEthereumChain".into(),
182                            [serde_json::json!({
183                                "chainId": format!("0x{:x}", config_chain_id),
184                            })],
185                        )
186                        .await?;
187                }
188            }
189
190            let (tx, _) = builder.build(config.sender).await?;
191
192            cast_send(provider, tx, cast_async, sync, confirmations, timeout).await
193        // Case 2:
194        // An option to use a local signer was provided.
195        // If we cannot successfully instantiate a local signer, then we will assume we don't have
196        // enough information to sign and we must bail.
197        } else {
198            // Retrieve the signer, and bail if it can't be constructed.
199            let signer = eth.wallet.signer().await?;
200            let from = signer.address();
201
202            tx::validate_from_address(eth.wallet.from, from)?;
203
204            // Browser wallets work differently as they sign and send the transaction in one step.
205            if eth.wallet.browser
206                && let WalletSigner::Browser(ref browser_signer) = signer
207            {
208                let (tx_request, _) = builder.build(from).await?;
209                let tx_hash = browser_signer.send_transaction_via_browser(tx_request.inner).await?;
210
211                if cast_async {
212                    sh_println!("{tx_hash:#x}")?;
213                } else {
214                    let receipt = Cast::new(&provider)
215                        .receipt(format!("{tx_hash:#x}"), None, confirmations, Some(timeout), false)
216                        .await?;
217                    sh_println!("{receipt}")?;
218                }
219
220                return Ok(());
221            }
222
223            let (tx_request, _) = builder.build(&signer).await?;
224
225            let wallet = EthereumWallet::from(signer);
226            let provider = ProviderBuilder::<_, _, AnyNetwork>::default()
227                .wallet(wallet)
228                .connect_provider(&provider);
229
230            cast_send(provider, tx_request, cast_async, sync, confirmations, timeout).await
231        }
232    }
233}

235async fn cast_send<P: Provider<AnyNetwork>>(
236    provider: P,
237    tx: WithOtherFields<TransactionRequest>,
238    cast_async: bool,
239    sync: bool,
240    confs: u64,
241    timeout: u64,
242) -> Result<()> {
243    let cast = Cast::new(&provider);
244
245    if sync {
246        // Send transaction and wait for receipt synchronously
247        let receipt = cast.send_sync(tx).await?;
248        sh_println!("{receipt}")?;
249    } else {
250        let pending_tx = cast.send(tx).await?;
251        let tx_hash = pending_tx.inner().tx_hash();
252
253        if cast_async {
254            sh_println!("{tx_hash:#x}")?;
255        } else {
256            let receipt =
257                cast.receipt(format!("{tx_hash:#x}"), None, confs, Some(timeout), false).await?;
258            sh_println!("{receipt}")?;
259        }
260    }
261
262    Ok(())
263}
