14/// Given a parameter type, returns a strategy for generating values for that type.
15///
16/// See [`fuzz_param_with_fixtures`] for more information.

21/// Given a parameter type and configured fixtures for param name, returns a strategy for generating
22/// values for that type.
23///
24/// Fixtures can be currently generated for uint, int, address, bytes and
25/// string types and are defined for parameter name.
26/// For example, fixtures for parameter `owner` of type `address` can be defined in a function with
27/// a `function fixture_owner() public returns (address[] memory)` signature.
28///
29/// Fixtures are matched on parameter name, hence fixtures defined in
30/// `fixture_owner` function can be used in a fuzzed test function with a signature like
31/// `function testFuzz_ownerAddress(address owner, uint amount)`.
32///
33/// Raises an error if all the fixture types are not of the same type as the input parameter.
34///
35/// Works with ABI Encoder v2 tuples.
36pub fn fuzz_param_with_fixtures(
37    param: &DynSolType,
38    fixtures: Option<&[DynSolValue]>,
39    name: &str,
40    max_fuzz_int: Option<U256>,
41) -> BoxedStrategy<DynSolValue> {
42    fuzz_param_inner(param, fixtures.map(|f| (f, name)), max_fuzz_int)
43}

45fn fuzz_param_inner(
46    param: &DynSolType,
47    mut fuzz_fixtures: Option<(&[DynSolValue], &str)>,
48    max_fuzz_int: Option<U256>,
49) -> BoxedStrategy<DynSolValue> {
50    if let Some((fixtures, name)) = fuzz_fixtures
51        && !fixtures.iter().all(|f| f.matches(param))
52    {
53        error!("fixtures for {name:?} do not match type {param}");
54        fuzz_fixtures = None;
55    }
56    let fuzz_fixtures = fuzz_fixtures.map(|(f, _)| f);
57
58    let value = || {
59        let default_strategy = DynSolValue::type_strategy(param);
60        if let Some(fixtures) = fuzz_fixtures {
61            proptest::prop_oneof![
62                50 => {
63                    let fixtures = fixtures.to_vec();
64                    any::<prop::sample::Index>()
65                        .prop_map(move |index| index.get(&fixtures).clone())
66                },
67                50 => default_strategy,
68            ]
69            .boxed()
70        } else {
71            default_strategy.boxed()
72        }
73    };
74
75    match *param {
76        DynSolType::Address => value(),
77        DynSolType::Int(n @ 8..=256) => super::IntStrategy::new(n, fuzz_fixtures, max_fuzz_int)
78            .prop_map(move |x| DynSolValue::Int(x, n))
79            .boxed(),
80        DynSolType::Uint(n @ 8..=256) => super::UintStrategy::new(n, fuzz_fixtures, max_fuzz_int)
81            .prop_map(move |x| DynSolValue::Uint(x, n))
82            .boxed(),
83        DynSolType::Function | DynSolType::Bool => DynSolValue::type_strategy(param).boxed(),
84        DynSolType::Bytes => value(),
85        DynSolType::FixedBytes(_size @ 1..=32) => value(),
86        DynSolType::String => value()
87            .prop_map(move |value| {
88                DynSolValue::String(
89                    value.as_str().unwrap().trim().trim_end_matches('\0').to_string(),
90                )
91            })
92            .boxed(),
93        DynSolType::Tuple(ref params) => params
94            .iter()
95            .map(|param| fuzz_param_inner(param, None, max_fuzz_int))
96            .collect::<Vec<_>>()
97            .prop_map(DynSolValue::Tuple)
98            .boxed(),
99        DynSolType::FixedArray(ref param, size) => {
100            proptest::collection::vec(fuzz_param_inner(param, None, max_fuzz_int), size)
101                .prop_map(DynSolValue::FixedArray)
102                .boxed()
103        }
104        DynSolType::Array(ref param) => {
105            proptest::collection::vec(fuzz_param_inner(param, None, max_fuzz_int), 0..MAX_ARRAY_LEN)
106                .prop_map(DynSolValue::Array)
107                .boxed()
108        }
109        _ => panic!("unsupported fuzz param type: {param}"),
110    }
111}

113/// Given a parameter type, returns a strategy for generating values for that type, given some EVM
114/// fuzz state.
115///
116/// Works with ABI Encoder v2 tuples.
117pub fn fuzz_param_from_state(
118    param: &DynSolType,
119    state: &EvmFuzzState,
120    max_fuzz_int: Option<U256>,
121) -> BoxedStrategy<DynSolValue> {
122    // Value strategy that uses the state.
123    let value = || {
124        let state = state.clone();
125        let param = param.clone();
126        // Generate a bias and use it to pick samples or non-persistent values (50 / 50).
127        // Use `Index` instead of `Selector` when selecting a value to avoid iterating over the
128        // entire dictionary.
129        any::<(bool, prop::sample::Index)>().prop_map(move |(bias, index)| {
130            let state = state.dictionary_read();
131            let values = if bias { state.samples(&param) } else { None }
132                .unwrap_or_else(|| state.values())
133                .as_slice();
134            values[index.index(values.len())]
135        })
136    };
137
138    // Convert the value based on the parameter type
139    match *param {
140        DynSolType::Address => {
141            let deployed_libs = state.deployed_libs.clone();
142            value()
143                .prop_map(move |value| {
144                    let mut fuzzed_addr = Address::from_word(value);
145                    if deployed_libs.contains(&fuzzed_addr) {
146                        let mut rng = StdRng::seed_from_u64(0x1337); // use deterministic rng
147
148                        // Do not use addresses of deployed libraries as fuzz input, instead return
149                        // a deterministically random address. We cannot filter out this value (via
150                        // `prop_filter_map`) as proptest can invoke this closure after test
151                        // execution, and returning a `None` will cause it to panic.
152                        // See <https://github.com/foundry-rs/foundry/issues/9764> and <https://github.com/foundry-rs/foundry/issues/8639>.
153                        loop {
154                            fuzzed_addr.randomize_with(&mut rng);
155                            if !deployed_libs.contains(&fuzzed_addr) {
156                                break;
157                            }
158                        }
159                    }
160                    DynSolValue::Address(fuzzed_addr)
161                })
162                .boxed()
163        }
164        DynSolType::Function => value()
165            .prop_map(move |value| {
166                DynSolValue::Function(alloy_primitives::Function::from_word(value))
167            })
168            .boxed(),
169        DynSolType::FixedBytes(size @ 1..=32) => value()
170            .prop_map(move |mut v| {
171                v[size..].fill(0);
172                DynSolValue::FixedBytes(B256::from(v), size)
173            })
174            .boxed(),
175        DynSolType::Bool => DynSolValue::type_strategy(param).boxed(),
176        DynSolType::String => {
177            let state = state.clone();
178            (proptest::bool::weighted(0.3), any::<prop::sample::Index>())
179                .prop_flat_map(move |(use_ast, select_index)| {
180                    let dict = state.dictionary_read();
181
182                    // AST string literals available: 30% probability
183                    let ast_strings = dict.ast_strings();
184                    if use_ast && !ast_strings.is_empty() {
185                        let s = &ast_strings.as_slice()[select_index.index(ast_strings.len())];
186                        return Just(DynSolValue::String(s.clone())).boxed();
187                    }
188
189                    // Fallback to random string generation
190                    DynSolValue::type_strategy(&DynSolType::String)
191                        .prop_map(|value| {
192                            DynSolValue::String(
193                                value.as_str().unwrap().trim().trim_end_matches('\0').to_string(),
194                            )
195                        })
196                        .boxed()
197                })
198                .boxed()
199        }
200        DynSolType::Bytes => {
201            let state_clone = state.clone();
202            (
203                value(),
204                proptest::bool::weighted(0.1),
205                proptest::bool::weighted(0.2),
206                any::<prop::sample::Index>(),
207            )
208                .prop_map(move |(word, use_ast_string, use_ast_bytes, select_index)| {
209                    let dict = state_clone.dictionary_read();
210
211                    // Try string literals as bytes: 10% chance
212                    let ast_strings = dict.ast_strings();
213                    if use_ast_string && !ast_strings.is_empty() {
214                        let s = &ast_strings.as_slice()[select_index.index(ast_strings.len())];
215                        return DynSolValue::Bytes(s.as_bytes().to_vec());
216                    }
217
218                    // Try hex literals: 20% chance
219                    let ast_bytes = dict.ast_bytes();
220                    if use_ast_bytes && !ast_bytes.is_empty() {
221                        let bytes = &ast_bytes.as_slice()[select_index.index(ast_bytes.len())];
222                        return DynSolValue::Bytes(bytes.to_vec());
223                    }
224
225                    // Fallback to the generated word from the dictionary: 70% chance
226                    DynSolValue::Bytes(word.0.into())
227                })
228                .boxed()
229        }
230        DynSolType::Int(n @ 8..=256) => match n / 8 {
231            32 => value()
232                .prop_map(move |value| {
233                    let num = I256::from_be_bytes(value.0);
234                    let num = max_fuzz_int.map(|max| clamp(num, max)).unwrap_or(num);
235                    DynSolValue::Int(num, 256)
236                })
237                .boxed(),
238            1..=31 => value()
239                .prop_map(move |value| {
240                    // Extract lower N bits
241                    let uint_n = U256::from_be_bytes(value.0) % U256::from(1).wrapping_shl(n);
242                    // Interpret as signed int (two's complement) --> check sign bit (bit N-1).
243                    let sign_bit = U256::from(1) << (n - 1);
244                    let num = if uint_n >= sign_bit {
245                        // Negative number in two's complement
246                        let modulus = U256::from(1) << n;
247                        I256::from_raw(uint_n.wrapping_sub(modulus))
248                    } else {
249                        // Positive number
250                        I256::from_raw(uint_n)
251                    };
252
253                    DynSolValue::Int(num, n)
254                })
255                .boxed(),
256            _ => unreachable!(),
257        },
258        DynSolType::Uint(n @ 8..=256) => match n / 8 {
259            32 => value()
260                .prop_map(move |value| {
261                    let uint = U256::from_be_bytes(value.0);
262                    let uint = max_fuzz_int.map(|max| uint.min(max)).unwrap_or(uint);
263                    DynSolValue::Uint(uint, 256)
264                })
265                .boxed(),
266            1..=31 => value()
267                .prop_map(move |value| {
268                    let type_max = U256::from(1).wrapping_shl(n) - U256::from(1);
269                    let uint = U256::from_be_bytes(value.0) % U256::from(1).wrapping_shl(n);
270                    let uint = max_fuzz_int.map(|max| uint.min(max).min(type_max)).unwrap_or(uint);
271                    DynSolValue::Uint(uint, n)
272                })
273                .boxed(),
274            _ => unreachable!(),
275        },
276        DynSolType::Tuple(ref params) => params
277            .iter()
278            .map(|p| fuzz_param_from_state(p, state, max_fuzz_int))
279            .collect::<Vec<_>>()
280            .prop_map(DynSolValue::Tuple)
281            .boxed(),
282        DynSolType::FixedArray(ref param, size) => {
283            proptest::collection::vec(fuzz_param_from_state(param, state, max_fuzz_int), size)
284                .prop_map(DynSolValue::FixedArray)
285                .boxed()
286        }
287        DynSolType::Array(ref param) => proptest::collection::vec(
288            fuzz_param_from_state(param, state, max_fuzz_int),
289            0..MAX_ARRAY_LEN,
290        )
291        .prop_map(DynSolValue::Array)
292        .boxed(),
293        _ => panic!("unsupported fuzz param type: {param}"),
294    }
295}

297/// Mutates the current value of the given parameter type and value.
298pub fn mutate_param_value(
299    param: &DynSolType,
300    value: DynSolValue,
301    test_runner: &mut TestRunner,
302    state: &EvmFuzzState,
303    max_fuzz_int: Option<U256>,
304) -> DynSolValue {
305    let new_value = |param: &DynSolType, test_runner: &mut TestRunner| {
306        fuzz_param_from_state(param, state, max_fuzz_int)
307            .new_tree(test_runner)
308            .expect("Could not generate case")
309            .current()
310    };
311
312    match value {
313        DynSolValue::Bool(val) => {
314            // flip boolean value
315            trace!(target: "mutator", "Bool flip {val}");
316            Some(DynSolValue::Bool(!val))
317        }
318        DynSolValue::Uint(val, size) => match test_runner.rng().random_range(0..=6) {
319            0 => U256::increment_decrement(val, size, test_runner),
320            1 => U256::flip_random_bit(val, size, test_runner),
321            2 => U256::mutate_interesting_byte(val, size, test_runner),
322            3 => U256::mutate_interesting_word(val, size, test_runner),
323            4 => U256::mutate_interesting_dword(val, size, test_runner),
324            5 => U256::mutate_with_gaussian_noise(val, size, test_runner),
325            6 => None,
326            _ => unreachable!(),
327        }
328        .map(|v| DynSolValue::Uint(v, size)),
329        DynSolValue::Int(val, size) => match test_runner.rng().random_range(0..=6) {
330            0 => I256::increment_decrement(val, size, test_runner),
331            1 => I256::flip_random_bit(val, size, test_runner),
332            2 => I256::mutate_interesting_byte(val, size, test_runner),
333            3 => I256::mutate_interesting_word(val, size, test_runner),
334            4 => I256::mutate_interesting_dword(val, size, test_runner),
335            5 => I256::mutate_with_gaussian_noise(val, size, test_runner),
336            6 => None,
337            _ => unreachable!(),
338        }
339        .map(|v| DynSolValue::Int(v, size)),
340        DynSolValue::Address(val) => match test_runner.rng().random_range(0..=4) {
341            0 => Address::flip_random_bit(val, 20, test_runner),
342            1 => Address::mutate_interesting_byte(val, 20, test_runner),
343            2 => Address::mutate_interesting_word(val, 20, test_runner),
344            3 => Address::mutate_interesting_dword(val, 20, test_runner),
345            4 => None,
346            _ => unreachable!(),
347        }
348        .map(DynSolValue::Address),
349        DynSolValue::Array(mut values) => {
350            if let DynSolType::Array(param_type) = param
351                && !values.is_empty()
352            {
353                match test_runner.rng().random_range(0..=2) {
354                    // Decrease array size by removing a random element.
355                    0 => {
356                        values.remove(test_runner.rng().random_range(0..values.len()));
357                    }
358                    // Increase array size.
359                    1 => values.push(new_value(param_type, test_runner)),
360                    // Mutate random array element.
361                    2 => mutate_random_array_value(
362                        &mut values,
363                        param_type,
364                        test_runner,
365                        state,
366                        max_fuzz_int,
367                    ),
368                    _ => unreachable!(),
369                }
370                Some(DynSolValue::Array(values))
371            } else {
372                None
373            }
374        }
375        DynSolValue::FixedArray(mut values) => {
376            if let DynSolType::FixedArray(param_type, _size) = param
377                && !values.is_empty()
378            {
379                mutate_random_array_value(
380                    &mut values,
381                    param_type,
382                    test_runner,
383                    state,
384                    max_fuzz_int,
385                );
386                Some(DynSolValue::FixedArray(values))
387            } else {
388                None
389            }
390        }
391        DynSolValue::FixedBytes(word, size) => match test_runner.rng().random_range(0..=4) {
392            0 => Word::flip_random_bit(word, size, test_runner),
393            1 => Word::mutate_interesting_byte(word, size, test_runner),
394            2 => Word::mutate_interesting_word(word, size, test_runner),
395            3 => Word::mutate_interesting_dword(word, size, test_runner),
396            4 => None,
397            _ => unreachable!(),
398        }
399        .map(|word| DynSolValue::FixedBytes(word, size)),
400        DynSolValue::CustomStruct { name, prop_names, tuple: mut values } => {
401            if let DynSolType::CustomStruct { name: _, prop_names: _, tuple: tuple_types }
402            | DynSolType::Tuple(tuple_types) = param
403                && !values.is_empty()
404            {
405                // Mutate random struct element.
406                mutate_random_tuple_value(
407                    &mut values,
408                    tuple_types,
409                    test_runner,
410                    state,
411                    max_fuzz_int,
412                );
413                Some(DynSolValue::CustomStruct { name, prop_names, tuple: values })
414            } else {
415                None
416            }
417        }
418        DynSolValue::Tuple(mut values) => {
419            if let DynSolType::Tuple(tuple_types) = param
420                && !values.is_empty()
421            {
422                // Mutate random tuple element.
423                mutate_random_tuple_value(
424                    &mut values,
425                    tuple_types,
426                    test_runner,
427                    state,
428                    max_fuzz_int,
429                );
430                Some(DynSolValue::Tuple(values))
431            } else {
432                None
433            }
434        }
435        _ => None,
436    }
437    .unwrap_or_else(|| new_value(param, test_runner))
438}

440/// Mutates random value from given tuples.
441fn mutate_random_tuple_value(
442    tuple_values: &mut [DynSolValue],
443    tuple_types: &[DynSolType],
444    test_runner: &mut TestRunner,
445    state: &EvmFuzzState,
446    max_fuzz_int: Option<U256>,
447) {
448    let id = test_runner.rng().random_range(0..tuple_values.len());
449    let param_type = &tuple_types[id];
450    let old_val = replace(&mut tuple_values[id], DynSolValue::Bool(false));
451    let new_val = mutate_param_value(param_type, old_val, test_runner, state, max_fuzz_int);
452    tuple_values[id] = new_val;
453}

455/// Mutates random value from given array.
456fn mutate_random_array_value(
457    array_values: &mut [DynSolValue],
458    element_type: &DynSolType,
459    test_runner: &mut TestRunner,
460    state: &EvmFuzzState,
461    max_fuzz_int: Option<U256>,
462) {
463    let elem = array_values.choose_mut(&mut test_runner.rng()).unwrap();
464    let old_val = replace(elem, DynSolValue::Bool(false));
465    let new_val = mutate_param_value(element_type, old_val, test_runner, state, max_fuzz_int);
466    *elem = new_val;
467}

496    #[test]
497    fn can_fuzz_string_and_bytes_with_ast_literals_and_hashes() {
498        use super::fuzz_param_from_state;
499        use crate::strategies::LiteralMaps;
500        use alloy_dyn_abi::DynSolType;
501        use alloy_primitives::keccak256;
502        use proptest::strategy::Strategy;
503
504        // Seed dict with string values and their hashes --> mimic `CheatcodeAnalysis` behavior.
505        let mut literals = LiteralMaps::default();
506        literals.strings.insert("hello".to_string());
507        literals.strings.insert("world".to_string());
508        literals.words.entry(DynSolType::FixedBytes(32)).or_default().insert(keccak256("hello"));
509        literals.words.entry(DynSolType::FixedBytes(32)).or_default().insert(keccak256("world"));
510
511        let db = CacheDB::new(EmptyDB::default());
512        let state = EvmFuzzState::new(&db, FuzzDictionaryConfig::default(), &[], None, None);
513        state.seed_literals(literals);
514
515        let cfg = proptest::test_runner::Config { failure_persistence: None, ..Default::default() };
516        let mut runner = proptest::test_runner::TestRunner::new(cfg);
517
518        // Verify strategies generates the seeded AST literals
519        let mut generated_bytes = HashSet::new();
520        let mut generated_hashes = HashSet::new();
521        let mut generated_strings = HashSet::new();
522        let bytes_strategy = fuzz_param_from_state(&DynSolType::Bytes, &state, None);
523        let string_strategy = fuzz_param_from_state(&DynSolType::String, &state, None);
524        let bytes32_strategy = fuzz_param_from_state(&DynSolType::FixedBytes(32), &state, None);
525
526        for _ in 0..256 {
527            let tree = bytes_strategy.new_tree(&mut runner).unwrap();
528            if let Some(bytes) = tree.current().as_bytes()
529                && let Ok(s) = std::str::from_utf8(bytes)
530            {
531                generated_bytes.insert(s.to_string());
532            }
533
534            let tree = string_strategy.new_tree(&mut runner).unwrap();
535            if let Some(s) = tree.current().as_str() {
536                generated_strings.insert(s.to_string());
537            }
538
539            let tree = bytes32_strategy.new_tree(&mut runner).unwrap();
540            if let Some((bytes, size)) = tree.current().as_fixed_bytes()
541                && size == 32
542            {
543                generated_hashes.insert(B256::from_slice(bytes));
544            }
545        }
546
547        assert!(generated_bytes.contains("hello"));
548        assert!(generated_bytes.contains("world"));
549        assert!(generated_strings.contains("hello"));
550        assert!(generated_strings.contains("world"));
551        assert!(generated_hashes.contains(&keccak256("hello")));
552        assert!(generated_hashes.contains(&keccak256("world")));
553    }
