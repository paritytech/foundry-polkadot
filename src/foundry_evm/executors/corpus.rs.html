26const METADATA_SUFFIX: &str = "metadata.json";
27const JSON_EXTENSION: &str = ".json";
28const FAVORABILITY_THRESHOLD: f64 = 0.3;
29const COVERAGE_MAP_SIZE: usize = 65536;

31/// Possible mutation strategies to apply on a call sequence.
32#[derive(Debug, Clone)]
33enum MutationType {
34    /// Splice original call sequence.
35    Splice,
36    /// Repeat selected call several times.
37    Repeat,
38    /// Interleave calls from two random call sequences.
39    Interleave,
40    /// Replace prefix of the original call sequence with new calls.
41    Prefix,
42    /// Replace suffix of the original call sequence with new calls.
43    Suffix,
44    /// ABI mutate random args of selected call in sequence.
45    Abi,
46}

48/// Holds Corpus information.
49#[derive(Serialize)]
50struct CorpusEntry {
51    // Unique corpus identifier.
52    uuid: Uuid,
53    // Total mutations of corpus as primary source.
54    total_mutations: usize,
55    // New coverage found as a result of mutating this corpus.
56    new_finds_produced: usize,
57    // Corpus call sequence.
58    #[serde(skip_serializing)]
59    tx_seq: Vec<BasicTxDetails>,
60    // Whether this corpus is favored, i.e. producing new finds more often than
61    // `FAVORABILITY_THRESHOLD`.
62    is_favored: bool,
63}
64
65impl CorpusEntry {
66    /// New corpus from given call sequence and corpus path to read uuid.
67    pub fn new(tx_seq: Vec<BasicTxDetails>, path: PathBuf) -> eyre::Result<Self> {
68        let uuid = if let Some(stem) = path.file_stem().and_then(|s| s.to_str()) {
69            Uuid::try_from(stem.strip_suffix(JSON_EXTENSION).unwrap_or(stem).to_string())?
70        } else {
71            Uuid::new_v4()
72        };
73        Ok(Self { uuid, total_mutations: 0, new_finds_produced: 0, tx_seq, is_favored: false })
74    }
75
76    /// New corpus with given call sequence and new uuid.
77    pub fn from_tx_seq(tx_seq: &[BasicTxDetails]) -> Self {
78        Self {
79            uuid: Uuid::new_v4(),
80            total_mutations: 0,
81            new_finds_produced: 0,
82            tx_seq: tx_seq.into(),
83            is_favored: false,
84        }
85    }
86}

88#[derive(Serialize, Default)]
89pub(crate) struct CorpusMetrics {
90    // Number of edges seen during the invariant run.
91    cumulative_edges_seen: usize,
92    // Number of features (new hitcount bin of previously hit edge) seen during the invariant run.
93    cumulative_features_seen: usize,
94    // Number of corpus entries.
95    corpus_count: usize,
96    // Number of corpus entries that are favored.
97    favored_items: usize,
98}
99
100impl fmt::Display for CorpusMetrics {
101    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
102        writeln!(f)?;
103        writeln!(f, "        - cumulative edges seen: {}", self.cumulative_edges_seen)?;
104        writeln!(f, "        - cumulative features seen: {}", self.cumulative_features_seen)?;
105        writeln!(f, "        - corpus count: {}", self.corpus_count)?;
106        write!(f, "        - favored items: {}", self.favored_items)?;
107        Ok(())
108    }
109}
110
111impl CorpusMetrics {
112    /// Records number of new edges or features explored during the campaign.
113    pub fn update_seen(&mut self, is_edge: bool) {
114        if is_edge {
115            self.cumulative_edges_seen += 1;
116        } else {
117            self.cumulative_features_seen += 1;
118        }
119    }
120
121    /// Updates campaign favored items.
122    pub fn update_favored(&mut self, is_favored: bool, corpus_favored: bool) {
123        if is_favored && !corpus_favored {
124            self.favored_items += 1;
125        } else if !is_favored && corpus_favored {
126            self.favored_items -= 1;
127        }
128    }
129}

131/// Fuzz corpus manager, used in coverage guided fuzzing mode by both stateless and stateful tests.
132pub(crate) struct CorpusManager {
133    // Fuzzed calls generator.
134    tx_generator: BoxedStrategy<BasicTxDetails>,
135    // Call sequence mutation strategy type generator.
136    mutation_generator: BoxedStrategy<MutationType>,
137    // Corpus configuration.
138    config: FuzzCorpusConfig,
139    // In-memory corpus, populated from persisted files and current runs.
140    // Mutation is performed on these.
141    in_memory_corpus: Vec<CorpusEntry>,
142    // Identifier of current mutated entry.
143    current_mutated: Option<Uuid>,
144    // Number of failed replays from persisted corpus.
145    failed_replays: usize,
146    // History of binned hitcount of edges seen during fuzzing.
147    history_map: Vec<u8>,
148    // Corpus metrics.
149    pub(crate) metrics: CorpusMetrics,
150    // Maximum value for fuzzed integers, used to simulate smaller integer types.
151    // Unsigned: [0, max], Signed: [-(max+1), max].
152    max_fuzz_int: Option<U256>,
153}

155impl CorpusManager {
156    pub fn new(
157        config: FuzzCorpusConfig,
158        tx_generator: BoxedStrategy<BasicTxDetails>,
159        executor: &Executor,
160        fuzzed_function: Option<&Function>,
161        fuzzed_contracts: Option<&FuzzRunIdentifiedContracts>,
162        max_fuzz_int: Option<U256>,
163    ) -> eyre::Result<Self> {
164        let mutation_generator = prop_oneof![
165            Just(MutationType::Splice),
166            Just(MutationType::Repeat),
167            Just(MutationType::Interleave),
168            Just(MutationType::Prefix),
169            Just(MutationType::Suffix),
170            Just(MutationType::Abi),
171        ]
172        .boxed();
173        let mut history_map = vec![0u8; COVERAGE_MAP_SIZE];
174        let mut metrics = CorpusMetrics::default();
175        let mut in_memory_corpus = vec![];
176        let mut failed_replays = 0;
177
178        // Early return if corpus dir / coverage guided fuzzing not configured.
179        let Some(corpus_dir) = &config.corpus_dir else {
180            return Ok(Self {
181                tx_generator,
182                mutation_generator,
183                config,
184                in_memory_corpus,
185                current_mutated: None,
186                failed_replays,
187                history_map,
188                metrics,
189                max_fuzz_int,
190            });
191        };
192
193        // Ensure corpus dir for current test is created.
194        if !corpus_dir.is_dir() {
195            foundry_common::fs::create_dir_all(corpus_dir)?;
196        }
197
198        let can_replay_tx = |tx: &BasicTxDetails| -> bool {
199            fuzzed_contracts.is_some_and(|contracts| contracts.targets.lock().can_replay(tx))
200                || fuzzed_function.is_some_and(|function| {
201                    tx.call_details
202                        .calldata
203                        .get(..4)
204                        .is_some_and(|selector| function.selector() == selector)
205                })
206        };
207
208        'corpus_replay: for entry in std::fs::read_dir(corpus_dir)? {
209            let path = entry?.path();
210            if path.is_file()
211                && let Some(name) = path.file_name().and_then(|s| s.to_str())
212                && name.contains(METADATA_SUFFIX)
213            {
214                // Ignore metadata files
215                continue;
216            }
217
218            let read_corpus_result = match path.extension().and_then(|ext| ext.to_str()) {
219                Some("gz") => foundry_common::fs::read_json_gzip_file::<Vec<BasicTxDetails>>(&path),
220                _ => foundry_common::fs::read_json_file::<Vec<BasicTxDetails>>(&path),
221            };
222
223            let Ok(tx_seq) = read_corpus_result else {
224                trace!(target: "corpus", "failed to load corpus from {}", path.display());
225                continue;
226            };
227
228            if !tx_seq.is_empty() {
229                // Warm up history map from loaded sequences.
230                let mut executor = executor.clone();
231                for tx in &tx_seq {
232                    if can_replay_tx(tx) {
233                        let mut call_result = executor
234                            .call_raw(
235                                tx.sender,
236                                tx.call_details.target,
237                                tx.call_details.calldata.clone(),
238                                U256::ZERO,
239                            )
240                            .map_err(|e| eyre!(format!("Could not make raw evm call: {e}")))?;
241
242                        let (new_coverage, is_edge) =
243                            call_result.merge_edge_coverage(&mut history_map);
244                        if new_coverage {
245                            metrics.update_seen(is_edge);
246                        }
247
248                        // Commit only when running invariant / stateful tests.
249                        if fuzzed_contracts.is_some() {
250                            executor.commit(&mut call_result);
251                        }
252                    } else {
253                        failed_replays += 1;
254
255                        // If the only input for fuzzed function cannot be replied, then move to
256                        // next one without adding it in memory.
257                        if fuzzed_function.is_some() {
258                            continue 'corpus_replay;
259                        }
260                    }
261                }
262
263                metrics.corpus_count += 1;
264
265                trace!(
266                    target: "corpus",
267                    "load sequence with len {} from corpus file {}",
268                    tx_seq.len(),
269                    path.display()
270                );
271
272                // Populate in memory corpus with the sequence from corpus file.
273                in_memory_corpus.push(CorpusEntry::new(tx_seq, path)?);
274            }
275        }
276
277        Ok(Self {
278            tx_generator,
279            mutation_generator,
280            config,
281            in_memory_corpus,
282            current_mutated: None,
283            failed_replays,
284            history_map,
285            metrics,
286            max_fuzz_int,
287        })
288    }

290    /// Updates stats for the given call sequence, if new coverage produced.
291    /// Persists the call sequence (if corpus directory is configured and new coverage) and updates
292    /// in-memory corpus.
293    pub fn process_inputs(&mut self, inputs: &[BasicTxDetails], new_coverage: bool) {
294        // Early return if corpus dir / coverage guided fuzzing is not configured.
295        let Some(corpus_dir) = &self.config.corpus_dir else {
296            return;
297        };
298
299        // Update stats of current mutated primary corpus.
300        if let Some(uuid) = &self.current_mutated {
301            if let Some(corpus) =
302                self.in_memory_corpus.iter_mut().find(|corpus| corpus.uuid.eq(uuid))
303            {
304                corpus.total_mutations += 1;
305                if new_coverage {
306                    corpus.new_finds_produced += 1
307                }
308                let is_favored = (corpus.new_finds_produced as f64 / corpus.total_mutations as f64)
309                    > FAVORABILITY_THRESHOLD;
310                self.metrics.update_favored(is_favored, corpus.is_favored);
311                corpus.is_favored = is_favored;
312
313                trace!(
314                    target: "corpus",
315                    "updated corpus {}, total mutations: {}, new finds: {}",
316                    corpus.uuid, corpus.total_mutations, corpus.new_finds_produced
317                );
318            }
319
320            self.current_mutated = None;
321        }
322
323        // Collect inputs only if current run produced new coverage.
324        if !new_coverage {
325            return;
326        }
327
328        let corpus = CorpusEntry::from_tx_seq(inputs);
329        let corpus_uuid = corpus.uuid;
330
331        // Persist to disk if corpus dir is configured.
332        let write_result = if self.config.corpus_gzip {
333            foundry_common::fs::write_json_gzip_file(
334                corpus_dir.join(format!("{corpus_uuid}{JSON_EXTENSION}.gz")).as_path(),
335                &corpus.tx_seq,
336            )
337        } else {
338            foundry_common::fs::write_json_file(
339                corpus_dir.join(format!("{corpus_uuid}{JSON_EXTENSION}")).as_path(),
340                &corpus.tx_seq,
341            )
342        };
343
344        if let Err(err) = write_result {
345            debug!(target: "corpus", %err, "Failed to record call sequence {:?}", &corpus.tx_seq);
346        } else {
347            trace!(
348                target: "corpus",
349                "persisted {} inputs for new coverage in {corpus_uuid} corpus",
350                &corpus.tx_seq.len()
351            );
352        }
353
354        // This includes reverting txs in the corpus and `can_continue` removes
355        // them. We want this as it is new coverage and may help reach the other branch.
356        self.metrics.corpus_count += 1;
357        self.in_memory_corpus.push(corpus);
358    }

360    /// Generates new call sequence from in memory corpus. Evicts oldest corpus mutated more than
361    /// configured max mutations value. Used by invariant test campaigns.
362    pub fn new_inputs(
363        &mut self,
364        test_runner: &mut TestRunner,
365        fuzz_state: &EvmFuzzState,
366        targeted_contracts: &FuzzRunIdentifiedContracts,
367    ) -> eyre::Result<Vec<BasicTxDetails>> {
368        let mut new_seq = vec![];
369
370        // Early return with first_input only if corpus dir / coverage guided fuzzing not
371        // configured.
372        if !self.config.is_coverage_guided() {
373            new_seq.push(self.new_tx(test_runner)?);
374            return Ok(new_seq);
375        };
376
377        if !self.in_memory_corpus.is_empty() {
378            self.evict_oldest_corpus()?;
379
380            let mutation_type = self
381                .mutation_generator
382                .new_tree(test_runner)
383                .map_err(|err| eyre!("Could not generate mutation type {err}"))?
384                .current();
385            let rng = test_runner.rng();
386            let corpus_len = self.in_memory_corpus.len();
387            let primary = &self.in_memory_corpus[rng.random_range(0..corpus_len)];
388            let secondary = &self.in_memory_corpus[rng.random_range(0..corpus_len)];
389
390            match mutation_type {
391                MutationType::Splice => {
392                    trace!(target: "corpus", "splice {} and {}", primary.uuid, secondary.uuid);
393
394                    self.current_mutated = Some(primary.uuid);
395
396                    let start1 = rng.random_range(0..primary.tx_seq.len());
397                    let end1 = rng.random_range(start1..primary.tx_seq.len());
398
399                    let start2 = rng.random_range(0..secondary.tx_seq.len());
400                    let end2 = rng.random_range(start2..secondary.tx_seq.len());
401
402                    for tx in primary.tx_seq.iter().take(end1).skip(start1) {
403                        new_seq.push(tx.clone());
404                    }
405                    for tx in secondary.tx_seq.iter().take(end2).skip(start2) {
406                        new_seq.push(tx.clone());
407                    }
408                }
409                MutationType::Repeat => {
410                    let corpus = if rng.random::<bool>() { primary } else { secondary };
411                    trace!(target: "corpus", "repeat {}", corpus.uuid);
412
413                    self.current_mutated = Some(corpus.uuid);
414
415                    new_seq = corpus.tx_seq.clone();
416                    let start = rng.random_range(0..corpus.tx_seq.len());
417                    let end = rng.random_range(start..corpus.tx_seq.len());
418                    let item_idx = rng.random_range(0..corpus.tx_seq.len());
419                    let repeated = vec![new_seq[item_idx].clone(); end - start];
420                    new_seq.splice(start..end, repeated);
421                }
422                MutationType::Interleave => {
423                    trace!(target: "corpus", "interleave {} with {}", primary.uuid, secondary.uuid);
424
425                    self.current_mutated = Some(primary.uuid);
426
427                    for (tx1, tx2) in primary.tx_seq.iter().zip(secondary.tx_seq.iter()) {
428                        // chunks?
429                        let tx = if rng.random::<bool>() { tx1.clone() } else { tx2.clone() };
430                        new_seq.push(tx);
431                    }
432                }
433                MutationType::Prefix => {
434                    let corpus = if rng.random::<bool>() { primary } else { secondary };
435                    trace!(target: "corpus", "overwrite prefix of {}", corpus.uuid);
436
437                    self.current_mutated = Some(corpus.uuid);
438
439                    new_seq = corpus.tx_seq.clone();
440                    for i in 0..rng.random_range(0..=new_seq.len()) {
441                        new_seq[i] = self.new_tx(test_runner)?;
442                    }
443                }
444                MutationType::Suffix => {
445                    let corpus = if rng.random::<bool>() { primary } else { secondary };
446                    trace!(target: "corpus", "overwrite suffix of {}", corpus.uuid);
447
448                    self.current_mutated = Some(corpus.uuid);
449
450                    new_seq = corpus.tx_seq.clone();
451                    for i in new_seq.len() - rng.random_range(0..new_seq.len())..corpus.tx_seq.len()
452                    {
453                        new_seq[i] = self.new_tx(test_runner)?;
454                    }
455                }
456                MutationType::Abi => {
457                    let targets = targeted_contracts.targets.lock();
458                    let corpus = if rng.random::<bool>() { primary } else { secondary };
459                    trace!(target: "corpus", "ABI mutate args of {}", corpus.uuid);
460
461                    self.current_mutated = Some(corpus.uuid);
462
463                    new_seq = corpus.tx_seq.clone();
464
465                    let idx = rng.random_range(0..new_seq.len());
466                    let tx = new_seq.get_mut(idx).unwrap();
467                    if let (_, Some(function)) = targets.fuzzed_artifacts(tx) {
468                        // TODO add call_value to call details and mutate it as well as sender some
469                        // of the time
470                        if !function.inputs.is_empty() {
471                            self.abi_mutate(tx, function, test_runner, fuzz_state)?;
472                        }
473                    }
474                }
475            }
476        }
477
478        // Make sure the new sequence contains at least one tx to start fuzzing from.
479        if new_seq.is_empty() {
480            new_seq.push(self.new_tx(test_runner)?);
481        }
482        trace!(target: "corpus", "new sequence of {} calls generated", new_seq.len());
483
484        Ok(new_seq)
485    }

487    /// Generates new input from in memory corpus. Evicts oldest corpus mutated more than
488    /// configured max mutations value. Used by fuzz test campaigns.
489    pub fn new_input(
490        &mut self,
491        test_runner: &mut TestRunner,
492        fuzz_state: &EvmFuzzState,
493        function: &Function,
494    ) -> eyre::Result<Bytes> {
495        // Early return if not running with coverage guided fuzzing.
496        if !self.config.is_coverage_guided() {
497            return Ok(self.new_tx(test_runner)?.call_details.calldata);
498        }
499
500        let tx = if !self.in_memory_corpus.is_empty() {
501            self.evict_oldest_corpus()?;
502
503            let corpus = &self.in_memory_corpus
504                [test_runner.rng().random_range(0..self.in_memory_corpus.len())];
505            self.current_mutated = Some(corpus.uuid);
506            let new_seq = corpus.tx_seq.clone();
507            let mut tx = new_seq.first().unwrap().clone();
508            self.abi_mutate(&mut tx, function, test_runner, fuzz_state)?;
509            tx
510        } else {
511            self.new_tx(test_runner)?
512        };
513
514        Ok(tx.call_details.calldata)
515    }

517    /// Returns the next call to be used in call sequence.
518    /// If coverage guided fuzzing is not configured or if previous input was discarded then this is
519    /// a new tx from strategy.
520    /// If running with coverage guided fuzzing it returns a new call only when sequence
521    /// does not have enough entries, or randomly. Otherwise, returns the next call from initial
522    /// sequence.
523    pub fn generate_next_input(
524        &mut self,
525        test_runner: &mut TestRunner,
526        sequence: &[BasicTxDetails],
527        discarded: bool,
528        depth: usize,
529    ) -> eyre::Result<BasicTxDetails> {
530        // Early return with new input if corpus dir / coverage guided fuzzing not configured or if
531        // call was discarded.
532        if self.config.corpus_dir.is_none() || discarded {
533            return self.new_tx(test_runner);
534        }
535
536        // When running with coverage guided fuzzing enabled then generate new sequence if initial
537        // sequence's length is less than depth or randomly, to occasionally intermix new txs.
538        if depth > sequence.len().saturating_sub(1) || test_runner.rng().random_ratio(1, 10) {
539            return self.new_tx(test_runner);
540        }
541
542        // Continue with the next call initial sequence
543        Ok(sequence[depth].clone())
544    }

560    /// Collects coverage from call result and updates metrics.
561    pub fn merge_edge_coverage(&mut self, call_result: &mut RawCallResult) -> bool {
562        if !self.config.collect_edge_coverage() {
563            return false;
564        }
565
566        let (new_coverage, is_edge) = call_result.merge_edge_coverage(&mut self.history_map);
567        if new_coverage {
568            self.metrics.update_seen(is_edge);
569        }
570        new_coverage
571    }

573    /// Flush the oldest corpus mutated more than configured max mutations unless they are
574    /// favored.
575    fn evict_oldest_corpus(&mut self) -> eyre::Result<()> {
576        if self.in_memory_corpus.len() > self.config.corpus_min_size.max(1)
577            && let Some(index) = self.in_memory_corpus.iter().position(|corpus| {
578                corpus.total_mutations > self.config.corpus_min_mutations && !corpus.is_favored
579            })
580        {
581            let corpus = self.in_memory_corpus.get(index).unwrap();
582
583            let uuid = corpus.uuid;
584            debug!(target: "corpus", "evict corpus {uuid}");
585
586            // Flush to disk the seed metadata at the time of eviction.
587            let eviction_time = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs();
588            foundry_common::fs::write_json_file(
589                self.config
590                    .corpus_dir
591                    .clone()
592                    .unwrap()
593                    .join(format!("{uuid}-{eviction_time}-{METADATA_SUFFIX}"))
594                    .as_path(),
595                &corpus,
596            )?;
597
598            // Remove corpus from memory.
599            self.in_memory_corpus.remove(index);
600        }
601        Ok(())
602    }

604    /// Mutates calldata of provided tx by abi decoding current values and randomly selecting the
605    /// inputs to change.
606    fn abi_mutate(
607        &self,
608        tx: &mut BasicTxDetails,
609        function: &Function,
610        test_runner: &mut TestRunner,
611        fuzz_state: &EvmFuzzState,
612    ) -> eyre::Result<()> {
613        // let rng = test_runner.rng();
614        let mut arg_mutation_rounds =
615            test_runner.rng().random_range(0..=function.inputs.len()).max(1);
616        let round_arg_idx: Vec<usize> = if function.inputs.len() <= 1 {
617            vec![0]
618        } else {
619            (0..arg_mutation_rounds)
620                .map(|_| test_runner.rng().random_range(0..function.inputs.len()))
621                .collect()
622        };
623        let mut prev_inputs = function
624            .abi_decode_input(&tx.call_details.calldata[4..])
625            .map_err(|err| eyre!("failed to load previous inputs: {err}"))?;
626
627        while arg_mutation_rounds > 0 {
628            let idx = round_arg_idx[arg_mutation_rounds - 1];
629            prev_inputs[idx] = mutate_param_value(
630                &function
631                    .inputs
632                    .get(idx)
633                    .expect("Could not get input to mutate")
634                    .selector_type()
635                    .parse()?,
636                prev_inputs[idx].clone(),
637                test_runner,
638                fuzz_state,
639                self.max_fuzz_int,
640            );
641            arg_mutation_rounds -= 1;
642        }
643
644        tx.call_details.calldata =
645            function.abi_encode_input(&prev_inputs).map_err(|e| eyre!(e.to_string()))?.into();
646        Ok(())
647    }
648}

767        // manager with two corpora

769        let config = FuzzCorpusConfig {

777        favored.total_mutations = 2;
778        favored.is_favored = true;
779
780        let mut non_favored = CorpusEntry::from_tx_seq(&[basic_tx()]);
781        non_favored.total_mutations = 2;
782        non_favored.is_favored = false;
783        let non_favored_uuid = non_favored.uuid;
784
785        let mut manager = CorpusManager {
786            tx_generator: tx_gen,
787            mutation_generator: Just(MutationType::Repeat).boxed(),
788            config,
789            in_memory_corpus: vec![favored, non_favored],
790            current_mutated: None,
791            failed_replays: 0,
792            history_map: vec![0u8; COVERAGE_MAP_SIZE],
793            metrics: CorpusMetrics::default(),
794            max_fuzz_int: None,
795        };
796
797        // First eviction should remove the non-favored one
798        manager.evict_oldest_corpus().unwrap();
799        assert_eq!(manager.in_memory_corpus.len(), 1);
800        assert!(manager.in_memory_corpus.iter().all(|c| c.is_favored));
801
802        // Attempt eviction again: only favored remains → should not remove
803        manager.evict_oldest_corpus().unwrap();

805
806        // ensure the evicted one was the non-favored uuid
807        assert!(manager.in_memory_corpus.iter().all(|c| c.uuid != non_favored_uuid));
