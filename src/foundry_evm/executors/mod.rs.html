85/// EVM executor.
86///
87/// The executor can be configured with various `revm::Inspector`s, like `Cheatcodes`.
88///
89/// There are multiple ways of interacting the EVM:
90/// - `call`: executes a transaction, but does not persist any state changes; similar to `eth_call`,
91///   where the EVM state is unchanged after the call.
92/// - `transact`: executes a transaction and persists the state changes
93/// - `deploy`: a special case of `transact`, specialized for persisting the state of a contract
94///   deployment
95/// - `setup`: a special case of `transact`, used to set up the environment for a test
96#[derive(Clone, Debug)]
97pub struct Executor {
98    /// The underlying `revm::Database` that contains the EVM storage.
99    // Note: We do not store an EVM here, since we are really
100    // only interested in the database. REVM's `EVM` is a thin
101    // wrapper around spawning a new EVM on every call anyway,
102    // so the performance difference should be negligible.
103    backend: Backend,
104    /// The EVM environment.
105    env: Env,
106    /// The Revm inspector stack.
107    inspector: InspectorStack,
108    /// The gas limit for calls and deployments.
109    gas_limit: u64,
110    /// Whether `failed()` should be called on the test contract to determine if the test failed.
111    legacy_assertions: bool,
112    /// Executor behavior strategy.
113    pub strategy: ExecutorStrategy,
114}

231    /// Creates the default CREATE2 Contract Deployer for local tests and scripts.
232    pub fn deploy_create2_deployer(&mut self) -> eyre::Result<()> {
233        trace!("deploying local create2 deployer");
234        let create2_deployer_account = self
235            .backend()
236            .basic_ref(DEFAULT_CREATE2_DEPLOYER)?
237            .ok_or_else(|| BackendError::MissingAccount(DEFAULT_CREATE2_DEPLOYER))?;
238
239        // If the deployer is not currently deployed, deploy the default one.
240        if create2_deployer_account.code.is_none_or(|code| code.is_empty()) {
241            let creator = DEFAULT_CREATE2_DEPLOYER_DEPLOYER;
242
243            // Probably 0, but just in case.
244            let initial_balance = self.get_balance(creator)?;
245            self.set_balance(creator, self.strategy.runner.max_balance())?;
246
247            let res =
248                self.deploy(creator, DEFAULT_CREATE2_DEPLOYER_CODE.into(), U256::ZERO, None)?;
249            trace!(create2=?res.address, "deployed local create2 deployer");
250
251            self.set_balance(creator, initial_balance)?;
252        }
253        Ok(())
254    }

349    /// Deploys a contract using the given `env` and commits the new state to the underlying
350    /// database.
351    ///
352    /// # Panics
353    ///
354    /// Panics if `env.tx.kind` is not `TxKind::Create(_)`.
355    #[instrument(name = "deploy", level = "debug", skip_all)]
356    pub fn deploy_with_env(
357        &mut self,
358        env: Env,
359        rd: Option<&RevertDecoder>,
360    ) -> Result<DeployResult, EvmError> {
361        assert!(
362            matches!(env.tx.kind, TxKind::Create),
363            "Expected create transaction, got {:?}",
364            env.tx.kind
365        );
366        trace!(sender=%env.tx.caller, "deploying contract");
367
368        let mut result = self.transact_with_env(env)?;
369        result = result.into_result(rd)?;
370        let Some(Output::Create(_, Some(address))) = result.out else {
371            panic!("Deployment succeeded, but no address was returned: {result:#?}");
372        };
373
374        // also mark this library as persistent, this will ensure that the state of the library is
375        // persistent across fork swaps in forking mode
376        self.backend_mut().add_persistent_account(address);
377
378        trace!(%address, "deployed contract");
379
380        Ok(DeployResult { raw: result, address })
381    }

383    /// Calls the `setUp()` function on a contract.
384    ///
385    /// This will commit any state changes to the underlying database.
386    ///
387    /// Ayn changes made during the setup call to env's block environment are persistent, for
388    /// example `vm.chainId()` will change the `block.chainId` for all subsequent test calls.
389    #[instrument(name = "setup", level = "debug", skip_all)]
390    pub fn setup(
391        &mut self,
392        from: Option<Address>,
393        to: Address,
394        rd: Option<&RevertDecoder>,
395    ) -> Result<RawCallResult, EvmError> {
396        trace!(?from, ?to, "setting up contract");
397
398        let from = from.unwrap_or(CALLER);
399        self.backend_mut().set_test_contract(to).set_caller(from);
400        let calldata = Bytes::from_static(&ITest::setUpCall::SELECTOR);
401        let mut res = self.transact_raw(from, to, calldata, U256::ZERO)?;
402        res = res.into_result(rd)?;
403
404        // record any changes made to the block's environment during setup
405        self.env_mut().evm_env.block_env = res.env.evm_env.block_env.clone();
406        // and also the chainid, which can be set manually
407        self.env_mut().evm_env.cfg_env.chain_id = res.env.evm_env.cfg_env.chain_id;
408
409        let success =
410            self.is_raw_call_success(to, Cow::Borrowed(&res.state_changeset), &res, false);
411        if !success {
412            return Err(res.into_execution_error("execution error".to_string()).into());
413        }
414
415        Ok(res)
416    }

519    /// Execute the transaction configured in `env.tx`.
520    ///
521    /// The state after the call is **not** persisted.
522    #[instrument(name = "call", level = "debug", skip_all)]
523    pub fn call_with_env(&self, mut env: Env) -> eyre::Result<RawCallResult> {
524        let mut stack = self.inspector().clone();
525        let mut backend = CowBackend::new_borrowed(self.backend());
526        let result = self.strategy.runner.call(
527            self.strategy.context.as_ref(),
528            &mut backend,
529            &mut env,
530            &self.env,
531            &mut stack,
532        )?;
533        convert_executed_result(env, stack, result, backend.has_state_snapshot_failure())
534    }

536    /// Execute the transaction configured in `env.tx`.
537    #[instrument(name = "transact", level = "debug", skip_all)]
538    pub fn transact_with_env(&mut self, mut env: Env) -> eyre::Result<RawCallResult> {
539        let mut stack = self.inspector().clone();
540        let backend = &mut self.backend;
541        let result = self.strategy.runner.transact(
542            self.strategy.context.as_mut(),
543            backend,
544            &mut env,
545            &self.env,
546            &mut stack,
547        )?;
548        let mut result =
549            convert_executed_result(env, stack, result, backend.has_state_snapshot_failure())?;
550        self.commit(&mut result);
551        Ok(result)
552    }

554    /// Commit the changeset to the database and adjust `self.inspector_config` values according to
555    /// the executed call result.
556    ///
557    /// This should not be exposed to the user, as it should be called only by `transact*`.
558    #[instrument(name = "commit", level = "debug", skip_all)]
559    fn commit(&mut self, result: &mut RawCallResult) {
560        // Persist changes to db.
561        self.backend_mut().commit(result.state_changeset.clone());
562
563        // Persist cheatcode state.
564        self.inspector_mut().cheatcodes = result.cheatcodes.take();
565        if let Some(cheats) = self.inspector_mut().cheatcodes.as_mut() {
566            // Clear broadcastable transactions
567            cheats.broadcastable_transactions.clear();
568            cheats.ignored_traces.ignored.clear();
569
570            // if tracing was paused but never unpaused, we should begin next frame with tracing
571            // still paused
572            if let Some(last_pause_call) = cheats.ignored_traces.last_pause_call.as_mut() {
573                *last_pause_call = (0, 0);
574            }
575        }
576
577        // Persist the changed environment.
578        self.inspector_mut().set_env(&result.env);
579    }

599    /// Returns `true` if a test can be considered successful.
600    ///
601    /// This is the same as [`Self::is_success`], but intended for outcomes of [`Self::call_raw`].
602    pub fn is_raw_call_success(
603        &self,
604        address: Address,
605        state_changeset: Cow<'_, StateChangeset>,
606        call_result: &RawCallResult,
607        should_fail: bool,
608    ) -> bool {
609        if call_result.has_state_snapshot_failure {
610            // a failure occurred in a reverted snapshot, which is considered a failed test
611            return should_fail;
612        }
613        self.is_success(address, call_result.reverted, state_changeset, should_fail)
614    }

616    /// Returns `true` if a test can be considered successful.
617    ///
618    /// If the call succeeded, we also have to check the global and local failure flags.
619    ///
620    /// These are set by the test contract itself when an assertion fails, using the internal `fail`
621    /// function. The global flag is located in [`CHEATCODE_ADDRESS`] at slot [`GLOBAL_FAIL_SLOT`],
622    /// and the local flag is located in the test contract at an unspecified slot.
623    ///
624    /// This behavior is inherited from Dapptools, where initially only a public
625    /// `failed` variable was used to track test failures, and later, a global failure flag was
626    /// introduced to track failures across multiple contracts in
627    /// [ds-test#30](https://github.com/dapphub/ds-test/pull/30).
628    ///
629    /// The assumption is that the test runner calls `failed` on the test contract to determine if
630    /// it failed. However, we want to avoid this as much as possible, as it is relatively
631    /// expensive to set up an EVM call just for checking a single boolean flag.
632    ///
633    /// See:
634    /// - Newer DSTest: <https://github.com/dapphub/ds-test/blob/e282159d5170298eb2455a6c05280ab5a73a4ef0/src/test.sol#L47-L63>
635    /// - Older DSTest: <https://github.com/dapphub/ds-test/blob/9ca4ecd48862b40d7b0197b600713f64d337af12/src/test.sol#L38-L49>
636    /// - forge-std: <https://github.com/foundry-rs/forge-std/blob/19891e6a0b5474b9ea6827ddb90bb9388f7acfc0/src/StdAssertions.sol#L38-L44>
637    pub fn is_success(
638        &self,
639        address: Address,
640        reverted: bool,
641        state_changeset: Cow<'_, StateChangeset>,
642        should_fail: bool,
643    ) -> bool {
644        let success = self.is_success_raw(address, reverted, state_changeset);
645        should_fail ^ success
646    }
647
648    #[instrument(name = "is_success", level = "debug", skip_all)]
649    fn is_success_raw(
650        &self,
651        address: Address,
652        reverted: bool,
653        state_changeset: Cow<'_, StateChangeset>,
654    ) -> bool {
655        // The call reverted.
656        if reverted {
657            return false;
658        }
659
660        // A failure occurred in a reverted snapshot, which is considered a failed test.
661        if self.backend().has_state_snapshot_failure() {
662            return false;
663        }
664
665        // Check the global failure slot.
666        if let Some(acc) = state_changeset.get(&CHEATCODE_ADDRESS)
667            && let Some(failed_slot) = acc.storage.get(&GLOBAL_FAIL_SLOT)
668            && !failed_slot.present_value().is_zero()
669        {
670            return false;
671        }
672        if let Ok(failed_slot) = self.backend().storage_ref(CHEATCODE_ADDRESS, GLOBAL_FAIL_SLOT)
673            && !failed_slot.is_zero()
674        {
675            return false;
676        }
677
678        if !self.legacy_assertions {
679            return true;
680        }
681
682        // Finally, resort to calling `DSTest::failed`.
683        {
684            // Construct a new bare-bones backend to evaluate success.
685            let mut backend = self.backend().clone_empty();
686
687            // We only clone the test contract and cheatcode accounts,
688            // that's all we need to evaluate success.
689            for address in [address, CHEATCODE_ADDRESS] {
690                let Ok(acc) = self.backend().basic_ref(address) else { return false };
691                backend.insert_account_info(address, acc.unwrap_or_default());
692            }
693
694            // If this test failed any asserts, then this changeset will contain changes
695            // `false -> true` for the contract's `failed` variable and the `globalFailure` flag
696            // in the state of the cheatcode address,
697            // which are both read when we call `"failed()(bool)"` in the next step.
698            backend.commit(state_changeset.into_owned());
699
700            // Check if a DSTest assertion failed
701            let executor = self.clone_with_backend(backend);
702            let call = executor.call_sol(CALLER, address, &ITest::failedCall {}, U256::ZERO, None);
703            match call {
704                Ok(CallResult { raw: _, decoded_result: failed }) => {
705                    trace!(failed, "DSTest::failed()");
706                    !failed
707                }
708                Err(err) => {
709                    trace!(%err, "failed to call DSTest::failed()");
710                    true
711                }
712            }
713        }
714    }
715
716    /// Creates the environment to use when executing a transaction in a test context
717    ///
718    /// If using a backend with cheatcodes, `tx.gas_price` and `block.number` will be overwritten by
719    /// the cheatcode state in between calls.
720    fn build_test_env(&self, caller: Address, kind: TxKind, data: Bytes, value: U256) -> Env {
721        Env {
722            evm_env: EvmEnv {
723                cfg_env: {
724                    let mut cfg = self.env().evm_env.cfg_env.clone();
725                    cfg.spec = self.spec_id();
726                    cfg
727                },
728                // We always set the gas price to 0 so we can execute the transaction regardless of
729                // network conditions - the actual gas price is kept in `self.block` and is applied
730                // by the cheatcode handler if it is enabled
731                block_env: BlockEnv {
732                    basefee: 0,
733                    gas_limit: self.gas_limit,
734                    ..self.env().evm_env.block_env.clone()
735                },
736            },
737            tx: TxEnv {
738                caller,
739                kind,
740                data,
741                value,
742                // As above, we set the gas price to 0.
743                gas_price: 0,
744                gas_priority_fee: None,
745                gas_limit: self.gas_limit,
746                chain_id: Some(self.env().evm_env.cfg_env.chain_id),
747                ..self.env().tx.clone()
748            },
749        }
750    }
751
752    pub fn call_sol_default<C: SolCall>(&self, to: Address, args: &C) -> C::Return
753    where
754        C::Return: Default,
755    {
756        self.call_sol(CALLER, to, args, U256::ZERO, None)
757            .map(|c| c.decoded_result)
758            .inspect_err(|e| warn!(target: "forge::test", "failed calling {:?}: {e}", C::SIGNATURE))
759            .unwrap_or_default()
760    }

924    /// Unpacks an EVM result.
925    pub fn from_evm_result(r: Result<Self, EvmError>) -> eyre::Result<(Self, Option<String>)> {
926        match r {
927            Ok(r) => Ok((r, None)),
928            Err(EvmError::Execution(e)) => Ok((e.raw, Some(e.reason))),
929            Err(e) => Err(e.into()),

933    /// Unpacks an execution result.
934    pub fn from_execution_result(r: Result<Self, ExecutionErr>) -> (Self, Option<String>) {

937            Err(e) => (e.raw, Some(e.reason)),

940
941    /// Converts the result of the call into an `EvmError`.
942    pub fn into_evm_error(self, rd: Option<&RevertDecoder>) -> EvmError {
943        if let Some(reason) = SkipReason::decode(&self.result) {
944            return EvmError::Skip(reason);
945        }
946        let reason = rd.unwrap_or_default().decode(&self.result, self.exit_reason);
947        EvmError::Execution(Box::new(self.into_execution_error(reason)))
948    }
949
950    /// Converts the result of the call into an `ExecutionErr`.
951    pub fn into_execution_error(self, reason: String) -> ExecutionErr {
952        ExecutionErr { raw: self, reason }
953    }
954
955    /// Returns an `EvmError` if the call failed, otherwise returns `self`.
956    pub fn into_result(self, rd: Option<&RevertDecoder>) -> Result<Self, EvmError> {
957        if let Some(reason) = self.exit_reason
958            && reason.is_ok()
959        {
960            Ok(self)
961        } else {
962            Err(self.into_evm_error(rd))
963        }
964    }
965
966    /// Decodes the result of the call with the given function.
967    pub fn into_decoded_result(
968        mut self,
969        func: &Function,
970        rd: Option<&RevertDecoder>,
971    ) -> Result<CallResult, EvmError> {
972        self = self.into_result(rd)?;
973        let mut result = func.abi_decode_output(&self.result)?;
974        let decoded_result = if result.len() == 1 {
975            result.pop().unwrap()
976        } else {
977            // combine results into a tuple
978            DynSolValue::Tuple(result)
979        };
980        Ok(CallResult { raw: self, decoded_result })
981    }
982
983    /// Returns the transactions generated from this call.
984    pub fn transactions(&self) -> Option<&BroadcastableTransactions> {
985        self.cheatcodes.as_ref().map(|c| &c.broadcastable_transactions)
986    }
987
988    /// Update provided history map with edge coverage info collected during this call.
989    /// Uses AFL binning algo <https://github.com/h0mbre/Lucid/blob/3026e7323c52b30b3cf12563954ac1eaa9c6981e/src/coverage.rs#L57-L85>
990    pub fn merge_edge_coverage(&mut self, history_map: &mut [u8]) -> (bool, bool) {
991        let mut new_coverage = false;
992        let mut is_edge = false;
993        if let Some(x) = &mut self.edge_coverage {
994            // Iterate over the current map and the history map together and update
995            // the history map, if we discover some new coverage, report true
996            for (curr, hist) in std::iter::zip(x, history_map) {
997                // If we got a hitcount of at least 1
998                if *curr > 0 {
999                    // Convert hitcount into bucket count
1000                    let bucket = match *curr {
1001                        0 => 0,
1002                        1 => 1,
1003                        2 => 2,
1004                        3 => 4,
1005                        4..=7 => 8,
1006                        8..=15 => 16,
1007                        16..=31 => 32,
1008                        32..=127 => 64,
1009                        128..=255 => 128,
1010                    };
1011
1012                    // If the old record for this edge pair is lower, update
1013                    if *hist < bucket {
1014                        if *hist == 0 {
1015                            // Counts as an edge the first time we see it, otherwise it's a feature.
1016                            is_edge = true;
1017                        }
1018                        *hist = bucket;
1019                        new_coverage = true;
1020                    }
1021
1022                    // Zero out the current map for next iteration.
1023                    *curr = 0;
1024                }
1025            }
1026        }
1027        (new_coverage, is_edge)
1028    }
1029}

1055/// Converts the data aggregated in the `inspector` and `call` to a `RawCallResult`
1056fn convert_executed_result(
1057    env: Env,
1058    inspector: InspectorStack,
1059    ResultAndState { result, state: state_changeset }: ResultAndState,
1060    has_state_snapshot_failure: bool,
1061) -> eyre::Result<RawCallResult> {
1062    let (exit_reason, gas_refunded, gas_used, out, exec_logs) = match result {
1063        ExecutionResult::Success { reason, gas_used, gas_refunded, output, logs, .. } => {
1064            (reason.into(), gas_refunded, gas_used, Some(output), logs)
1065        }
1066        ExecutionResult::Revert { gas_used, output } => {
1067            // Need to fetch the unused gas
1068            (InstructionResult::Revert, 0_u64, gas_used, Some(Output::Call(output)), vec![])
1069        }
1070        ExecutionResult::Halt { reason, gas_used } => {
1071            (reason.into(), 0_u64, gas_used, None, vec![])
1072        }
1073    };
1074    let gas = revm::interpreter::gas::calculate_initial_tx_gas(
1075        env.evm_env.cfg_env.spec,
1076        &env.tx.data,
1077        env.tx.kind.is_create(),
1078        env.tx.access_list.len().try_into()?,
1079        0,
1080        0,
1081    );
1082
1083    let result = match &out {
1084        Some(Output::Call(data)) => data.clone(),
1085        _ => Bytes::new(),
1086    };
1087
1088    let InspectorData {
1089        mut logs,
1090        labels,
1091        traces,
1092        line_coverage,
1093        edge_coverage,
1094        cheatcodes,
1095        chisel_state,
1096        reverter,
1097    } = inspector.collect();
1098
1099    if logs.is_empty() {
1100        logs = exec_logs;
1101    }
1102
1103    let transactions = cheatcodes
1104        .as_ref()
1105        .map(|c| c.broadcastable_transactions.clone())
1106        .filter(|txs| !txs.is_empty());
1107
1108    Ok(RawCallResult {
1109        exit_reason: Some(exit_reason),
1110        reverted: !matches!(exit_reason, return_ok!()),
1111        has_state_snapshot_failure,
1112        result,
1113        gas_used,
1114        gas_refunded,
1115        stipend: gas.initial_gas,
1116        logs,
1117        labels,
1118        traces,
1119        line_coverage,
1120        edge_coverage,
1121        transactions,
1122        state_changeset,
1123        env,
1124        cheatcodes,
1125        out,
1126        chisel_state,
1127        reverter,
1128    })
1129}

1153/// Helper struct to enable early exit behavior: when one test fails or run is interrupted,
1154/// all other tests stop early.
1155#[derive(Clone, Debug)]
1156pub struct EarlyExit {
1157    /// Shared atomic flag set to `true` when a failure occurs or ctrl-c received.
1158    /// None if running without fail-fast or show-progress.
1159    inner: Option<Arc<AtomicBool>>,
1160}
1161
1162impl EarlyExit {
1163    pub fn new(early_exit: bool) -> Self {
1164        Self { inner: early_exit.then_some(Arc::new(AtomicBool::new(false))) }
1165    }
1166
1167    /// Returns `true` if fail-fast is enabled.
1168    pub fn is_enabled(&self) -> bool {
1169        self.inner.is_some()
1170    }
1171
1172    /// Sets the exit flag. Used by other tests to stop early.
1173    pub fn record_exit(&self) {
1174        if let Some(early_exit) = &self.inner {
1175            early_exit.store(true, Ordering::Relaxed);
1176        }
1177    }

1179    /// Whether tests should stop and exit early.
1180    pub fn should_stop(&self) -> bool {
1181        self.inner.as_ref().map(|flag| flag.load(Ordering::Relaxed)).unwrap_or(false)
1182    }
