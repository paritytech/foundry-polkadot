123/// Contains data collected during invariant test runs.
124struct InvariantTestData {
125    // Consumed gas and calldata of every successful fuzz call.
126    fuzz_cases: Vec<FuzzedCases>,
127    // Data related to reverts or failed assertions of the test.
128    failures: InvariantFailures,
129    // Calldata in the last invariant run.
130    last_run_inputs: Vec<BasicTxDetails>,
131    // Additional traces for gas report.
132    gas_report_traces: Vec<Vec<CallTraceArena>>,
133    // Last call results of the invariant test.
134    last_call_results: Option<RawCallResult>,
135    // Line coverage information collected from all fuzzed calls.
136    line_coverage: Option<HitMaps>,
137    // Metrics for each fuzzed selector.
138    metrics: Map<String, InvariantMetrics>,
139
140    // Proptest runner to query for random values.
141    // The strategy only comes with the first `input`. We fill the rest of the `inputs`
142    // until the desired `depth` so we can use the evolving fuzz dictionary
143    // during the run.
144    branch_runner: TestRunner,
145}

147/// Contains invariant test data.
148struct InvariantTest {
149    // Fuzz state of invariant test.
150    fuzz_state: EvmFuzzState,
151    // Contracts fuzzed by the invariant test.
152    targeted_contracts: FuzzRunIdentifiedContracts,
153    // Data collected during invariant runs.
154    test_data: InvariantTestData,
155}
156
157impl InvariantTest {
158    /// Instantiates an invariant test.
159    fn new(
160        fuzz_state: EvmFuzzState,
161        targeted_contracts: FuzzRunIdentifiedContracts,
162        failures: InvariantFailures,
163        last_call_results: Option<RawCallResult>,
164        branch_runner: TestRunner,
165    ) -> Self {
166        let mut fuzz_cases = vec![];
167        if last_call_results.is_none() {
168            fuzz_cases.push(FuzzedCases::new(vec![]));
169        }
170        let test_data = InvariantTestData {
171            fuzz_cases,
172            failures,
173            last_run_inputs: vec![],
174            gas_report_traces: vec![],
175            last_call_results,
176            line_coverage: None,
177            metrics: Map::default(),
178            branch_runner,
179        };
180        Self { fuzz_state, targeted_contracts, test_data }
181    }
182
183    /// Returns number of invariant test reverts.
184    fn reverts(&self) -> usize {
185        self.test_data.failures.reverts
186    }
187
188    /// Whether invariant test has errors or not.
189    fn has_errors(&self) -> bool {
190        self.test_data.failures.error.is_some()
191    }
192
193    /// Set invariant test error.
194    fn set_error(&mut self, error: InvariantFuzzError) {
195        self.test_data.failures.error = Some(error);
196    }
197
198    /// Set last invariant test call results.
199    fn set_last_call_results(&mut self, call_result: Option<RawCallResult>) {
200        self.test_data.last_call_results = call_result;
201    }
202
203    /// Set last invariant run call sequence.
204    fn set_last_run_inputs(&mut self, inputs: &Vec<BasicTxDetails>) {
205        self.test_data.last_run_inputs.clone_from(inputs);
206    }
207
208    /// Merge current collected line coverage with the new coverage from last fuzzed call.
209    fn merge_line_coverage(&mut self, new_coverage: Option<HitMaps>) {
210        HitMaps::merge_opt(&mut self.test_data.line_coverage, new_coverage);
211    }
212
213    /// Update metrics for a fuzzed selector, extracted from tx details.
214    /// Always increments number of calls; discarded runs (through assume cheatcodes) are tracked
215    /// separated from reverts.
216    fn record_metrics(&mut self, tx_details: &BasicTxDetails, reverted: bool, discarded: bool) {
217        if let Some(metric_key) =
218            self.targeted_contracts.targets.lock().fuzzed_metric_key(tx_details)
219        {
220            let test_metrics = &mut self.test_data.metrics;
221            let invariant_metrics = test_metrics.entry(metric_key).or_default();
222            invariant_metrics.calls += 1;
223            if discarded {
224                invariant_metrics.discards += 1;
225            } else if reverted {
226                invariant_metrics.reverts += 1;
227            }
228        }
229    }
230
231    /// End invariant test run by collecting results, cleaning collected artifacts and reverting
232    /// created fuzz state.
233    fn end_run(&mut self, run: InvariantTestRun, gas_samples: usize) {
234        // We clear all the targeted contracts created during this run.
235        self.targeted_contracts.clear_created_contracts(run.created_contracts);
236
237        if self.test_data.gas_report_traces.len() < gas_samples {
238            self.test_data
239                .gas_report_traces
240                .push(run.run_traces.into_iter().map(|arena| arena.arena).collect());
241        }
242        self.test_data.fuzz_cases.push(FuzzedCases::new(run.fuzz_runs));
243
244        // Revert state to not persist values between runs.
245        self.fuzz_state.revert();
246    }
247}

250struct InvariantTestRun {
251    // Invariant run call sequence.
252    inputs: Vec<BasicTxDetails>,
253    // Current invariant run executor.
254    executor: Executor,
255    // Invariant run stat reports (eg. gas usage).

258    created_contracts: Vec<Address>,
259    // Traces of each call of the invariant run call sequence.
260    run_traces: Vec<SparsedTraceArena>,
261    // Current depth of invariant run.
262    depth: u32,
263    // Current assume rejects of the invariant run.
264    rejects: u32,

267}

280            new_coverage: false,
281        }

285/// Wrapper around any [`Executor`] implementer which provides fuzzing support using [`proptest`].
286///
287/// After instantiation, calling `invariant_fuzz` will proceed to hammer the deployed smart
288/// contracts with inputs, until it finds a counterexample sequence. The provided [`TestRunner`]
289/// contains all the configuration which can be overridden via [environment
290/// variables](proptest::test_runner::Config)
291pub struct InvariantExecutor<'a> {
292    pub executor: Executor,
293    /// Proptest runner.
294    runner: TestRunner,
295    /// The invariant configuration
296    config: InvariantConfig,
297    /// Contracts deployed with `setUp()`
298    setup_contracts: &'a ContractsByAddress,
299    /// Contracts that are part of the project but have not been deployed yet. We need the bytecode
300    /// to identify them from the stateset changes.
301    project_contracts: &'a ContractsByArtifact,
302    /// Filters contracts to be fuzzed through their artifact identifiers.
303    artifact_filters: ArtifactFilters,
304}
305
306impl<'a> InvariantExecutor<'a> {

308    pub fn new(
309        executor: Executor,
310        runner: TestRunner,
311        config: InvariantConfig,
312        setup_contracts: &'a ContractsByAddress,
313        project_contracts: &'a ContractsByArtifact,
314    ) -> Self {
315        Self {
316            executor,
317            runner,
318            config,
319            setup_contracts,
320            project_contracts,
321            artifact_filters: ArtifactFilters::default(),
322        }
323    }

329    /// Fuzzes any deployed contract and checks any broken invariant at `invariant_address`.
330    pub fn invariant_fuzz(
331        &mut self,
332        invariant_contract: InvariantContract<'_>,
333        fuzz_fixtures: &FuzzFixtures,
334        deployed_libs: &[Address],
335        progress: Option<&ProgressBar>,
336        early_exit: &EarlyExit,
337    ) -> Result<InvariantFuzzTestResult> {
338        // Throw an error to abort test run if the invariant function accepts input params
339        if !invariant_contract.invariant_function.inputs.is_empty() {
340            return Err(eyre!("Invariant test function should have no inputs"));
341        }
342
343        let (mut invariant_test, mut corpus_manager) =
344            self.prepare_test(&invariant_contract, fuzz_fixtures, deployed_libs)?;
345
346        // Start timer for this invariant test.
347        let mut runs = 0;
348        let timer = FuzzTestTimer::new(self.config.timeout);
349        let mut last_metrics_report = Instant::now();
350        let continue_campaign = |runs: u32| {
351            if early_exit.should_stop() {
352                return false;
353            }
354
355            if timer.is_enabled() { !timer.is_timed_out() } else { runs < self.config.runs }
356        };
357
358        // Invariant runs with edge coverage if corpus dir is set or showing edge coverage.
359        let edge_coverage_enabled = self.config.corpus.collect_edge_coverage();
360
361        'stop: while continue_campaign(runs) {
362            let initial_seq = corpus_manager.new_inputs(
363                &mut invariant_test.test_data.branch_runner,
364                &invariant_test.fuzz_state,
365                &invariant_test.targeted_contracts,
366            )?;
367
368            // Create current invariant run data.
369            let mut current_run = InvariantTestRun::new(
370                initial_seq[0].clone(),
371                // Before each run, we must reset the backend state.
372                self.executor.clone(),
373                self.config.depth as usize,
374            );
375
376            // We stop the run immediately if we have reverted, and `fail_on_revert` is set.
377            if self.config.fail_on_revert && invariant_test.reverts() > 0 {
378                return Err(eyre!("call reverted"));
379            }
380
381            while current_run.depth < self.config.depth {
382                // Check if the timeout has been reached.
383                if timer.is_timed_out() {
384                    // Since we never record a revert here the test is still considered
385                    // successful even though it timed out. We *want*
386                    // this behavior for now, so that's ok, but
387                    // future developers should be aware of this.
388                    break 'stop;
389                }
390
391                let tx = current_run
392                    .inputs
393                    .last()
394                    .ok_or_else(|| eyre!("no input generated to call fuzzed target."))?;
395                // Execute call from the randomly generated sequence without committing state.
396                // State is committed only if call is not a magic assume.
397                current_run
398                    .executor
399                    .strategy
400                    .runner
401                    .start_transaction(current_run.executor.strategy.context.as_ref());
402                let call_result = current_run
403                    .executor
404                    .call_raw(
405                        tx.sender,
406                        tx.call_details.target,
407                        tx.call_details.calldata.clone(),
408                        U256::ZERO,
409                    )
410                    .map_err(|e| eyre!(format!("Could not make raw evm call: {e}")));
411                current_run
412                    .executor
413                    .strategy
414                    .runner
415                    .rollback_transaction(current_run.executor.strategy.context.as_ref());
416
417                let mut call_result = call_result?;
418                let discarded = call_result.result.as_ref() == MAGIC_ASSUME;
419                if self.config.show_metrics {
420                    invariant_test.record_metrics(tx, call_result.reverted, discarded);
421                }
422
423                // Collect line coverage from last fuzzed call.
424                invariant_test.merge_line_coverage(call_result.line_coverage.clone());
425                // Collect edge coverage and set the flag in the current run.
426                if corpus_manager.merge_edge_coverage(&mut call_result) {
427                    current_run.new_coverage = true;
428                }
429
430                if discarded {
431                    current_run.inputs.pop();
432                    current_run.rejects += 1;
433                    if current_run.rejects > self.config.max_assume_rejects {
434                        invariant_test.set_error(InvariantFuzzError::MaxAssumeRejects(
435                            self.config.max_assume_rejects,
436                        ));
437                        break 'stop;
438                    }
439                } else {
440                    // Commit executed call result.
441                    current_run.executor.commit(&mut call_result);
442
443                    // Collect data for fuzzing from the state changeset.
444                    // This step updates the state dictionary and therefore invalidates the
445                    // ValueTree in use by the current run. This manifestsitself in proptest
446                    // observing a different input case than what it was called with, and creates
447                    // inconsistencies whenever proptest tries to use the input case after test
448                    // execution.
449                    // See <https://github.com/foundry-rs/foundry/issues/9764>.
450                    let mut state_changeset = call_result.state_changeset.clone();
451                    if !call_result.reverted {
452                        collect_data(
453                            &invariant_test,
454                            &mut state_changeset,
455                            tx,
456                            &call_result,
457                            self.config.depth,
458                        );
459                    }
460
461                    // Collect created contracts and add to fuzz targets only if targeted contracts
462                    // are updatable.
463                    if let Err(error) =
464                        &invariant_test.targeted_contracts.collect_created_contracts(
465                            &state_changeset,
466                            self.project_contracts,
467                            self.setup_contracts,
468                            &self.artifact_filters,
469                            &mut current_run.created_contracts,
470                        )
471                    {
472                        warn!(target: "forge::test", "{error}");
473                    }
474                    current_run.fuzz_runs.push(FuzzCase {
475                        calldata: tx.call_details.calldata.clone(),
476                        gas: call_result.gas_used,
477                        stipend: call_result.stipend,
478                    });
479
480                    // Determine if test can continue or should exit.
481                    let result = can_continue(
482                        &invariant_contract,
483                        &mut invariant_test,
484                        &mut current_run,
485                        &self.config,
486                        call_result,
487                        &state_changeset,
488                    )
489                    .map_err(|e| eyre!(e.to_string()))?;
490                    if !result.can_continue || current_run.depth == self.config.depth - 1 {
491                        invariant_test.set_last_run_inputs(&current_run.inputs);
492                    }
493                    // If test cannot continue then stop current run and exit test suite.
494                    if !result.can_continue {
495                        break 'stop;
496                    }
497
498                    invariant_test.set_last_call_results(result.call_result);
499                    current_run.depth += 1;
500                }
501
502                current_run.inputs.push(corpus_manager.generate_next_input(
503                    &mut invariant_test.test_data.branch_runner,
504                    &initial_seq,
505                    discarded,
506                    current_run.depth as usize,
507                )?);
508            }
509
510            // Extend corpus with current run data.
511            corpus_manager.process_inputs(&current_run.inputs, current_run.new_coverage);
512
513            // Call `afterInvariant` only if it is declared and test didn't fail already.
514            if invariant_contract.call_after_invariant && !invariant_test.has_errors() {
515                assert_after_invariant(
516                    &invariant_contract,
517                    &mut invariant_test,
518                    &current_run,
519                    &self.config,
520                )
521                .map_err(|_| eyre!("Failed to call afterInvariant"))?;
522            }
523
524            // End current invariant test run.
525            invariant_test.end_run(current_run, self.config.gas_report_samples as usize);
526            if let Some(progress) = progress {
527                // If running with progress then increment completed runs.
528                progress.inc(1);
529                // Display metrics in progress bar.
530                if edge_coverage_enabled {
531                    progress.set_message(format!("{}", &corpus_manager.metrics));
532                }
533            } else if edge_coverage_enabled
534                && last_metrics_report.elapsed() > DURATION_BETWEEN_METRICS_REPORT
535            {
536                // Display metrics inline if corpus dir set.
537                let metrics = json!({
538                    "timestamp": SystemTime::now()
539                        .duration_since(UNIX_EPOCH)?
540                        .as_secs(),
541                    "invariant": invariant_contract.invariant_function.name,
542                    "metrics": &corpus_manager.metrics,
543                });
544                let _ = sh_println!("{}", serde_json::to_string(&metrics)?);
545                last_metrics_report = Instant::now();
546            }
547
548            runs += 1;
549        }
550
551        trace!(?fuzz_fixtures);
552        invariant_test.fuzz_state.log_stats();
553
554        let result = invariant_test.test_data;
555        Ok(InvariantFuzzTestResult {
556            error: result.failures.error,
557            cases: result.fuzz_cases,
558            reverts: result.failures.reverts,
559            last_run_inputs: result.last_run_inputs,
560            gas_report_traces: result.gas_report_traces,
561            line_coverage: result.line_coverage,
562            metrics: result.metrics,
563            failed_corpus_replays: corpus_manager.failed_replays(),
564        })
565    }

567    /// Prepares certain structures to execute the invariant tests:
568    /// * Invariant Fuzz Test.
569    /// * Invariant Corpus Manager.
570    fn prepare_test(
571        &mut self,
572        invariant_contract: &InvariantContract<'_>,
573        fuzz_fixtures: &FuzzFixtures,
574        deployed_libs: &[Address],
575    ) -> Result<(InvariantTest, CorpusManager)> {
576        // Finds out the chosen deployed contracts and/or senders.
577        self.select_contract_artifacts(invariant_contract.address)?;
578        let (targeted_senders, targeted_contracts) =
579            self.select_contracts_and_senders(invariant_contract.address)?;
580
581        // Stores fuzz state for use with [fuzz_calldata_from_state].
582        let inspector = self.executor.inspector();
583        let fuzz_state = EvmFuzzState::new(
584            self.executor.backend().mem_db(),
585            self.config.dictionary,
586            deployed_libs,
587            inspector.analysis.as_ref(),
588            inspector.paths_config(),
589        );
590
591        // Creates the invariant strategy.
592        let strategy = invariant_strat(
593            fuzz_state.clone(),
594            targeted_senders,
595            targeted_contracts.clone(),
596            self.config.dictionary.dictionary_weight,
597            fuzz_fixtures.clone(),
598            self.config.max_fuzz_int,
599        )
600        .no_shrink();
601
602        // Allows `override_call_strat` to use the address given by the Fuzzer inspector during
603        // EVM execution.
604        let mut call_generator = None;
605        if self.config.call_override {
606            let target_contract_ref = Arc::new(RwLock::new(Address::ZERO));
607
608            call_generator = Some(RandomCallGenerator::new(
609                invariant_contract.address,
610                self.runner.clone(),
611                override_call_strat(
612                    fuzz_state.clone(),
613                    targeted_contracts.clone(),
614                    target_contract_ref.clone(),
615                    fuzz_fixtures.clone(),
616                    self.config.max_fuzz_int,
617                ),
618                target_contract_ref,
619            ));
620        }
621
622        // If any of the targeted contracts have the storage layout enabled then we can sample
623        // mapping values. To accomplish, we need to record the mapping storage slots and keys.
624        let fuzz_state =
625            if targeted_contracts.targets.lock().iter().any(|(_, t)| t.storage_layout.is_some()) {
626                fuzz_state.with_mapping_slots(AddressMap::default())
627            } else {
628                fuzz_state
629            };
630
631        self.executor.inspector_mut().set_fuzzer(Fuzzer {
632            call_generator,
633            fuzz_state: fuzz_state.clone(),
634            collect: true,
635        });
636
637        // Let's make sure the invariant is sound before actually starting the run:
638        // We'll assert the invariant in its initial state, and if it fails, we'll
639        // already know if we can early exit the invariant run.
640        // This does not count as a fuzz run. It will just register the revert.
641        let mut failures = InvariantFailures::new();
642        let last_call_results = assert_invariants(
643            invariant_contract,
644            &self.config,
645            &targeted_contracts,
646            &self.executor,
647            &[],
648            &mut failures,
649        )?;
650        if let Some(error) = failures.error {
651            return Err(eyre!(error.revert_reason().unwrap_or_default()));
652        }
653
654        let corpus_manager = CorpusManager::new(
655            self.config.corpus.clone(),
656            strategy.boxed(),
657            &self.executor,
658            None,
659            Some(&targeted_contracts),
660            self.config.max_fuzz_int,
661        )?;
662        let invariant_test = InvariantTest::new(
663            fuzz_state,
664            targeted_contracts,
665            failures,
666            last_call_results,
667            self.runner.clone(),
668        );
669
670        Ok((invariant_test, corpus_manager))
671    }

673    /// Fills the `InvariantExecutor` with the artifact identifier filters (in `path:name` string
674    /// format). They will be used to filter contracts after the `setUp`, and more importantly,
675    /// during the runs.
676    ///
677    /// Also excludes any contract without any mutable functions.
678    ///
679    /// Priority:
680    ///
681    /// targetArtifactSelectors > excludeArtifacts > targetArtifacts
682    pub fn select_contract_artifacts(&mut self, invariant_address: Address) -> Result<()> {
683        let targeted_artifact_selectors = self
684            .executor
685            .call_sol_default(invariant_address, &IInvariantTest::targetArtifactSelectorsCall {});
686
687        // Insert them into the executor `targeted_abi`.
688        for IInvariantTest::FuzzArtifactSelector { artifact, selectors } in
689            targeted_artifact_selectors
690        {
691            let identifier = self.validate_selected_contract(artifact, &selectors)?;
692            self.artifact_filters.targeted.entry(identifier).or_default().extend(selectors);
693        }
694
695        let targeted_artifacts = self
696            .executor
697            .call_sol_default(invariant_address, &IInvariantTest::targetArtifactsCall {});
698        let excluded_artifacts = self
699            .executor
700            .call_sol_default(invariant_address, &IInvariantTest::excludeArtifactsCall {});
701
702        // Insert `excludeArtifacts` into the executor `excluded_abi`.
703        for contract in excluded_artifacts {
704            let identifier = self.validate_selected_contract(contract, &[])?;
705
706            if !self.artifact_filters.excluded.contains(&identifier) {
707                self.artifact_filters.excluded.push(identifier);
708            }
709        }
710
711        // Exclude any artifact without mutable functions.
712        for (artifact, contract) in self.project_contracts.iter() {
713            if contract
714                .abi
715                .functions()
716                .filter(|func| {
717                    !matches!(
718                        func.state_mutability,
719                        alloy_json_abi::StateMutability::Pure
720                            | alloy_json_abi::StateMutability::View
721                    )
722                })
723                .count()
724                == 0
725                && !self.artifact_filters.excluded.contains(&artifact.identifier())
726            {
727                self.artifact_filters.excluded.push(artifact.identifier());
728            }
729        }
730
731        // Insert `targetArtifacts` into the executor `targeted_abi`, if they have not been seen
732        // before.
733        for contract in targeted_artifacts {
734            let identifier = self.validate_selected_contract(contract, &[])?;
735
736            if !self.artifact_filters.targeted.contains_key(&identifier)
737                && !self.artifact_filters.excluded.contains(&identifier)
738            {
739                self.artifact_filters.targeted.insert(identifier, vec![]);
740            }
741        }
742        Ok(())
743    }

745    /// Makes sure that the contract exists in the project. If so, it returns its artifact
746    /// identifier.
747    fn validate_selected_contract(
748        &mut self,
749        contract: String,
750        selectors: &[FixedBytes<4>],
751    ) -> Result<String> {
752        if let Some((artifact, contract_data)) =
753            self.project_contracts.find_by_name_or_identifier(&contract)?
754        {
755            // Check that the selectors really exist for this contract.
756            for selector in selectors {
757                contract_data
758                    .abi
759                    .functions()
760                    .find(|func| func.selector().as_slice() == selector.as_slice())
761                    .wrap_err(format!("{contract} does not have the selector {selector:?}"))?;
762            }
763
764            return Ok(artifact.identifier());
765        }
766        eyre::bail!(
767            "{contract} not found in the project. Allowed format: `contract_name` or `contract_path:contract_name`."
768        );
769    }

771    /// Selects senders and contracts based on the contract methods `targetSenders() -> address[]`,
772    /// `targetContracts() -> address[]` and `excludeContracts() -> address[]`.
773    pub fn select_contracts_and_senders(
774        &self,
775        to: Address,
776    ) -> Result<(SenderFilters, FuzzRunIdentifiedContracts)> {
777        let targeted_senders =
778            self.executor.call_sol_default(to, &IInvariantTest::targetSendersCall {});
779        let mut excluded_senders =
780            self.executor.call_sol_default(to, &IInvariantTest::excludeSendersCall {});
781        // Extend with default excluded addresses - https://github.com/foundry-rs/foundry/issues/4163
782        excluded_senders.extend([
783            CHEATCODE_ADDRESS,
784            HARDHAT_CONSOLE_ADDRESS,
785            DEFAULT_CREATE2_DEPLOYER,
786        ]);
787        // Extend with precompiles - https://github.com/foundry-rs/foundry/issues/4287
788        excluded_senders.extend(PRECOMPILES);
789        let sender_filters = SenderFilters::new(targeted_senders, excluded_senders);
790
791        let selected = self.executor.call_sol_default(to, &IInvariantTest::targetContractsCall {});
792        let excluded = self.executor.call_sol_default(to, &IInvariantTest::excludeContractsCall {});
793
794        let contracts = self
795            .setup_contracts
796            .iter()
797            .filter(|&(addr, (identifier, _))| {
798                // Include to address if explicitly set as target.
799                if *addr == to && selected.contains(&to) {
800                    return true;
801                }
802
803                *addr != to
804                    && *addr != CHEATCODE_ADDRESS
805                    && *addr != HARDHAT_CONSOLE_ADDRESS
806                    && (selected.is_empty() || selected.contains(addr))
807                    && (excluded.is_empty() || !excluded.contains(addr))
808                    && self.artifact_filters.matches(identifier)
809            })
810            .map(|(addr, (identifier, abi))| {
811                (
812                    *addr,
813                    TargetedContract::new(identifier.clone(), abi.clone())
814                        .with_project_contracts(self.project_contracts),
815                )
816            })
817            .collect();
818        let mut contracts = TargetedContracts { inner: contracts };
819
820        self.target_interfaces(to, &mut contracts)?;
821
822        self.select_selectors(to, &mut contracts)?;
823
824        // There should be at least one contract identified as target for fuzz runs.
825        if contracts.is_empty() {
826            eyre::bail!("No contracts to fuzz.");
827        }
828
829        Ok((sender_filters, FuzzRunIdentifiedContracts::new(contracts, selected.is_empty())))
830    }

832    /// Extends the contracts and selectors to fuzz with the addresses and ABIs specified in
833    /// `targetInterfaces() -> (address, string[])[]`. Enables targeting of addresses that are
834    /// not deployed during `setUp` such as when fuzzing in a forked environment. Also enables
835    /// targeting of delegate proxies and contracts deployed with `create` or `create2`.
836    pub fn target_interfaces(
837        &self,
838        invariant_address: Address,
839        targeted_contracts: &mut TargetedContracts,
840    ) -> Result<()> {
841        let interfaces = self
842            .executor
843            .call_sol_default(invariant_address, &IInvariantTest::targetInterfacesCall {});
844
845        // Since `targetInterfaces` returns a tuple array there is no guarantee
846        // that the addresses are unique this map is used to merge functions of
847        // the specified interfaces for the same address. For example:
848        // `[(addr1, ["IERC20", "IOwnable"])]` and `[(addr1, ["IERC20"]), (addr1, ("IOwnable"))]`
849        // should be equivalent.
850        let mut combined = TargetedContracts::new();
851
852        // Loop through each address and its associated artifact identifiers.
853        // We're borrowing here to avoid taking full ownership.
854        for IInvariantTest::FuzzInterface { addr, artifacts } in &interfaces {
855            // Identifiers are specified as an array, so we loop through them.
856            for identifier in artifacts {
857                // Try to find the contract by name or identifier in the project's contracts.
858                if let Some((_, contract_data)) =
859                    self.project_contracts.iter().find(|(artifact, _)| {
860                        &artifact.name == identifier || &artifact.identifier() == identifier
861                    })
862                {
863                    let abi = &contract_data.abi;
864                    combined
865                        // Check if there's an entry for the given key in the 'combined' map.
866                        .entry(*addr)
867                        // If the entry exists, extends its ABI with the function list.
868                        .and_modify(|entry| {
869                            // Extend the ABI's function list with the new functions.
870                            entry.abi.functions.extend(abi.functions.clone());
871                        })
872                        // Otherwise insert it into the map.
873                        .or_insert_with(|| {
874                            let mut contract =
875                                TargetedContract::new(identifier.to_string(), abi.clone());
876                            contract.storage_layout =
877                                contract_data.storage_layout.as_ref().map(Arc::clone);
878                            contract
879                        });
880                }
881            }
882        }
883
884        targeted_contracts.extend(combined.inner);
885
886        Ok(())
887    }

889    /// Selects the functions to fuzz based on the contract method `targetSelectors()` and
890    /// `targetArtifactSelectors()`.
891    pub fn select_selectors(
892        &self,
893        address: Address,
894        targeted_contracts: &mut TargetedContracts,
895    ) -> Result<()> {
896        for (address, (identifier, _)) in self.setup_contracts {
897            if let Some(selectors) = self.artifact_filters.targeted.get(identifier) {
898                self.add_address_with_functions(*address, selectors, false, targeted_contracts)?;
899            }
900        }
901
902        let mut target_test_selectors = vec![];
903        let mut excluded_test_selectors = vec![];
904
905        // Collect contract functions marked as target for fuzzing campaign.
906        let selectors =
907            self.executor.call_sol_default(address, &IInvariantTest::targetSelectorsCall {});
908        for IInvariantTest::FuzzSelector { addr, selectors } in selectors {
909            if addr == address {
910                target_test_selectors = selectors.clone();
911            }
912            self.add_address_with_functions(addr, &selectors, false, targeted_contracts)?;
913        }
914
915        // Collect contract functions excluded from fuzzing campaign.
916        let excluded_selectors =
917            self.executor.call_sol_default(address, &IInvariantTest::excludeSelectorsCall {});
918        for IInvariantTest::FuzzSelector { addr, selectors } in excluded_selectors {
919            if addr == address {
920                // If fuzz selector address is the test contract, then record selectors to be
921                // later excluded if needed.
922                excluded_test_selectors = selectors.clone();
923            }
924            self.add_address_with_functions(addr, &selectors, true, targeted_contracts)?;
925        }
926
927        if target_test_selectors.is_empty()
928            && let Some(target) = targeted_contracts.get(&address)
929        {
930            // If test contract is marked as a target and no target selector explicitly set, then
931            // include only state-changing functions that are not reserved and selectors that are
932            // not explicitly excluded.
933            let selectors: Vec<_> = target
934                .abi
935                .functions()
936                .filter_map(|func| {
937                    if matches!(
938                        func.state_mutability,
939                        alloy_json_abi::StateMutability::Pure
940                            | alloy_json_abi::StateMutability::View
941                    ) || func.is_reserved()
942                        || excluded_test_selectors.contains(&func.selector())
943                    {
944                        None
945                    } else {
946                        Some(func.selector())
947                    }
948                })
949                .collect();
950            self.add_address_with_functions(address, &selectors, false, targeted_contracts)?;
951        }
952
953        Ok(())
954    }
955
956    /// Adds the address and fuzzed or excluded functions to `TargetedContracts`.
957    fn add_address_with_functions(
958        &self,
959        address: Address,
960        selectors: &[Selector],
961        should_exclude: bool,
962        targeted_contracts: &mut TargetedContracts,
963    ) -> eyre::Result<()> {
964        // Do not add address in target contracts if no function selected.
965        if selectors.is_empty() {
966            return Ok(());
967        }
968
969        let contract = match targeted_contracts.entry(address) {
970            Entry::Occupied(entry) => entry.into_mut(),
971            Entry::Vacant(entry) => {
972                let (identifier, abi) = self.setup_contracts.get(&address).ok_or_else(|| {
973                    eyre::eyre!(
974                        "[{}] address does not have an associated contract: {}",
975                        if should_exclude { "excludeSelectors" } else { "targetSelectors" },
976                        address
977                    )
978                })?;
979                entry.insert(
980                    TargetedContract::new(identifier.clone(), abi.clone())
981                        .with_project_contracts(self.project_contracts),
982                )
983            }
984        };
985        contract.add_selectors(selectors.iter().copied(), should_exclude)?;
986        Ok(())
987    }
988}

990/// Collects data from call for fuzzing. However, it first verifies that the sender is not an EOA
991/// before inserting it into the dictionary. Otherwise, we flood the dictionary with
992/// randomly generated addresses.
993fn collect_data(
994    invariant_test: &InvariantTest,
995    state_changeset: &mut HashMap<Address, Account>,
996    tx: &BasicTxDetails,
997    call_result: &RawCallResult,
998    run_depth: u32,
999) {
1000    // Verify it has no code.
1001    let mut has_code = false;
1002    if let Some(Some(code)) =
1003        state_changeset.get(&tx.sender).map(|account| account.info.code.as_ref())
1004    {
1005        has_code = !code.is_empty();
1006    }
1007
1008    // We keep the nonce changes to apply later.
1009    let mut sender_changeset = None;
1010    if !has_code {
1011        sender_changeset = state_changeset.remove(&tx.sender);
1012    }
1013
1014    // Collect values from fuzzed call result and add them to fuzz dictionary.
1015    invariant_test.fuzz_state.collect_values_from_call(
1016        &invariant_test.targeted_contracts,
1017        tx,
1018        &call_result.result,
1019        &call_result.logs,
1020        &*state_changeset,
1021        run_depth,
1022    );
1023
1024    // Re-add changes
1025    if let Some(changed) = sender_changeset {
1026        state_changeset.insert(tx.sender, changed);
1027    }
1028}

1030/// Calls the `afterInvariant()` function on a contract.
1031/// Returns call result and if call succeeded.
1032/// The state after the call is not persisted.
1033pub(crate) fn call_after_invariant_function(
1034    executor: &Executor,
1035    to: Address,
1036) -> Result<(RawCallResult, bool), EvmError> {
1037    let calldata = Bytes::from_static(&IInvariantTest::afterInvariantCall::SELECTOR);
1038    let mut call_result = executor.call_raw(CALLER, to, calldata, U256::ZERO)?;
1039    let success = executor.is_raw_call_mut_success(to, &mut call_result, false);
1040    Ok((call_result, success))
1041}
