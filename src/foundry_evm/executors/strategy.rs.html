20/// Context for [ExecutorStrategy].
21pub trait ExecutorStrategyContext: Debug + Send + Sync + Any {
22    /// Clone the strategy context.
23    fn new_cloned(&self) -> Box<dyn ExecutorStrategyContext>;
24    /// Alias as immutable reference of [Any].
25    fn as_any_ref(&self) -> &dyn Any;
26    /// Alias as mutable reference of [Any].
27    fn as_any_mut(&mut self) -> &mut dyn Any;
28}
29
30impl Clone for Box<dyn ExecutorStrategyContext> {
31    fn clone(&self) -> Self {
32        self.new_cloned()
33    }
34}
35
36/// Default strategy context object.
37impl ExecutorStrategyContext for () {
38    fn new_cloned(&self) -> Box<dyn ExecutorStrategyContext> {
39        Box::new(())
40    }
41
42    fn as_any_ref(&self) -> &dyn Any {
43        self
44    }
45
46    fn as_any_mut(&mut self) -> &mut dyn Any {
47        self
48    }
49}

51/// Stateless strategy runner for [ExecutorStrategy].
52pub trait ExecutorStrategyRunner: Debug + Send + Sync + ExecutorStrategyExt {
53    /// Creates a new [BackendStrategy].
54    fn new_backend_strategy(&self, ctx: &dyn ExecutorStrategyContext) -> BackendStrategy;
55
56    /// Creates a new [CheatcodesStrategy].
57    fn new_cheatcodes_strategy(&self, ctx: &dyn ExecutorStrategyContext) -> CheatcodesStrategy;
58
59    /// Set the balance of an account.
60    fn set_balance(
61        &self,
62        executor: &mut Executor,
63        address: Address,
64        amount: U256,
65    ) -> BackendResult<()>;
66
67    /// Gets the balance of an account
68    fn get_balance(&self, executor: &mut Executor, address: Address) -> BackendResult<U256>;
69
70    /// Set the nonce of an account.
71    fn set_nonce(&self, executor: &mut Executor, address: Address, nonce: u64)
72    -> BackendResult<()>;
73
74    /// Returns the nonce of an account.
75    fn get_nonce(&self, executor: &mut Executor, address: Address) -> BackendResult<u64>;
76
77    /// Execute a transaction and *WITHOUT* applying state changes.
78    fn call(
79        &self,
80        ctx: &dyn ExecutorStrategyContext,
81        backend: &mut CowBackend<'_>,
82        env: &mut Env,
83        executor_env: &Env,
84        inspector: &mut InspectorStack,
85    ) -> Result<ResultAndState>;
86
87    /// Execute a transaction and apply state changes.
88    fn transact(
89        &self,
90        ctx: &mut dyn ExecutorStrategyContext,
91        backend: &mut Backend,
92        env: &mut Env,
93        executor_env: &Env,
94        inspector: &mut InspectorStack,
95    ) -> Result<ResultAndState>;
96}

145/// Implements [ExecutorStrategyRunner] for EVM.
146#[derive(Debug, Default, Clone)]
147pub struct EvmExecutorStrategyRunner;
148
149impl ExecutorStrategyRunner for EvmExecutorStrategyRunner {
150    fn set_balance(
151        &self,
152        executor: &mut Executor,
153        address: Address,
154        amount: U256,
155    ) -> BackendResult<()> {
156        let mut account = executor.backend().basic_ref(address)?.unwrap_or_default();
157        account.balance = amount;
158        executor.backend_mut().insert_account_info(address, account);
159        Ok(())
160    }
161
162    fn get_balance(&self, executor: &mut Executor, address: Address) -> BackendResult<U256> {
163        Ok(executor.backend().basic_ref(address)?.map(|acc| acc.balance).unwrap_or_default())
164    }
165
166    fn set_nonce(
167        &self,
168        executor: &mut Executor,
169        address: Address,
170        nonce: u64,
171    ) -> BackendResult<()> {
172        let mut account = executor.backend().basic_ref(address)?.unwrap_or_default();
173        account.nonce = nonce;
174        executor.backend_mut().insert_account_info(address, account);
175        Ok(())
176    }
177
178    fn get_nonce(&self, executor: &mut Executor, address: Address) -> BackendResult<u64> {
179        Ok(executor.backend().basic_ref(address)?.map(|acc| acc.nonce).unwrap_or_default())
180    }
181
182    fn call(
183        &self,
184        _ctx: &dyn ExecutorStrategyContext,
185        backend: &mut CowBackend<'_>,
186        env: &mut Env,
187        _executor_env: &Env,
188        inspector: &mut InspectorStack,
189    ) -> Result<ResultAndState> {
190        backend.inspect(env, inspector, Box::new(()))
191    }
192
193    fn transact(
194        &self,
195        _ctx: &mut dyn ExecutorStrategyContext,
196        backend: &mut Backend,
197        env: &mut Env,
198        _executor_env: &Env,
199        inspector: &mut InspectorStack,
200    ) -> Result<ResultAndState> {
201        backend.inspect(env, inspector, Box::new(()))
202    }
203
204    fn new_backend_strategy(&self, _ctx: &dyn ExecutorStrategyContext) -> BackendStrategy {
205        BackendStrategy::new_evm()
206    }
207
208    fn new_cheatcodes_strategy(&self, _ctx: &dyn ExecutorStrategyContext) -> CheatcodesStrategy {
209        CheatcodesStrategy::new_evm()
210    }
211}

215/// Defines the strategy for an [Executor].
216#[derive(Debug)]
217pub struct ExecutorStrategy {
218    /// Strategy runner.
219    pub runner: &'static dyn ExecutorStrategyRunner,
220    /// Strategy context.
221    pub context: Box<dyn ExecutorStrategyContext>,
222}
223
224impl ExecutorStrategy {
225    /// Creates a new EVM strategy for the [Executor].
226    pub fn new_evm() -> Self {
227        Self { runner: &EvmExecutorStrategyRunner, context: Box::new(()) }
228    }
229}
230
231impl Clone for ExecutorStrategy {
232    fn clone(&self) -> Self {
233        Self { runner: self.runner, context: self.context.clone() }
234    }
