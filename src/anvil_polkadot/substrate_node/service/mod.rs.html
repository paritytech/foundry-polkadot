53/// Builds a new service for a full client.
54pub fn new(
55    anvil_config: &AnvilNodeConfig,
56    config: Configuration,
57) -> Result<(Service, TaskManager), ServiceError> {
58    let storage_overrides =
59        Arc::new(Mutex::new(StorageOverrides::new(anvil_config.revive_rpc_block_limit)));
60
61    let (client, backend, keystore, mut task_manager) = client::new_client(
62        anvil_config.get_genesis_number(),
63        &config,
64        sc_service::new_wasm_executor(&config.executor),
65        storage_overrides.clone(),
66    )?;
67
68    let transaction_pool = Arc::from(
69        sc_transaction_pool::Builder::new(
70            task_manager.spawn_essential_handle(),
71            client.clone(),
72            config.role.is_authority().into(),
73        )
74        .with_options(config.transaction_pool.clone())
75        .build(),
76    );
77
78    // Inform the tx pool about imported and finalized blocks.
79    task_manager.spawn_handle().spawn(
80        "txpool-notifications",
81        Some("transaction-pool"),
82        sc_transaction_pool::notification_future(client.clone(), transaction_pool.clone()),
83    );
84
85    let (seal_engine_command_sender, commands_stream) = tokio::sync::mpsc::channel(1024);
86    let commands_stream = ReceiverStream::new(commands_stream);
87
88    let mining_mode =
89        MiningMode::new(anvil_config.block_time, anvil_config.mixed_mining, anvil_config.no_mining);
90    let time_manager = Arc::new(TimeManager::new_with_milliseconds(
91        sp_timestamp::Timestamp::from(
92            anvil_config
93                .get_genesis_timestamp()
94                .checked_mul(1000)
95                .ok_or(ServiceError::Application("Genesis timestamp overflow".into()))?,
96        )
97        .into(),
98    ));
99
100    let mining_engine =
101        Arc::new(MiningEngine::new(mining_mode, transaction_pool.clone(), time_manager.clone()));
102
103    let rpc_handlers = spawn_rpc_server(
104        anvil_config.get_genesis_number(),
105        &mut task_manager,
106        client.clone(),
107        config,
108        transaction_pool.clone(),
109        keystore,
110        backend.clone(),
111    )?;
112
113    let proposer = sc_basic_authorship::ProposerFactory::new(
114        task_manager.spawn_handle(),
115        client.clone(),
116        transaction_pool.clone(),
117        None,
118        None,
119    );
120
121    let create_inherent_data_providers = {
122        move |_, ()| {
123            let next_timestamp = time_manager.next_timestamp();
124            async move { Ok(sp_timestamp::InherentDataProvider::new(next_timestamp.into())) }
125        }
126    };
127
128    let params = sc_consensus_manual_seal::ManualSealParams {
129        block_import: client.clone(),
130        env: proposer,
131        client: client.clone(),
132        pool: transaction_pool.clone(),
133        select_chain: SelectChain::new(backend.clone()),
134        commands_stream: Box::pin(commands_stream),
135        consensus_data_provider: Some(Box::new(SameSlotConsensusDataProvider::new())),
136        create_inherent_data_providers,
137    };
138    let authorship_future = sc_consensus_manual_seal::run_manual_seal(params);
139
140    task_manager.spawn_essential_handle().spawn_blocking(
141        "manual-seal",
142        "substrate",
143        authorship_future,
144    );
145
146    Ok((
147        Service {
148            spawn_handle: task_manager.spawn_handle(),
149            client,
150            backend,
151            tx_pool: transaction_pool,
152            rpc_handlers,
153            mining_engine,
154            storage_overrides,
155            genesis_block_number: anvil_config.get_genesis_number(),
156            seal_command_sender: seal_engine_command_sender,
157        },
158        task_manager,
159    ))
160}
