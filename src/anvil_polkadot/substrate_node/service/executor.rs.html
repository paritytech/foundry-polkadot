36/// State-injecting executor implementation.
37pub struct Executor {
38    inner: InnerLocalCallExecutor,
39    storage_overrides: Arc<Mutex<StorageOverrides>>,
40    backend: Arc<Backend>,
41}
42
43impl Executor {
44    pub fn new(
45        inner: InnerLocalCallExecutor,
46        storage_overrides: Arc<Mutex<StorageOverrides>>,
47        backend: Arc<Backend>,
48    ) -> Self {
49        Self { inner, storage_overrides, backend }
50    }
51
52    fn apply_overrides(&self, hash: &Hash, overlay: &mut OverlayedChanges<HashingFor<Block>>) {
53        let overrides = { self.storage_overrides.lock().get(hash) };
54        let Some(overrides) = overrides else { return };
55
56        Self::apply_block_overrides(overrides, overlay);
57    }
58
59    /// Take and apply one-shot pending overrides from `StorageOverrides`.
60    /// Called during `Core_initialize_block` for actual block production.
61    fn apply_and_consume_pending_overrides(
62        &self,
63        overlay: &mut OverlayedChanges<HashingFor<Block>>,
64    ) {
65        let pending = self.storage_overrides.lock().take_pending();
66        let Some(pending) = pending else { return };
67
68        Self::apply_block_overrides(pending, overlay);
69    }
70
71    fn apply_block_overrides(
72        overrides: BlockOverrides,
73        overlay: &mut OverlayedChanges<HashingFor<Block>>,
74    ) {
75        for (key, val) in overrides.top {
76            overlay.set_storage(key, val);
77        }
78
79        for (child_key, child_map) in overrides.children {
80            let child_info = ChildInfo::new_default_from_vec(child_key);
81
82            for (key, val) in child_map {
83                overlay.set_child_storage(&child_info, key, val);
84            }
85        }
86    }
87}

89impl CallExecutor<Block> for Executor {
90    type Error = <InnerLocalCallExecutor as CallExecutor<Block>>::Error;
91
92    type Backend = Backend;
93
94    fn execution_extensions(&self) -> &ExecutionExtensions<Block> {
95        self.inner.execution_extensions()
96    }
97
98    fn call(
99        &self,
100        at_hash: Hash,
101        method: &str,
102        call_data: &[u8],
103        context: CallContext,
104    ) -> sp_blockchain::Result<Vec<u8>> {
105        let at_number =
106            self.backend.blockchain().expect_block_number_from_id(&BlockId::Hash(at_hash))?;
107        let extensions = self.execution_extensions().extensions(at_hash, at_number);
108
109        let mut changes = OverlayedChanges::default();
110
111        self.apply_overrides(&at_hash, &mut changes);
112
113        self.contextual_call(
114            at_hash,
115            method,
116            call_data,
117            &RefCell::new(changes),
118            &None,
119            context,
120            &RefCell::new(extensions),
121        )
122    }
123
124    fn contextual_call(
125        &self,
126        at_hash: Hash,
127        method: &str,
128        call_data: &[u8],
129        changes: &RefCell<OverlayedChanges<HashingFor<Block>>>,
130        recorder: &Option<ProofRecorder<Block>>,
131        call_context: CallContext,
132        extensions: &RefCell<sp_externalities::Extensions>,
133    ) -> Result<Vec<u8>, sp_blockchain::Error> {
134        let is_block_init =
135            method == "Core_initialize_block" && call_context == CallContext::Onchain;
136        let apply_overrides = is_block_init || call_context == CallContext::Offchain;
137
138        if apply_overrides {
139            self.apply_overrides(&at_hash, &mut changes.borrow_mut());
140        }
141
142        // Apply and consume pending overrides only during actual block production.
143        if is_block_init {
144            self.apply_and_consume_pending_overrides(&mut changes.borrow_mut());
145        }
146
147        self.inner.contextual_call(
148            at_hash,
149            method,
150            call_data,
151            changes,
152            recorder,
153            call_context,
154            extensions,
155        )
156    }
157
158    fn runtime_version(&self, at_hash: Hash) -> sp_blockchain::Result<RuntimeVersion> {
159        CallExecutor::runtime_version(&self.inner, at_hash)
160    }
161
162    fn prove_execution(
163        &self,
164        at_hash: Hash,
165        method: &str,
166        call_data: &[u8],
167    ) -> sp_blockchain::Result<(Vec<u8>, StorageProof)> {
168        self.inner.prove_execution(at_hash, method, call_data)
169    }
170}
