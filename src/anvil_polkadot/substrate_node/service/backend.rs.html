37    #[error("Could not find the next fee multiplier in the state")]

43    #[error("Unable to decode chain id {0}")]
44    DecodeChainId(codec::Error),
45    #[error("Unable to decode balance {0}")]
46    DecodeBalance(codec::Error),

48    DecodeReviveAccountInfo(codec::Error),

101    pub fn read_aura_authority(&self, hash: Hash) -> Result<AccountId> {
102        let key = well_known_keys::AURA_AUTHORITIES;
103
104        let value =
105            self.read_top_state(hash, key.to_vec())?.ok_or(BackendError::MissingAuraAuthorities)?;
106        let authorities = <Vec<[u8; 32]>>::decode(&mut &value[..])
107            .map_err(BackendError::DecodeAuraAuthorities)?;
108        // Read the first authority, since that's what we modify via RPC, or at genesis,
109        // and instruct the runtime to pick via the consensus data provider, whenever it
110        // needs the block author.
111        let authority = *authorities.first().ok_or(BackendError::MissingAuraAuthorities)?;
112        Ok(authority.into())
113    }

176    /// Store a one-shot `pallet_timestamp::Now` override for the **next** mined block.
177    /// Unlike `inject_timestamp`, this does not modify any existing block's observable state.
178    pub fn set_pending_timestamp(&self, desired_timestamp: u64) {
179        let min_period = <substrate_runtime::Runtime as polkadot_sdk::pallet_timestamp::Config>::MinimumPeriod::get();
180        // Compensate for the minimum period to ensure the timestamp is as expected.
181        let override_value = desired_timestamp.saturating_sub(min_period);
182
183        let mut overrides = self.overrides.lock();
184        let mut changeset = BlockOverrides::default();
185        changeset.top.insert(well_known_keys::TIMESTAMP.to_vec(), Some(override_value.encode()));
186        overrides.set_pending(changeset);
187    }

240    fn read_top_state(&self, hash: Hash, key: StorageKey) -> Result<Option<StorageValue>> {
241        let maybe_overridden_val = {
242            let mut guard = self.overrides.lock();
243
244            guard.per_block.get(&hash).and_then(|overrides| overrides.top.get(&key).cloned())
245        };
246
247        if let Some(overridden_val) = maybe_overridden_val {
248            return Ok(overridden_val);
249        }
250
251        let state = self.backend.state_at(hash, TrieCacheContext::Trusted)?;
252        Ok(state
253            .storage(key.as_slice())
254            .map_err(|e| sp_blockchain::Error::from_state(Box::new(e)))?)
255    }

258pub type Storage = HashMap<StorageKey, Option<StorageValue>>;

260#[derive(Default, Clone)]
261pub struct BlockOverrides {
262    pub top: Storage,
263    pub children: HashMap<StorageKey, Storage>,
264}
265
266pub struct StorageOverrides {
267    // We keep N most recently used block state overrides because we may later get RPC calls which
268    // query the state of past blocks. When state is mutated by the `set_*` RPCs, it gets committed
269    // to the state DB only in the next block.
270    per_block: LruCache<Hash, BlockOverrides>,
271    // One-shot overrides applied at the next `Core_initialize_block` and then consumed.
272    // It can be used to affect the state of the pending block.
273    pending: Option<BlockOverrides>,
274}
275
276impl StorageOverrides {
277    pub fn new(lru_capacity: Option<usize>) -> Self {
278        Self {
279            per_block: LruCache::new(
280                lru_capacity.and_then(NonZeroUsize::new).unwrap_or(DEFAULT_LRU_CAP),
281            ),
282            pending: None,
283        }
284    }
285
286    pub fn get(&mut self, block: &Hash) -> Option<BlockOverrides> {
287        self.per_block.get(block).cloned()
288    }
289
290    pub fn set_pending(&mut self, overrides: BlockOverrides) {
291        self.pending = Some(overrides);
292    }
293
294    pub fn take_pending(&mut self) -> Option<BlockOverrides> {
295        self.pending.take()
296    }

301
302        self.add(latest_block, changeset);
303    }

387    fn set_immutable_data(&mut self, latest_block: Hash, address: Address, data: Vec<u8>) {
388        let mut changeset = BlockOverrides::default();
389
390        // SCALE-encode as BoundedVec<u8>: compact_length prefix followed by raw bytes
391        let mut encoded = codec::Compact(data.len() as u32).encode();
392        encoded.extend_from_slice(&data);
393
394        changeset.top.insert(
395            well_known_keys::immutable_data_of(H160::from_slice(address.as_slice())),
396            Some(encoded),
397        );
398
399        self.add(latest_block, changeset);
400    }
401
402    fn set_child_storage(
403        &mut self,
404        latest_block: Hash,
405        child_key: StorageKey,
406        key: StorageKey,
407        value: StorageValue,
408    ) {
409        let mut changeset = BlockOverrides::default();
410
411        let mut child_map = Storage::with_capacity(1);
412        child_map.insert(blake2_256(key.as_slice()).to_vec(), Some(value));
413
414        changeset.children.insert(child_key, child_map);
415
416        self.add(latest_block, changeset);
417    }
418
419    fn add(&mut self, latest_block: Hash, changeset: BlockOverrides) {
420        if let Some(per_block) = self.per_block.get_mut(&latest_block) {
421            per_block.top.extend(changeset.top);
422
423            for (child_key, child_map) in changeset.children {
424                per_block.children.entry(child_key).or_default().extend(child_map);
425            }
426        } else {
427            self.per_block.put(latest_block, changeset);
428        }
429    }
