23// Errors that can happen during the block production.
24#[derive(Debug, thiserror::Error)]
25pub enum MiningError {
26    #[error("Block production failed: {0}")]
27    BlockProducing(#[from] BlockProducingError),
28    #[error("Current mining mode can not answer this query.")]
29    MiningModeMismatch,
30    #[error("Current timestamp is newer.")]
31    Timestamp,
32    #[error("Closed channel")]
33    ClosedChannel,
34}

36/// Mining modes supported by the MiningEngine.
37#[derive(Debug, Clone, Copy, PartialEq, Eq)]
38pub enum MiningMode {
39    /// Blocks are produced only in response to RPC calls.
40    None,
41    /// Automatic block productiona at fixed time intervals.
42    Interval { tick: Duration },
43    /// Automatic block production triggered by new transactions.
44    AutoMining,
45    /// Hybrid mode combining interval and transaction-based mining.
46    MixedMining { tick: Duration },
47}
48
49impl MiningMode {
50    /// Create a mining mode based on configuration parameters.
51    ///
52    /// This method determines the appropriate mining mode based on
53    /// the provided timing and behavioral preferences.
54    ///
55    /// # Arguments
56    /// * `block_time` - Optional duration between blocks for interval mining
57    /// * `mixed_mining` - Enable mixed mode when block_time is provided
58    /// * `no_mining` - Disable automatic mining when no block_time is set
59    ///
60    /// # Returns
61    /// The appropriate `MiningMode` variant based on input parameters
62    pub fn new(block_time: Option<Duration>, mixed_mining: bool, no_mining: bool) -> Self {
63        block_time.map_or_else(
64            || if no_mining { Self::None } else { Self::AutoMining },
65            |time| {
66                if mixed_mining {
67                    Self::MixedMining { tick: time }
68                } else {
69                    Self::Interval { tick: time }
70                }
71            },
72        )
73    }
74}

76/// Controller for blockchain block production operations.
77///
78/// The `MiningEngine` provides a high-level interface for managing block production
79/// It supports multiple mining modes, time manipulation for testing, and
80/// Ethereum-compatible RPC methods.
81///
82/// The engine coordinates between the transaction pool, consensus layer, and time
83/// management to provide flexible block production strategies suitable for both
84/// development and production environments.
85pub struct MiningEngine {
86    /// Coordination mechanism between the MiningEngine and the background
87    /// task that runs the mining loop in `ApiServer::run()`.
88    /// Calls to waker.wake() will unpark the task and force a
89    /// recheck of the current mining mode and rebuild of the polled streams.
90    waker: Arc<AtomicWaker>,
91    mining_mode: Arc<RwLock<MiningMode>>,
92    transaction_pool: Arc<TransactionPoolHandle>,
93    time_manager: Arc<TimeManager>,
94}
95
96impl MiningEngine {
97    /// Create a new mining engine controller.
98    ///
99    /// Initializes a mining engine with the specified components and configuration.
100    /// The engine will coordinate block production according to the initial mining mode.
101    ///
102    /// # Arguments
103    /// * `mining_mode` - Initial mining strategy
104    /// * `transaction_pool` - Handle for monitoring transaction pool changes
105    /// * `time_manager` - Component for blockchain time management
106    ///
107    /// # Returns
108    /// A new `MiningEngine` instance ready for use
109    pub fn new(
110        mining_mode: MiningMode,
111        transaction_pool: Arc<TransactionPoolHandle>,
112        time_manager: Arc<TimeManager>,
113    ) -> Self {
114        Self {
115            waker: Default::default(),
116            mining_mode: Arc::new(RwLock::new(mining_mode)),
117            transaction_pool,
118            time_manager,
119        }
120    }

122    /// Mine a specified number of blocks manually.
123    ///
124    /// This method implements the `anvil_mine` RPC call, allowing manual control
125    /// over block production. Blocks are mined sequentially, with optional time
126    /// advancement between each block.
127    ///
128    /// # Arguments
129    /// * `num_blocks` - Number of blocks to mine (defaults to 1 if None)
130    /// * `interval` - Optional time to advance between blocks (in seconds)
131    /// * `seal_sender` - Channel for sending block sealing commands.
132    ///
133    /// # Returns
134    /// * `Ok(H256)` - The hash of the last block mined successfully.
135    /// * `Err(MiningError)` - Block production failed
136    pub async fn mine(
137        &self,
138        num_blocks: Option<u64>,
139        interval: Option<Duration>,
140        seal_sender: &Sender<EngineCommand<sp_core::H256>>,
141    ) -> Result<H256, MiningError> {
142        let blocks = num_blocks.unwrap_or(1);
143        let mut last_hash = H256::zero();
144        for _ in 0..blocks {
145            if let Some(interval) = interval {
146                self.time_manager.increase_time(interval.as_secs());
147            }
148            last_hash = seal_now(seal_sender).await?.hash;
149        }
150        Ok(last_hash)
151    }

153    /// Ethereum-compatible block mining RPC method.
154    ///
155    /// Implements the `evm_mine` RPC call from the Ethereum JSON-RPC API.
156    /// This method provides compatibility with Ethereum development tools
157    /// and testing frameworks.
158    ///
159    /// # Arguments
160    /// * `opts` - Optional mining parameters including timestamp and block count
161    /// * `seal_sender` - Channel for sending block sealing commands.
162    ///
163    /// # Returns
164    /// * `Ok(H256)` - The hash of the last block mined successfully.
165    /// * `Err(MiningError)` - Mining operation failed
166    pub async fn evm_mine(
167        &self,
168        opts: Option<MineOptions>,
169        seal_sender: &Sender<EngineCommand<sp_core::H256>>,
170    ) -> Result<H256, MiningError> {
171        self.do_evm_mine(opts, seal_sender).await.map(|res| res.1)
172    }

174    /// Configure interval-based mining mode.
175    ///
176    /// Sets the mining engine to produce blocks at regular time intervals.
177    /// An interval of 0 disables interval mining. Changes take effect immediately
178    /// and will wake the background mining task.
179    ///
180    /// # Arguments
181    /// * `interval` - Block production interval in seconds (0 to disable)
182    pub fn set_interval_mining(&self, interval: Duration) {
183        let old_mode = *self.mining_mode.read();
184        let new_mode = if interval.as_secs() == 0 {
185            MiningMode::None
186        } else {
187            MiningMode::Interval { tick: interval }
188        };
189        trace!(target: "miner", "updated mining mode from {:?} to {:?}", old_mode, new_mode);
190        *self.mining_mode.write() = new_mode;
191        self.wake();
192    }

194    /// Get the current interval mining configuration.
195    ///
196    /// Returns the current block production interval if interval or mixed
197    /// mining is active, or None if interval mining is disabled.
198    ///
199    /// # Returns
200    /// * `Some(seconds)` - Interval mining active with specified interval
201    /// * `None` - Interval mining is disabled
202    pub fn get_interval_mining(&self) -> Option<u64> {
203        let mode = *self.mining_mode.read();
204        match mode {
205            MiningMode::Interval { tick } | MiningMode::MixedMining { tick } => {
206                Some(tick.as_secs())
207            }
208            _ => None,
209        }
210    }

212    /// Check if automatic mining is enabled.
213    ///
214    /// Returns true if the mining engine will automatically produce blocks
215    /// when new transactions are added to the transaction pool.
216    pub fn is_automine(&self) -> bool {
217        matches!(*self.mining_mode.read(), MiningMode::AutoMining)
218    }

220    /// Check if auto or mixed mining is active (single lock acquisition).
221    pub fn is_auto_or_mixed_mining(&self) -> bool {
222        matches!(*self.mining_mode.read(), MiningMode::AutoMining | MiningMode::MixedMining { .. })
223    }

225    /// Enable or disable automatic mining mode.
226    ///
227    /// When enabled, the mining engine will automatically produce a new block
228    /// whenever a transaction is added to the transaction pool. This provides
229    /// instant transaction confirmation in development environments.
230    ///
231    /// # Arguments
232    /// * `enabled` - True to enable auto-mining, false to disable
233    pub fn set_auto_mine(&self, enabled: bool) {
234        let old_mode = *self.mining_mode.read();
235        let new_mode = match (self.is_automine(), enabled) {
236            (true, false) => Some(MiningMode::None),
237            (false, true) => Some(MiningMode::AutoMining),
238            _ => None,
239        };
240        if let Some(new_mode) = new_mode {
241            trace!(target: "miner", "updated mining mode from {:?} to {:?}", old_mode, new_mode);
242            *self.mining_mode.write() = new_mode;
243            self.wake();
244        }
245    }

247    /// Set the timestamp for the next block to be mined.
248    ///
249    /// Allows precise control over block timestamps for testing scenarios.
250    /// The timestamp must not be older than the current blockchain time.
251    ///
252    /// # Arguments
253    /// * `time_in_seconds` - Unix timestamp in seconds for the next block
254    ///
255    /// # Returns
256    /// * `Ok(())` - Timestamp set successfully
257    /// * `Err(MiningError::Timestamp)` - Invalid timestamp
258    pub fn set_next_block_timestamp(&self, time: Duration) -> Result<(), MiningError> {
259        self.time_manager
260            // this will convert the time_in_seconds in milliseconds. It is transparent
261            // to the user
262            .set_next_block_timestamp(time.as_secs())
263            .map_err(|_| MiningError::Timestamp)
264    }
265
266    /// Advance the blockchain time by a specified duration.
267    ///
268    /// Increases the current blockchain time, affecting timestamps of future blocks.
269    ///
270    /// # Arguments
271    /// * `time_in_seconds` - Duration to advance in seconds
272    ///
273    /// # Returns
274    /// * `new_timestamp` - The new offset to the current time as i64
275    pub fn increase_time(&self, time: Duration) -> i64 {
276        self.time_manager.increase_time(time.as_secs()).saturating_div(1000) as i64
277    }
278
279    /// Set the blockchain time to a specific timestamp.
280    ///
281    /// Resets the blockchain time to the specified timestamp and returns
282    /// the time difference from the previous timestamp.
283    ///
284    /// # Arguments
285    /// * `timestamp` - Target timestamp in seconds since Unix epoch
286    ///
287    /// # Returns
288    /// * `offset_seconds` - Time difference from previous timestamp
289    pub fn set_time(&self, timestamp: Duration) -> u64 {
290        let now = self.time_manager.current_call_timestamp();
291        self.time_manager.reset(timestamp.as_secs());
292        let offset = (timestamp.as_millis() as u64).saturating_sub(now);
293        Duration::from_millis(offset).as_secs()
294    }
295
296    /// Configure automatic timestamp intervals between blocks.
297    ///
298    /// Sets a fixed time interval that will be automatically added to each
299    /// block's timestamp relative to the previous block. This ensures
300    /// consistent time progression in the blockchain.
301    ///
302    /// # Arguments
303    /// * `interval_in_seconds` - Time interval to add between block timestamps
304    pub fn set_block_timestamp_interval(&self, interval_in_seconds: Duration) {
305        self.time_manager.set_block_timestamp_interval(interval_in_seconds.as_secs())
306    }
307
308    /// Remove automatic timestamp intervals between blocks.
309    ///
310    /// Disables the automatic timestamp interval feature, allowing blocks
311    /// to have timestamps based on the actual mining time rather than
312    /// fixed intervals.
313    ///
314    /// # Returns
315    /// * `true` - Timestamp interval was removed
316    /// * `false` - No timestamp interval was configured
317    pub fn remove_block_timestamp_interval(&self) -> bool {
318        self.time_manager.remove_block_timestamp_interval()
319    }

327    pub async fn do_evm_mine(
328        &self,
329        opts: Option<MineOptions>,
330        seal_sender: &Sender<EngineCommand<sp_core::H256>>,
331    ) -> Result<(u64, H256), MiningError> {
332        let mut blocks_to_mine = 1u64;
333        let mut last_hash = H256::zero();
334
335        if let Some(opts) = opts {
336            let timestamp = match opts {
337                MineOptions::Timestamp(timestamp) => timestamp,
338                MineOptions::Options { timestamp, blocks } => {
339                    if let Some(blocks) = blocks {
340                        blocks_to_mine = blocks;
341                    }
342                    timestamp
343                }
344            };
345            if let Some(timestamp) = timestamp {
346                // timestamp was explicitly provided to be the next timestamp
347                self.time_manager
348                    .set_next_block_timestamp(timestamp)
349                    .map_err(|_| MiningError::Timestamp)?;
350            }
351        }
352
353        for _ in 0..blocks_to_mine {
354            last_hash = seal_now(seal_sender).await?.hash;
355        }
356
357        Ok((blocks_to_mine, last_hash))
358    }
359}

361pub async fn seal_now(
362    seal_command_sender: &Sender<EngineCommand<sp_core::H256>>,
363) -> Result<CreatedBlock<Hash>, MiningError> {
364    let (sender, receiver) = oneshot::channel();
365    let seal_command = EngineCommand::SealNewBlock {
366        create_empty: true,
367        finalize: true,
368        parent_hash: None,
369        sender: Some(sender),
370    };
371    seal_command_sender.send(seal_command).await.map_err(|_| MiningError::ClosedChannel)?;
372    match receiver.await {
373        Ok(Ok(rx)) => Ok(rx),
374        Ok(Err(e)) => Err(MiningError::BlockProducing(e)),
375        Err(_e) => Err(MiningError::ClosedChannel),
376    }
377}

381/// Identifies the source of a mining trigger in the select loop.
382#[derive(Debug, Clone, Copy, PartialEq, Eq)]
383pub enum MiningTrigger {
384    /// An interval tick fired (produce a block regardless of pool state).
385    Interval,
386    /// A new transaction was imported into the pool.
387    PoolNotification,
388}
389
390pub type SealCommandStream = Pin<Box<dyn FusedStream<Item = MiningTrigger> + Send>>;
391
392fn build_interval_stream(interval: Duration) -> SealCommandStream {
393    let mut interval_ticker = interval_at(Instant::now() + interval, interval);
394    interval_ticker.set_missed_tick_behavior(MissedTickBehavior::Delay);
395
396    let stream = unfold(interval_ticker, |mut interval_tick| async {
397        interval_tick.tick().await;
398        Some((MiningTrigger::Interval, interval_tick))
399    });
400    Box::pin(stream.fuse())
401}
402
403fn build_auto_stream(engine: &Arc<MiningEngine>) -> SealCommandStream {
404    let stream = engine
405        .transaction_pool
406        .import_notification_stream()
407        .map(|_| MiningTrigger::PoolNotification);
408    Box::pin(stream.fuse())
409}
410
411pub fn build_streams_for_mode(
412    mode: MiningMode,
413    engine: &Arc<MiningEngine>,
414) -> SelectAll<SealCommandStream> {
415    let mut streams: Vec<SealCommandStream> = Vec::new();
416    if let Some(stream) = match mode {
417        MiningMode::Interval { tick } | MiningMode::MixedMining { tick } => Some(tick),
418        _ => None,
419    }
420    .map(build_interval_stream)
421    {
422        streams.push(stream)
423    }
424    if let Some(stream) = matches!(mode, MiningMode::AutoMining | MiningMode::MixedMining { .. })
425        .then(|| build_auto_stream(engine))
426    {
427        streams.push(stream)
428    }
429    select_all(streams)
430}

432pub async fn wait_for_mode_change(
433    engine: &Arc<MiningEngine>,
434    current: Option<MiningMode>,
435) -> MiningMode {
436    futures::future::poll_fn(|cx| {
437        let mode = *engine.mining_mode.read();
438        if current.as_ref().is_none_or(|m| *m != mode) {
439            return std::task::Poll::Ready(mode);
440        }
441        engine.waker.register(cx.waker());
442        std::task::Poll::Pending
443    })
444    .await
