24/// Transaction info for txpool RPCs with Option fields to match Anvil's null values
25#[derive(Debug, Clone, Serialize, Deserialize)]
26pub struct TxpoolTransactionInfo {
27    pub hash: H256,
28    pub block_hash: Option<H256>,
29    pub block_number: Option<sp_core::U256>,
30    pub transaction_index: Option<sp_core::U256>,
31    pub from: sp_core::H160,
32    pub transaction_signed: TransactionSigned,
33}

35/// Decode extrinsic into ETH transaction payload and signed transaction
36pub(super) fn decode_eth_transaction(
37    tx_data: &Arc<polkadot_sdk::sp_runtime::OpaqueExtrinsic>,
38) -> Option<(Vec<u8>, TransactionSigned)> {
39    let encoded = tx_data.encode();
40    let ext =
41        UncheckedExtrinsic::decode_all_with_depth_limit(MAX_EXTRINSIC_DEPTH, &mut &encoded[..])
42            .ok()?;
43
44    let polkadot_sdk::sp_runtime::generic::UncheckedExtrinsic {
45        function: RuntimeCall::Revive(polkadot_sdk::pallet_revive::Call::eth_transact { payload }),
46        ..
47    } = ext.0
48    else {
49        return None;
50    };
51
52    let signed_tx = TransactionSigned::decode(&payload).ok()?;
53
54    Some((payload, signed_tx))
55}

57/// Check if transaction matches ETH hash
58pub(super) fn transaction_matches_eth_hash(
59    tx_data: &Arc<polkadot_sdk::sp_runtime::OpaqueExtrinsic>,
60    target_eth_hash: B256,
61) -> bool {
62    let Some((payload, _signed_tx)) = decode_eth_transaction(tx_data) else {
63        return false;
64    };
65
66    let tx_eth_hash = keccak256(&payload);
67    B256::from_slice(tx_eth_hash.as_ref()) == target_eth_hash
68}

79/// Extract fields from ETH transaction
80fn extract_tx_fields(signed_tx: &TransactionSigned) -> TransactionFields {
81    match signed_tx {
82        TransactionSigned::TransactionLegacySigned(tx) => {
83            let t = &tx.transaction_legacy_unsigned;
84            TransactionFields {
85                nonce: t.nonce,
86                to: t.to,
87                value: t.value,
88                gas: t.gas,
89                gas_price: t.gas_price,
90            }
91        }
92        TransactionSigned::Transaction2930Signed(tx) => {
93            let t = &tx.transaction_2930_unsigned;
94            TransactionFields {
95                nonce: t.nonce,
96                to: t.to,
97                value: t.value,
98                gas: t.gas,
99                gas_price: t.gas_price,
100            }
101        }
102        TransactionSigned::Transaction1559Signed(tx) => {
103            let t = &tx.transaction_1559_unsigned;
104            TransactionFields {
105                nonce: t.nonce,
106                to: t.to,
107                value: t.value,
108                gas: t.gas,
109                gas_price: t.max_fee_per_gas,
110            }
111        }
112        TransactionSigned::Transaction4844Signed(tx) => {
113            let t = &tx.transaction_4844_unsigned;
114            TransactionFields {
115                nonce: t.nonce,
116                to: Some(t.to),
117                value: t.value,
118                gas: t.gas,
119                gas_price: t.max_fee_per_gas,
120            }
121        }
122        TransactionSigned::Transaction7702Signed(tx) => {
123            let t = &tx.transaction_7702_unsigned;
124            TransactionFields {
125                nonce: t.nonce,
126                to: Some(t.to),
127                value: t.value,
128                gas: t.gas,
129                gas_price: t.max_fee_per_gas,
130            }
131        }
132    }
133}

135/// Extract transaction summary from extrinsic
136pub(super) fn extract_tx_summary(
137    tx_data: &Arc<polkadot_sdk::sp_runtime::OpaqueExtrinsic>,
138) -> Option<(Address, u64, TxpoolInspectSummary)> {
139    let (_payload, signed_tx) = decode_eth_transaction(tx_data)?;
140
141    let from = recover_maybe_impersonated_address(&signed_tx).ok()?;
142    let sender = Address::from_slice(from.as_bytes());
143
144    let fields = extract_tx_fields(&signed_tx);
145
146    let to_addr = fields.to.map(|addr| Address::from_slice(addr.as_bytes()));
147    let value_u256 = U256::from_limbs(fields.value.0);
148    let gas_u64 = fields.gas.as_u64();
149    let gas_price_u128 = fields.gas_price.as_u128();
150    let nonce_u64 = fields.nonce.as_u64();
151
152    Some((
153        sender,
154        nonce_u64,
155        TxpoolInspectSummary {
156            to: to_addr,
157            value: value_u256,
158            gas: gas_u64,
159            gas_price: gas_price_u128,
160        },
161    ))
162}

164/// Extract full transaction info from extrinsic
165pub(super) fn extract_tx_info(
166    tx_data: &Arc<polkadot_sdk::sp_runtime::OpaqueExtrinsic>,
167) -> Option<(Address, u64, TxpoolTransactionInfo)> {
168    let (payload, signed_tx) = decode_eth_transaction(tx_data)?;
169
170    let eth_hash = keccak256(&payload);
171    let eth_hash_h256 = H256::from_slice(eth_hash.as_ref());
172
173    let from = recover_maybe_impersonated_address(&signed_tx).ok()?;
174    let sender = Address::from_slice(from.as_bytes());
175
176    let fields = extract_tx_fields(&signed_tx);
177    let nonce_u64 = fields.nonce.as_u64();
178
179    let tx_info = TxpoolTransactionInfo {
180        hash: eth_hash_h256,
181        block_hash: None,
182        block_number: None,
183        transaction_index: None,
184        from,
185        transaction_signed: signed_tx,
186    };
187
188    Some((sender, nonce_u64, tx_info))
189}

191/// Extract sender address from extrinsic as Alloy Address type.
192/// Helper for `anvil_remove_pool_transactions` to compare sender addresses.
193pub(super) fn extract_sender(
194    tx_data: &Arc<polkadot_sdk::sp_runtime::OpaqueExtrinsic>,
195) -> Option<Address> {
196    let (_payload, signed_tx) = decode_eth_transaction(tx_data)?;
197
198    let from = recover_maybe_impersonated_address(&signed_tx).ok()?;
199    let sender = Address::from_slice(from.as_bytes());
200
201    Some(sender)
202}

204/// Get the next nonce for an address based on pending pool transactions.
205/// Returns `Some(max_nonce + 1)` if the address has pending txs, `None` otherwise.
206pub(super) fn get_pool_nonce(pool: &TransactionPoolHandle, address: Address) -> Option<u64> {
207    pool.ready()
208        .filter_map(|tx| extract_tx_summary(tx.data()))
209        .filter(|(sender, _, _)| *sender == address)
210        .map(|(_, nonce, _)| nonce)
211        .max()
212        .map(|nonce| nonce.saturating_add(1))
213}
