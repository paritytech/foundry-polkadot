17/// A Wrapper around [TracingInspector] to allow adding zkEVM traces.
18#[derive(Clone, Debug, Default)]
19pub struct TraceCollector {
20    inner: TracingInspector,
21}
22
23impl TraceCollector {
24    /// Returns a new instance for the given config
25    pub fn new(config: TracingInspectorConfig) -> Self {
26        Self { inner: TracingInspector::new(config) }
27    }
28
29    /// Returns the inner [`TracingInspector`]
30    #[inline]
31    pub fn inner(&mut self) -> &mut TracingInspector {
32        &mut self.inner
33    }
34
35    /// Resets the inspector to its initial state of [Self::new].
36    /// This makes the inspector ready to be used again.
37    ///
38    /// Note that this method has no effect on the allocated capacity of the vector.
39    #[inline]
40    pub fn fuse(&mut self) {
41        self.inner.fuse()
42    }
43
44    /// Resets the inspector to it's initial state of [Self::new].
45    #[inline]
46    pub fn fused(self) -> Self {
47        Self { inner: self.inner.fused() }
48    }
49
50    /// Returns the config of the inspector.
51    pub const fn config(&self) -> &TracingInspectorConfig {
52        self.inner.config()
53    }
54
55    /// Returns a mutable reference to the config of the inspector.
56    pub fn config_mut(&mut self) -> &mut TracingInspectorConfig {
57        self.inner.config_mut()
58    }
59
60    /// Updates the config of the inspector.
61    pub fn update_config(
62        &mut self,
63        f: impl FnOnce(TracingInspectorConfig) -> TracingInspectorConfig,
64    ) {
65        self.inner.update_config(f);
66    }
67
68    /// Gets a reference to the recorded call traces.
69    pub const fn traces(&self) -> &CallTraceArena {
70        self.inner.traces()
71    }
72
73    /// Gets a mutable reference to the recorded call traces.
74    pub fn traces_mut(&mut self) -> &mut CallTraceArena {
75        self.inner.traces_mut()
76    }
77
78    /// Consumes the inspector and returns the recorded call traces.
79    pub fn into_traces(self) -> CallTraceArena {
80        self.inner.into_traces()
81    }
82
83    /// Manually the gas used of the root trace.
84    ///
85    /// This is useful if the root trace's gasUsed should mirror the actual gas used by the
86    /// transaction.
87    ///
88    /// This allows setting it manually by consuming the execution result's gas for example.
89    #[inline]
90    pub fn set_transaction_gas_used(&mut self, gas_used: u64) {

99    }

101    /// Consumes the Inspector and returns a [ParityTraceBuilder].

111    }

160impl InspectorExt for TraceCollector {
161    fn trace_revive(
162        &mut self,
163        context: Ecx<'_, '_, '_>,
164        call_traces: Box<dyn std::any::Any>,
165        record_top_call: bool,
166    ) {
167        let call_traces = *call_traces
168            .downcast::<CallTrace>()
169            .expect("TraceCollector::trace_revive expected call traces to be a CallTrace");
170        use revm::Inspector;
171        fn trace_call_recursive(
172            tracer: &mut TracingInspector,
173            context: Ecx<'_, '_, '_>,
174            call: CallTrace,
175            suppressed_top_call: bool,
176        ) -> u64 {
177            let inputs = &mut CallInputs {
178                input: revm::interpreter::CallInput::Bytes(call.input.0.clone().into()),
179                gas_limit: call.gas,
180                scheme: revm::interpreter::CallScheme::Call,
181                caller: call.from.0.into(),
182                value: revm::interpreter::CallValue::Transfer(RU256::from_be_bytes(
183                    call.value.unwrap_or_default().to_big_endian(),
184                )),
185                target_address: call.to.0.into(),
186                bytecode_address: call.to.0.into(),
187                is_static: false,
188                return_memory_offset: Default::default(),
189                known_bytecode: None,
190            };
191            let is_first_non_system_call = !suppressed_top_call;
192
193            // We ignore traces from system addresses, the default account abstraction calls on
194            // caller address, and the original call (identified when neither `to` or
195            // `from` are system addresses) since it is already included in EVM trace.
196            let record_trace =
197                !is_first_non_system_call && inputs.target_address != context.tx.caller;
198
199            let mut outcome = if let Some(reason) = &call.revert_reason {
200                CallOutcome {
201                    result: InterpreterResult {
202                        result: InstructionResult::Revert,
203                        output: reason.as_bytes().to_owned().into(),
204                        gas: Gas::new_spent(call.gas_used.as_u64()),
205                    },
206                    memory_offset: Default::default(),
207                    was_precompile_called: false,
208                    precompile_call_logs: vec![],
209                }
210            } else {
211                CallOutcome {
212                    result: InterpreterResult {
213                        result: InstructionResult::Return,
214                        output: call.output.clone().0.into(),
215                        gas: Gas::new_spent(call.gas_used.as_u64()),
216                    },
217                    memory_offset: Default::default(),
218                    was_precompile_called: false,
219                    precompile_call_logs: vec![],
220                }
221            };
222
223            let mut create_inputs =
224                if matches!(call.call_type, CallType::Create | CallType::Create2) {
225                    let scheme = match call.call_type {
226                        CallType::Create => CreateScheme::Create,
227                        CallType::Create2 => {
228                            // For Create2 traces from Polkadot Revive, the CallTrace doesn't
229                            // provide a separate salt field. We derive a salt by hashing the input
230                            // to ensure it fits in U256 and is deterministic.
231                            let salt =
232                                RU256::from_be_bytes(
233                                    B256::from_slice(
234                                        &alloy_primitives::keccak256::<&[u8]>(
235                                            call.input.0.as_ref(),
236                                        )[..],
237                                    )
238                                    .0,
239                                );
240                            CreateScheme::Create2 { salt }
241                        }
242                        _ => panic!("impossible"),
243                    };
244                    Some(CreateInputs {
245                        caller: inputs.caller,
246                        scheme,
247                        value: inputs.value.get(),
248                        init_code: inputs.input.bytes(context),
249                        gas_limit: inputs.gas_limit,
250                    })
251                } else {
252                    None
253                };
254
255            // start span
256            if record_trace {
257                if let Some(inputs) = &mut create_inputs {
258                    tracer.create(context, inputs);
259                } else {
260                    tracer.call(context, inputs);
261                }
262            }
263            for log in call.logs {
264                tracer.log(
265                    context,
266                    Log::new_unchecked(
267                        log.address.0.into(),
268                        log.topics.iter().map(|x| B256::from_slice(x.as_bytes())).collect(),
269                        log.data.0.into(),
270                    ),
271                );
272            }
273
274            // We increment the depth for inner calls as normally traces are processed
275            // during execution, where the environment takes care of updating the context
276            let (new_depth, overflow) = context.journaled_state.depth.overflowing_add(1);
277            if !overflow && record_trace {
278                context.journaled_state.depth = new_depth;
279            }
280
281            // recurse into inner calls
282            // record extra gas from ignored traces, to add it at end
283            let mut extra_gas = if record_trace { 0u64 } else { call.gas_used.as_u64() };
284            for inner_call in call.calls {
285                let inner_extra_gas = trace_call_recursive(
286                    tracer,
287                    context,
288                    inner_call,
289                    suppressed_top_call || is_first_non_system_call,
290                );
291                extra_gas = extra_gas.saturating_add(inner_extra_gas);
292            }
293
294            // We then decrement the call depth so `call_end`/`create_end` has the correct context
295            if !overflow && record_trace {
296                context.journaled_state.depth = context.journaled_state.depth.saturating_sub(1);
297            }
298
299            // finish span
300            if record_trace {
301                if let Some(inputs) = &mut create_inputs {
302                    let mut outcome = if let Some(reason) = call.revert_reason {
303                        CreateOutcome {
304                            result: InterpreterResult {
305                                result: InstructionResult::Revert,
306                                output: reason.as_bytes().to_owned().into(),
307                                gas: Gas::new_spent(call.gas_used.as_u64() + extra_gas),
308                            },
309                            address: None,
310                        }
311                    } else {
312                        CreateOutcome {
313                            result: InterpreterResult {
314                                result: InstructionResult::Return,
315                                output: Bytes::from(call.output.clone().0),
316                                gas: Gas::new_spent(call.gas_used.as_u64() + extra_gas),
317                            },
318                            address: Some(call.to.0.into()),
319                        }
320                    };
321
322                    tracer.create_end(context, inputs, &mut outcome);
323                } else {
324                    if extra_gas != 0 {
325                        outcome.result.gas = Gas::new_spent(outcome.result.gas.spent() + extra_gas);
326                    }
327                    tracer.call_end(context, inputs, &mut outcome);
328                }
329            }
330
331            extra_gas
332        }
333
334        let (new_depth, overflow) = context.journaled_state.depth.overflowing_add(1);
335        // If we are going to record the top call then we don't want to change the call depth
336        if !overflow && !record_top_call {
337            context.journaled_state.depth = new_depth;
338        }
339
340        trace_call_recursive(&mut self.inner, context, call_traces, record_top_call);
341
342        if !overflow && !record_top_call {
343            context.journaled_state.depth = context.journaled_state.depth.saturating_sub(1);
344        }
345    }
346}
