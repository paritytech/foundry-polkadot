36/// Builder type to configure how to compile a project.
37///
38/// This is merely a wrapper for [`Project::compile()`] which also prints to stdout depending on its
39/// settings.

156    /// Compiles the project.
157    #[instrument(target = "forge::compile", skip_all)]
158    pub fn compile<C: Compiler<CompilerContract = Contract>>(
159        mut self,
160        project: &Project<C>,
161    ) -> Result<ProjectCompileOutput<C>>
162    where
163        DynamicTestLinkingPreprocessor: Preprocessor<C>,
164    {
165        self.project_root = project.root().to_path_buf();
166
167        // TODO: Avoid using std::process::exit(0).
168        // Replacing this with a return (e.g., Ok(ProjectCompileOutput::default())) would be more
169        // idiomatic, but it currently requires a `Default` bound on `C::Language`, which
170        // breaks compatibility with downstream crates like `foundry-cli`. This would need a
171        // broader refactor across the call chain. Leaving it as-is for now until a larger
172        // refactor is feasible.
173        if !project.paths.has_input_files() && self.files.is_empty() {
174            sh_println!("Nothing to compile")?;
175            std::process::exit(0);
176        }
177
178        // Taking is fine since we don't need these in `compile_with`.
179        let files = std::mem::take(&mut self.files);
180        let preprocess = self.dynamic_test_linking;
181        self.compile_with(|| {
182            let sources = if !files.is_empty() {
183                Source::read_all(files)?
184            } else {
185                project.paths.read_input_files()?
186            };
187
188            let mut compiler =
189                foundry_compilers::project::ProjectCompiler::with_sources(project, sources)?;
190            if preprocess {
191                compiler = compiler.with_preprocessor(DynamicTestLinkingPreprocessor);
192            }
193            compiler.compile().map_err(Into::into)
194        })
195    }
196
197    /// Compiles the project with the given closure
198    fn compile_with<C: Compiler<CompilerContract = Contract>, F>(
199        self,
200        f: F,
201    ) -> Result<ProjectCompileOutput<C>>
202    where
203        F: FnOnce() -> Result<ProjectCompileOutput<C>>,
204    {
205        let quiet = self.quiet.unwrap_or(false);
206        let bail = self.bail.unwrap_or(true);
207
208        let output = with_compilation_reporter(quiet, Some(self.project_root.clone()), || {
209            tracing::debug!("compiling project");
210
211            let timer = Instant::now();
212            let r = f();
213            let elapsed = timer.elapsed();
214
215            tracing::debug!("finished compiling in {:.3}s", elapsed.as_secs_f64());
216            r
217        })?;
218
219        if bail && output.has_compiler_errors() {
220            eyre::bail!("{output}")
221        }
222
223        if !quiet {
224            if !shell::is_json() {
225                if output.is_unchanged() {
226                    sh_println!("No files changed, compilation skipped")?;
227                } else {
228                    // print the compiler output / warnings
229                    sh_println!("{output}")?;
230                }
231            }
232
233            self.handle_output(&output)?;
234        }
235
236        Ok(output)
237    }

239    /// If configured, this will print sizes or names
240    fn handle_output<C: Compiler<CompilerContract = Contract>>(
241        &self,
242        output: &ProjectCompileOutput<C>,
243    ) -> Result<()> {
244        let print_names = self.print_names.unwrap_or(false);
245        let print_sizes = self.print_sizes.unwrap_or(false);
246
247        // print any sizes or names
248        if print_names {
249            let mut artifacts: BTreeMap<_, Vec<_>> = BTreeMap::new();
250            for (name, (_, version)) in output.versioned_artifacts() {
251                artifacts.entry(version).or_default().push(name);
252            }
253
254            if shell::is_json() {
255                sh_println!("{}", serde_json::to_string(&artifacts).unwrap())?;
256            } else {
257                for (version, names) in artifacts {
258                    sh_println!(
259                        "  compiler version: {}.{}.{}",
260                        version.major,
261                        version.minor,
262                        version.patch
263                    )?;
264                    for name in names {
265                        sh_println!("    - {name}")?;
266                    }
267                }
268            }
269        }
270
271        if print_sizes {
272            // add extra newline if names were already printed
273            if print_names && !shell::is_json() {
274                sh_println!()?;
275            }
276
277            let mut size_report = SizeReport::new(
278                BTreeMap::new(),
279                self.runtime_size_limit.unwrap_or(CONTRACT_RUNTIME_SIZE_LIMIT),
280                self.initcode_size_limit.unwrap_or(CONTRACT_INITCODE_SIZE_LIMIT),
281            );
282
283            let mut artifacts: BTreeMap<String, Vec<_>> = BTreeMap::new();
284            for (id, artifact) in output.artifact_ids().filter(|(id, _)| {
285                // filter out forge-std specific contracts
286                !id.source.to_string_lossy().contains("/forge-std/src/")
287            }) {
288                artifacts.entry(id.name.clone()).or_default().push((id.source.clone(), artifact));
289            }
290
291            for (name, artifact_list) in artifacts {
292                for (path, artifact) in &artifact_list {
293                    let runtime_size = contract_size(*artifact, false).unwrap_or_default();
294                    let init_size = contract_size(*artifact, true).unwrap_or_default();
295
296                    let is_dev_contract = artifact
297                        .abi
298                        .as_ref()
299                        .map(|abi| {
300                            abi.functions().any(|f| {
301                                f.test_function_kind().is_known()
302                                    || matches!(f.name.as_str(), "IS_TEST" | "IS_SCRIPT")
303                            })
304                        })
305                        .unwrap_or(false);
306
307                    let unique_name = if artifact_list.len() > 1 {
308                        format!(
309                            "{} ({})",
310                            name,
311                            path.strip_prefix(&self.project_root).unwrap_or(path).display()
312                        )
313                    } else {
314                        name.clone()
315                    };
316
317                    size_report.contracts.insert(
318                        unique_name,
319                        ContractInfo { runtime_size, init_size, is_dev_contract },
320                    );
321                }
322            }
323
324            sh_println!("{size_report}")?;
325
326            eyre::ensure!(
327                !size_report.exceeds_runtime_size_limit(),
328                "some contracts exceed the runtime size limit \
329                 (EIP-170: {CONTRACT_RUNTIME_SIZE_LIMIT} bytes)"
330            );
331            // Check size limits only if not ignoring EIP-3860
332            eyre::ensure!(
333                self.ignore_eip_3860 || !size_report.exceeds_initcode_size_limit(),
334                "some contracts exceed the initcode size limit \
335                 (EIP-3860: {CONTRACT_INITCODE_SIZE_LIMIT} bytes)"
336            );
337        }
338
339        Ok(())
340    }
341}

343// https://eips.ethereum.org/EIPS/eip-170
344const CONTRACT_RUNTIME_SIZE_LIMIT: usize = 24576;
345
346// https://eips.ethereum.org/EIPS/eip-3860
347const CONTRACT_INITCODE_SIZE_LIMIT: usize = 49152;
348
349const CONTRACT_SIZE_LIMIT_MARGIN: f64 = 0.73;
350/// Contracts with info about their size
351pub struct SizeReport {
352    /// `contract name -> info`
353    pub contracts: BTreeMap<String, ContractInfo>,
354    /// Runtime size limit
355    runtime_size_limit: usize,
356    /// Initcode size limit
357    initcode_size_limit: usize,
358}
359
360impl SizeReport {
361    pub fn new(
362        contracts: BTreeMap<String, ContractInfo>,
363        runtime_size_limit: usize,
364        initcode_size_limit: usize,
365    ) -> Self {
366        Self { contracts, runtime_size_limit, initcode_size_limit }
367    }
368    /// Returns the maximum runtime code size, excluding dev contracts.
369    pub fn max_runtime_size(&self) -> usize {
370        self.contracts
371            .values()
372            .filter(|c| !c.is_dev_contract)
373            .map(|c| c.runtime_size)
374            .max()
375            .unwrap_or(0)
376    }
377
378    /// Returns the maximum initcode size, excluding dev contracts.
379    pub fn max_init_size(&self) -> usize {
380        self.contracts
381            .values()
382            .filter(|c| !c.is_dev_contract)
383            .map(|c| c.init_size)
384            .max()
385            .unwrap_or(0)
386    }
387
388    /// Returns true if any contract exceeds the runtime size limit, excluding dev contracts.
389    pub fn exceeds_runtime_size_limit(&self) -> bool {
390        self.max_runtime_size() > self.runtime_size_limit
391    }
392
393    /// Returns true if any contract exceeds the initcode size limit, excluding dev contracts.
394    pub fn exceeds_initcode_size_limit(&self) -> bool {
395        self.max_init_size() > self.initcode_size_limit
396    }
397}

410impl SizeReport {
411    fn format_json_output(&self) -> String {
412        let contracts = self
413            .contracts
414            .iter()
415            .filter(|(_, c)| !c.is_dev_contract && (c.runtime_size > 0 || c.init_size > 0))
416            .map(|(name, contract)| {
417                (
418                    name.clone(),
419                    serde_json::json!({
420                        "runtime_size": contract.runtime_size,
421                        "init_size": contract.init_size,
422                        "runtime_margin": self.runtime_size_limit as isize - contract.runtime_size as isize,
423                        "init_margin": self.initcode_size_limit as isize - contract.init_size as isize,
424                    }),
425                )
426            })
427            .collect::<serde_json::Map<_, _>>();
428
429        serde_json::to_string(&contracts).unwrap()
430    }
431
432    fn format_table_output(&self) -> Table {
433        let mut table = Table::new();
434        if shell::is_markdown() {
435            table.load_preset(ASCII_MARKDOWN);
436        } else {
437            table.apply_modifier(UTF8_ROUND_CORNERS);
438        }
439
440        table.set_header(vec![
441            Cell::new("Contract"),
442            Cell::new("Runtime Size (B)"),
443            Cell::new("Initcode Size (B)"),
444            Cell::new("Runtime Margin (B)"),
445            Cell::new("Initcode Margin (B)"),
446        ]);
447
448        // Filters out dev contracts (Test or Script)
449        let contracts = self
450            .contracts
451            .iter()
452            .filter(|(_, c)| !c.is_dev_contract && (c.runtime_size > 0 || c.init_size > 0));
453        for (name, contract) in contracts {
454            let runtime_margin = self.runtime_size_limit as isize - contract.runtime_size as isize;
455            let init_margin = self.initcode_size_limit as isize - contract.init_size as isize;
456
457            let runtime_size_warning =
458                (CONTRACT_SIZE_LIMIT_MARGIN * self.runtime_size_limit as f64) as usize;
459            let initcode_size_warning =
460                (CONTRACT_SIZE_LIMIT_MARGIN * self.initcode_size_limit as f64) as usize;
461
462            let runtime_color = if contract.runtime_size < runtime_size_warning {
463                Color::Reset
464            } else if contract.runtime_size <= self.runtime_size_limit {
465                Color::Yellow
466            } else {
467                Color::Red
468            };
469
470            let init_color = if contract.init_size < initcode_size_warning {
471                Color::Reset
472            } else if contract.init_size <= self.initcode_size_limit {
473                Color::Yellow
474            } else {
475                Color::Red
476            };
477
478            let locale = &Locale::en;
479            table.add_row([
480                Cell::new(name),
481                Cell::new(contract.runtime_size.to_formatted_string(locale)).fg(runtime_color),
482                Cell::new(contract.init_size.to_formatted_string(locale)).fg(init_color),
483                Cell::new(runtime_margin.to_formatted_string(locale)).fg(runtime_color),
484                Cell::new(init_margin.to_formatted_string(locale)).fg(init_color),
485            ]);
486        }
487
488        table
489    }
490}

492/// Returns the deployed or init size of the contract.
493fn contract_size<T: Artifact>(artifact: &T, initcode: bool) -> Option<usize> {
494    let bytecode = if initcode {
495        artifact.get_bytecode_object()?
496    } else {
497        artifact.get_deployed_bytecode_object()?
498    };
499
500    let size = match bytecode.as_ref() {
501        BytecodeObject::Bytecode(bytes) => bytes.len(),
502        BytecodeObject::Unlinked(unlinked) => {
503            // we don't need to account for placeholders here, because library placeholders take up
504            // 40 characters: `__$<library hash>$__` which is the same as a 20byte address in hex.
505            let mut size = unlinked.len();
506            if unlinked.starts_with("0x") {
507                size -= 2;
508            }
509            // hex -> bytes
510            size / 2
511        }
512    };
513
514    Some(size)
515}

544/// Creates a [Project] from an Etherscan source.
545pub fn etherscan_project(metadata: &Metadata, target_path: &Path) -> Result<Project> {
546    let target_path = dunce::canonicalize(target_path)?;
547    let sources_path = target_path.join(&metadata.contract_name);
548    metadata.source_tree().write_to(&target_path)?;
549
550    let mut settings = metadata.settings()?;
551
552    // make remappings absolute with our root
553    for remapping in &mut settings.remappings {
554        let new_path = sources_path.join(remapping.path.trim_start_matches('/'));
555        remapping.path = new_path.display().to_string();
556    }
557
558    // add missing remappings
559    if !settings.remappings.iter().any(|remapping| remapping.name.starts_with("@openzeppelin/")) {
560        let oz = Remapping {
561            context: None,
562            name: "@openzeppelin/".into(),
563            path: sources_path.join("@openzeppelin").display().to_string(),
564        };
565        settings.remappings.push(oz);
566    }
567
568    // root/
569    //   ContractName/
570    //     [source code]
571    let paths = ProjectPathsConfig::builder()
572        .sources(sources_path.clone())
573        .remappings(settings.remappings.clone())
574        .build_with_root(sources_path);
575
576    // TODO: detect vyper
577    let v = metadata.compiler_version()?;
578    let solc = Solc::find_or_install(&v)?;
579
580    let compiler = MultiCompiler {
581        solidity: foundry_compilers::multi::SolidityCompiler::Solc(SolcCompiler::Specific(solc)),
582        vyper: None,
583    };
584
585    Ok(ProjectBuilder::<MultiCompiler>::default()
586        .settings(MultiCompilerSettings {
587            solc: SolcSettings {
588                settings: SolcConfig::builder().settings(settings).build(),
589                ..Default::default()
590            },
591            ..Default::default()
592        })
593        .paths(paths)
594        .ephemeral()
595        .no_artifacts()
596        .build(compiler)?)
597}

599/// Configures the reporter and runs the given closure.
600pub fn with_compilation_reporter<O>(
601    quiet: bool,
602    project_root: Option<PathBuf>,
603    f: impl FnOnce() -> O,
604) -> O {
605    #[expect(clippy::collapsible_else_if)]
606    let reporter = if quiet || shell::is_json() {
607        Report::new(NoReporter::default())
608    } else {
609        if std::io::stdout().is_terminal() {
610            Report::new(SpinnerReporter::spawn(project_root))
611        } else {
612            Report::new(BasicStdoutReporter::default())
613        }
614    };
615
616    foundry_compilers::report::with_scoped(&reporter, f)
617}

619/// Container type for parsing contract identifiers from CLI.
620///
621/// Passed string can be of the following forms:
622/// - `src/Counter.sol` - path to the contract file, in the case where it only contains one contract
623/// - `src/Counter.sol:Counter` - path to the contract file and the contract name
624/// - `Counter` - contract name only
625#[derive(Clone, PartialEq, Eq)]
626pub enum PathOrContractInfo {
627    /// Non-canonicalized path provided via CLI.
628    Path(PathBuf),
629    /// Contract info provided via CLI.
630    ContractInfo(CompilerContractInfo),
631}
632
633impl PathOrContractInfo {
634    /// Returns the path to the contract file if provided.
635    pub fn path(&self) -> Option<PathBuf> {
636        match self {
637            Self::Path(path) => Some(path.to_path_buf()),
638            Self::ContractInfo(info) => info.path.as_ref().map(PathBuf::from),
639        }
640    }
641
642    /// Returns the contract name if provided.
643    pub fn name(&self) -> Option<&str> {
644        match self {
645            Self::Path(_) => None,
646            Self::ContractInfo(info) => Some(&info.name),
647        }
648    }
649}
650
651impl FromStr for PathOrContractInfo {
652    type Err = eyre::Error;
653
654    fn from_str(s: &str) -> Result<Self> {
655        if let Ok(contract) = CompilerContractInfo::from_str(s) {
656            return Ok(Self::ContractInfo(contract));
657        }
658        let path = PathBuf::from(s);
659        if path.extension().is_some_and(|ext| ext == "sol" || ext == "vy") {
660            return Ok(Self::Path(path));
661        }
662        Err(eyre::eyre!("Invalid contract identifier, file is not *.sol or *.vy: {}", s))
663    }
664}
665
666impl std::fmt::Debug for PathOrContractInfo {
667    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
668        match self {
669            Self::Path(path) => write!(f, "Path({})", path.display()),
670            Self::ContractInfo(info) => {
671                write!(f, "ContractInfo({info})")
672            }
673        }
674    }
675}
