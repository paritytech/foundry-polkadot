50/// A multi contract runner receives a set of contracts deployed in an EVM instance and proceeds
51/// to run all test functions in these contracts.
52#[derive(Clone, Debug)]
53pub struct MultiContractRunner {
54    /// Mapping of contract name to JsonAbi, creation bytecode and library bytecode which
55    /// needs to be deployed & linked against
56    pub contracts: DeployableContracts,

59    /// Revert decoder. Contains all known errors and their selectors.
60    pub revert_decoder: RevertDecoder,
61    /// Libraries to deploy.

63    /// Library addresses used to link contracts.
64    pub libraries: Libraries,

66    pub analysis: Arc<solar::sema::Compiler>,

69    pub fork: Option<CreateFork>,
70
71    /// The base configuration for the test runner.

92impl MultiContractRunner {
93    /// Returns an iterator over all contracts that match the filter.
94    pub fn matching_contracts<'a: 'b, 'b>(
95        &'a self,
96        filter: &'b dyn TestFilter,
97    ) -> impl Iterator<Item = (&'a ArtifactId, &'a TestContract)> + 'b {
98        self.contracts.iter().filter(|&(id, c)| matches_artifact(filter, id, &c.abi))
99    }
100
101    /// Returns an iterator over all test functions that match the filter.
102    pub fn matching_test_functions<'a: 'b, 'b>(
103        &'a self,
104        filter: &'b dyn TestFilter,
105    ) -> impl Iterator<Item = &'a Function> + 'b {
106        self.matching_contracts(filter)
107            .flat_map(|(_, c)| c.abi.functions())
108            .filter(|func| filter.matches_test_function(func))
109    }
110
111    /// Returns an iterator over all test functions in contracts that match the filter.
112    pub fn all_test_functions<'a: 'b, 'b>(
113        &'a self,
114        filter: &'b dyn TestFilter,
115    ) -> impl Iterator<Item = &'a Function> + 'b {
116        self.contracts
117            .iter()
118            .filter(|(id, _)| filter.matches_path(&id.source) && filter.matches_contract(&id.name))
119            .flat_map(|(_, c)| c.abi.functions())
120            .filter(|func| func.is_any_test())
121    }
122
123    /// Returns all matching tests grouped by contract grouped by file (file -> (contract -> tests))
124    pub fn list(&self, filter: &dyn TestFilter) -> BTreeMap<String, BTreeMap<String, Vec<String>>> {
125        self.matching_contracts(filter)
126            .map(|(id, c)| {
127                let source = id.source.as_path().display().to_string();
128                let name = id.name.clone();
129                let tests = c
130                    .abi
131                    .functions()
132                    .filter(|func| filter.matches_test_function(func))
133                    .map(|func| func.name.clone())
134                    .collect::<Vec<_>>();
135                (source, name, tests)
136            })
137            .fold(BTreeMap::new(), |mut acc, (source, name, tests)| {
138                acc.entry(source).or_default().insert(name, tests);
139                acc
140            })
141    }
142
143    /// Executes _all_ tests that match the given `filter`.
144    ///
145    /// The same as [`test`](Self::test), but returns the results instead of streaming them.
146    ///
147    /// Note that this method returns only when all tests have been executed.
148    pub fn test_collect(
149        &mut self,
150        filter: &dyn TestFilter,
151    ) -> Result<BTreeMap<String, SuiteResult>> {
152        Ok(self.test_iter(filter)?.collect())
153    }
154
155    /// Executes _all_ tests that match the given `filter`.
156    ///
157    /// The same as [`test`](Self::test), but returns the results instead of streaming them.
158    ///
159    /// Note that this method returns only when all tests have been executed.
160    pub fn test_iter(
161        &mut self,
162        filter: &dyn TestFilter,
163    ) -> Result<impl Iterator<Item = (String, SuiteResult)>> {
164        let (tx, rx) = mpsc::channel();
165        self.test(filter, tx, false)?;
166        Ok(rx.into_iter())
167    }
168
169    /// Executes _all_ tests that match the given `filter`.
170    ///
171    /// This will create the runtime based on the configured `evm` ops and create the `Backend`
172    /// before executing all contracts and their tests in _parallel_.
173    ///
174    /// Each Executor gets its own instance of the `Backend`.
175    pub fn test(
176        &mut self,
177        filter: &dyn TestFilter,
178        tx: mpsc::Sender<(String, SuiteResult)>,
179        show_progress: bool,
180    ) -> Result<()> {
181        let tokio_handle = tokio::runtime::Handle::current();
182        trace!("running all tests");
183
184        // The DB backend that serves all the data.
185        let db = Backend::spawn(
186            self.fork.take(),
187            self.strategy.runner.new_backend_strategy(self.strategy.context.as_ref()),
188        )?;
189
190        let find_timer = Instant::now();
191        let contracts = self.matching_contracts(filter).collect::<Vec<_>>();
192        let find_time = find_timer.elapsed();
193        debug!(
194            "Found {} test contracts out of {} in {:?}",
195            contracts.len(),
196            self.contracts.len(),
197            find_time,
198        );
199
200        if show_progress {
201            let tests_progress = TestsProgress::new(contracts.len(), rayon::current_num_threads());
202            // Collect test suite results to stream at the end of test run.
203            let results: Vec<(String, SuiteResult)> = contracts
204                .par_iter()
205                .map(|&(id, contract)| {
206                    let _guard = tokio_handle.enter();
207                    tests_progress.inner.lock().start_suite_progress(&id.identifier());
208
209                    let result = self.run_test_suite(
210                        id,
211                        contract,
212                        &db,
213                        filter,
214                        &tokio_handle,
215                        Some(&tests_progress),
216                    );
217
218                    tests_progress
219                        .inner
220                        .lock()
221                        .end_suite_progress(&id.identifier(), result.summary());
222
223                    (id.identifier(), result)
224                })
225                .collect();
226
227            tests_progress.inner.lock().clear();
228
229            results.iter().for_each(|result| {
230                let _ = tx.send(result.to_owned());
231            });
232        } else {
233            contracts.par_iter().for_each(|&(id, contract)| {
234                let _guard = tokio_handle.enter();
235                let result = self.run_test_suite(id, contract, &db, filter, &tokio_handle, None);
236                let _ = tx.send((id.identifier(), result));
237            })
238        }
239
240        Ok(())
241    }
242
243    fn run_test_suite(
244        &self,
245        artifact_id: &ArtifactId,
246        contract: &TestContract,
247        db: &Backend,
248        filter: &dyn TestFilter,
249        tokio_handle: &tokio::runtime::Handle,
250        progress: Option<&TestsProgress>,
251    ) -> SuiteResult {
252        let identifier = artifact_id.identifier();
253        let mut span_name = identifier.as_str();
254
255        if !enabled!(tracing::Level::TRACE) {
256            span_name = get_contract_name(&identifier);
257        }
258        let span = debug_span!("suite", name = %span_name);
259        let span_local = span.clone();
260        let _guard = span_local.enter();
261
262        debug!("start executing all tests in contract");
263
264        let executor = self.tcfg.executor(
265            self.strategy.clone(),
266            self.known_contracts.clone(),
267            self.analysis.clone(),
268            artifact_id,
269            db.clone(),
270        );
271        let runner = ContractRunner::new(
272            &identifier,
273            contract,
274            executor,
275            progress,
276            tokio_handle,
277            span,
278            self,
279        );
280        let r = runner.run_tests(filter);
281
282        debug!(duration=?r.duration, "executed all tests in contract");
283
284        r
285    }

367    /// Creates a new executor with this configuration.
368    pub fn executor(
369        &self,
370        strategy: ExecutorStrategy,
371        known_contracts: ContractsByArtifact,
372        analysis: Arc<solar::sema::Compiler>,
373        artifact_id: &ArtifactId,
374        db: Backend,
375    ) -> Executor {
376        let cheats_config = Arc::new(CheatsConfig::new(
377            strategy.runner.new_cheatcodes_strategy(strategy.context.as_ref()),
378            &self.config,
379            self.evm_opts.clone(),
380            Some(known_contracts),
381            Some(artifact_id.clone()),
382        ));
383        ExecutorBuilder::new()
384            .inspectors(|stack| {
385                stack
386                    .cheatcodes(cheats_config)
387                    .trace_mode(self.trace_mode())
388                    .line_coverage(self.line_coverage)
389                    .enable_isolation(self.isolation)
390                    .networks(self.networks)
391                    .create2_deployer(self.evm_opts.create2_deployer)
392                    .set_analysis(analysis)
393            })
394            .spec_id(self.spec_id)
395            .gas_limit(self.evm_opts.gas_limit())
396            .legacy_assertions(self.config.legacy_assertions)
397            .build(self.env.clone(), db, strategy)
398    }

400    fn trace_mode(&self) -> TraceMode {

405            .with_state_changes(verbosity() > 4)
406    }

505    /// Given an EVM, proceeds to return a runner which is able to execute all tests
506    /// against that evm
507    pub fn build<C: Compiler<CompilerContract = Contract>>(
508        self,
509        mut strategy: ExecutorStrategy,
510        output: std::sync::Arc<ProjectCompileOutput>,
511        resolc_output: Option<std::sync::Arc<ProjectCompileOutput>>,
512        env: Env,
513        evm_opts: EvmOpts,
514    ) -> Result<MultiContractRunner> {
515        strategy.runner.revive_set_compilation_output(strategy.context.as_mut(), output.clone());
516        if let Some(resolc_output) = resolc_output {
517            strategy.runner.revive_set_resolc_output(strategy.context.as_mut(), resolc_output);
518        }
519
520        let root = &self.config.root;
521        let contracts = output
522            .artifact_ids()
523            .map(|(id, v)| (id.with_stripped_file_prefixes(root), v))
524            .collect();
525        let linker = Linker::new(root, contracts);
526
527        // Build revert decoder from ABIs of all artifacts.
528        let abis = linker
529            .contracts
530            .values()
531            .filter_map(|contract| contract.abi.as_ref().map(|abi| abi.borrow()));
532        let revert_decoder = RevertDecoder::new().with_abis(abis);
533
534        let LinkOutput { libraries, libs_to_deploy } = linker.link_with_nonce_or_address(
535            Default::default(),
536            LIBRARY_DEPLOYER,
537            0,
538            linker.contracts.keys(),
539        )?;
540
541        let linked_contracts = linker.get_linked_artifacts_cow(&libraries)?;
542
543        strategy.runner.revive_link_libraries(
544            strategy.context.as_mut(),
545            &self.config,
546            root,
547            &linked_contracts,
548            &libraries,
549        )?;
550
551        // Create a mapping of name => (abi, deployment code, Vec<library deployment code>)
552        let mut deployable_contracts = DeployableContracts::default();
553
554        for (id, contract) in linked_contracts.iter() {
555            let Some(abi) = contract.abi.as_ref() else { continue };
556
557            // if it's a test, link it and add to deployable contracts
558            if abi.constructor.as_ref().map(|c| c.inputs.is_empty()).unwrap_or(true)
559                && abi.functions().any(|func| func.name.is_any_test())
560            {
561                linker.ensure_linked(contract, id)?;
562
563                let Some(bytecode) =
564                    contract.get_bytecode_bytes().map(|b| b.into_owned()).filter(|b| !b.is_empty())
565                else {
566                    continue;
567                };
568
569                deployable_contracts
570                    .insert(id.clone(), TestContract { abi: abi.clone().into_owned(), bytecode });
571            }
572        }
573
574        // Create known contracts from linked contracts and storage layout information (if any).
575        let known_contracts =
576            ContractsByArtifactBuilder::new(linked_contracts).with_output(&output, root).build();
577
578        // Initialize and configure the solar compiler.
579        let mut analysis = solar::sema::Compiler::new(
580            solar::interface::Session::builder().with_stderr_emitter().build(),
581        );
582        let dcx = analysis.dcx_mut();
583        dcx.set_emitter(Box::new(
584            solar::interface::diagnostics::HumanEmitter::stderr(Default::default())
585                .source_map(Some(dcx.source_map().unwrap())),
586        ));
587        dcx.set_flags_mut(|f| f.track_diagnostics = false);
588
589        // Populate solar's global context by parsing and lowering the sources.
590        let files: Vec<_> =
591            output.output().sources.as_ref().keys().map(|path| path.to_path_buf()).collect();
592
593        analysis.enter_mut(|compiler| -> Result<()> {
594            let mut pcx = compiler.parse();
595            configure_pcx_from_compile_output(
596                &mut pcx,
597                &self.config,
598                &output,
599                if files.is_empty() { None } else { Some(&files) },
600            )?;
601            pcx.parse();
602            let _ = compiler.lower_asts();
603            Ok(())
604        })?;
605
606        Ok(MultiContractRunner {
607            contracts: deployable_contracts,
608            revert_decoder,
609            known_contracts,
610            libs_to_deploy,
611            libraries,
612            analysis: Arc::new(analysis),
613
614            tcfg: TestRunnerConfig {
615                evm_opts,
616                env,
617                spec_id: self.evm_spec.unwrap_or_else(|| self.config.evm_spec_id()),
618                sender: self.sender.unwrap_or(self.config.sender),
619                line_coverage: self.line_coverage,
620                debug: self.debug,
621                decode_internal: self.decode_internal,
622                inline_config: Arc::new(InlineConfig::new_parsed(&output, &self.config)?),
623                isolation: self.isolation,
624                networks: self.networks,
625                early_exit: EarlyExit::new(self.fail_fast || self.config.show_progress),
626                config: self.config,
627            },
628            strategy,
629
630            fork: self.fork,
631        })
632    }

635pub fn matches_artifact(filter: &dyn TestFilter, id: &ArtifactId, abi: &JsonAbi) -> bool {
636    matches_contract(filter, &id.source, &id.name, abi.functions())
637}

639pub(crate) fn matches_contract(
640    filter: &dyn TestFilter,
641    path: &Path,
642    contract_name: &str,
643    functions: impl IntoIterator<Item = impl std::borrow::Borrow<Function>>,
644) -> bool {
645    (filter.matches_path(path) && filter.matches_contract(contract_name))
646        && functions.into_iter().any(|func| filter.matches_test_function(func.borrow()))
647}
