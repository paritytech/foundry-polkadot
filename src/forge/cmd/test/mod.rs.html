231    /// Returns a list of files that need to be compiled in order to run all the tests that match
232    /// the given filter.
233    ///
234    /// This means that it will return all sources that are not test contracts or that match the
235    /// filter. We want to compile all non-test sources always because tests might depend on them
236    /// dynamically through cheatcodes.
237    #[instrument(target = "forge::test", skip_all)]
238    pub fn get_sources_to_compile(
239        &self,
240        config: &Config,
241        test_filter: &ProjectPathsAwareFilter,
242    ) -> Result<BTreeSet<PathBuf>> {
243        // An empty filter doesn't filter out anything.
244        // We can still optimize slightly by excluding scripts.
245        if test_filter.is_empty() {
246            return Ok(source_files_iter(&config.src, MultiCompilerLanguage::FILE_EXTENSIONS)
247                .chain(source_files_iter(&config.test, MultiCompilerLanguage::FILE_EXTENSIONS))
248                .collect());
249        }
250
251        let mut project = config.create_project(true, true)?;
252        project.update_output_selection(|selection| {
253            *selection = OutputSelection::common_output_selection(["abi".to_string()]);
254        });
255        let output = project.compile()?;
256        if output.has_compiler_errors() {
257            sh_println!("{output}")?;
258            eyre::bail!("Compilation failed");
259        }
260
261        Ok(output
262            .artifact_ids()
263            .filter_map(|(id, artifact)| artifact.abi.as_ref().map(|abi| (id, abi)))
264            .filter(|(id, abi)| {
265                id.source.starts_with(&config.src) || matches_artifact(test_filter, id, abi)
266            })
267            .map(|(id, _)| id.source)
268            .collect())
269    }

271    /// Executes all the tests in the project.
272    ///
273    /// This will trigger the build process first. On success all test contracts that match the
274    /// configured filter will be executed
275    ///
276    /// Returns the test results for all matching tests.
277    pub async fn compile_and_run(&mut self) -> Result<TestOutcome> {
278        // Merge all configs.
279        let (mut config, mut evm_opts) = self.load_config_and_evm_opts()?;
280
281        // Override polkadot mode from CLI flag if provided
282        if let Some(polkadot_mode) = self.polkadot {
283            config.polkadot.polkadot = Some(polkadot_mode);
284            // Auto-enable resolc_compile when using --polkadot=pvm (required for dual compilation)
285            if polkadot_mode == PolkadotMode::Pvm {
286                tracing::warn!(
287                    "Using 'pvm' backend is an experimental feature and may lead to unexpected behavior in tests."
288                );
289                config.polkadot.resolc_compile = true;
290            }
291        }
292
293        // Auto-set polkadot=pvm when --resolc is used without explicit --polkadot flag
294        if config.polkadot.resolc_compile && config.polkadot.polkadot.is_none() {
295            tracing::warn!(
296                "Using 'pvm' backend is an experimental feature and may lead to unexpected behavior in tests."
297            );
298            config.polkadot.polkadot = Some(PolkadotMode::Pvm);
299        }
300
301        // Explicitly enable isolation for gas reports for more correct gas accounting.
302        if self.gas_report {
303            evm_opts.isolate = true;
304        } else {
305            // Do not collect gas report traces if gas report is not enabled.
306            config.fuzz.gas_report_samples = 0;
307            config.invariant.gas_report_samples = 0;
308        }
309
310        // Install missing dependencies.
311        if install::install_missing_dependencies(&mut config).await && config.auto_detect_remappings
312        {
313            // need to re-configure here to also catch additional remappings
314            config = self.load_config()?;
315        }
316        if config.polkadot.resolc_compile {
317            config.extra_output.push(ContractOutputSelection::StorageLayout);
318        }
319        // Set up the project.
320        let project = config.project()?;
321
322        let filter = self.filter(&config)?;
323        trace!(target: "forge::test", ?filter, "using filter");
324
325        let sources_to_compile = self.get_sources_to_compile(&config, &filter)?;
326
327        // Handle compilation based on whether dual compilation is enabled
328        let (output, dual_compiled_contracts, resolc_output) = if config.polkadot.resolc_compile {
329            // Dual compilation mode: compile both solc and resolc
330
331            // Compile with solc to a subdirectory
332            let mut solc_config = config.clone();
333            solc_config.out = solc_config.out.join(revive::SOLC_ARTIFACTS_SUBDIR);
334            solc_config.polkadot = Default::default();
335            solc_config.build_info_path = Some(solc_config.out.join("build-info"));
336            let solc_project = solc_config.project()?;
337            let compiler = ProjectCompiler::new()
338                .dynamic_test_linking(config.dynamic_test_linking)
339                .quiet(shell::is_json() || self.junit)
340                .files(sources_to_compile.clone());
341
342            let solc_output = compiler.compile(&solc_project)?;
343
344            let resolc_project = config.project()?;
345
346            let resolc_compiler = ProjectCompiler::new()
347                .quiet(shell::is_json() || self.junit)
348                .files(sources_to_compile)
349                .size_limits(revive::CONTRACT_SIZE_LIMIT, revive::CONTRACT_SIZE_LIMIT);
350
351            let resolc_output = resolc_compiler.compile(&resolc_project)?;
352
353            let dual_compiled_contracts = DualCompiledContracts::new(
354                &solc_output,
355                &resolc_output,
356                &solc_project.paths,
357                &resolc_project.paths,
358            );
359
360            (solc_output, Some(dual_compiled_contracts), Some(std::sync::Arc::new(resolc_output)))
361        } else {
362            // Single compilation mode: compile only with solc
363
364            let compiler: ProjectCompiler = ProjectCompiler::new()
365                .dynamic_test_linking(config.dynamic_test_linking)
366                .quiet(shell::is_json() || self.junit)
367                .files(sources_to_compile.clone());
368
369            let solc_output = compiler.compile(&project)?;
370
371            (solc_output, None, None)
372        };
373
374        self.run_tests(
375            &project.paths.root,
376            config,
377            evm_opts,
378            std::sync::Arc::new(output),
379            resolc_output,
380            &filter,
381            false,
382            dual_compiled_contracts,
383        )
384        .await
385    }

387    /// Executes all the tests in the project.
388    ///
389    /// See [`Self::compile_and_run`] for more details.
390    #[allow(clippy::too_many_arguments)]
391    pub async fn run_tests(
392        &mut self,
393        project_root: &Path,
394        mut config: Config,
395        mut evm_opts: EvmOpts,
396        output: std::sync::Arc<ProjectCompileOutput>,
397        resolc_output: Option<std::sync::Arc<ProjectCompileOutput>>,
398        filter: &ProjectPathsAwareFilter,
399        coverage: bool,
400        dual_compiled_contracts: Option<DualCompiledContracts>,
401    ) -> Result<TestOutcome> {
402        let mut strategy = utils::get_executor_strategy(&config);
403
404        // Explicitly enable isolation for gas reports for more correct gas accounting.
405        if self.gas_report {
406            evm_opts.isolate = true;
407        } else {
408            // Do not collect gas report traces if gas report is not enabled.
409            config.fuzz.gas_report_samples = 0;
410            config.invariant.gas_report_samples = 0;
411        }
412
413        // Create test options from general project settings and compiler output.
414        let should_debug = self.debug;
415        let should_draw = self.flamegraph || self.flamechart;
416
417        // Determine print verbosity and executor verbosity.
418        let verbosity = evm_opts.verbosity;
419        if (self.gas_report && evm_opts.verbosity < 3) || self.flamegraph || self.flamechart {
420            evm_opts.verbosity = 3;
421        }
422
423        let env = evm_opts.evm_env().await?;
424
425        // Enable internal tracing for more informative flamegraph.
426        if should_draw && !self.decode_internal {
427            self.decode_internal = true;
428        }
429
430        // Choose the internal function tracing mode, if --decode-internal is provided.
431        let decode_internal = if self.decode_internal {
432            // If more than one function matched, we enable simple tracing.
433            // If only one function matched, we enable full tracing. This is done in `run_tests`.
434            InternalTraceMode::Simple
435        } else {
436            InternalTraceMode::None
437        };
438
439        // Prepare the test builder.
440        let config = Arc::new(config);
441
442        // Set dual compiled contracts on the strategy
443        strategy.runner.revive_set_dual_compiled_contracts(
444            strategy.context.as_mut(),
445            dual_compiled_contracts.unwrap_or_default(),
446        );
447
448        let runner = MultiContractRunnerBuilder::new(config.clone())
449            .set_debug(should_debug)
450            .set_decode_internal(decode_internal)
451            .initial_balance(evm_opts.initial_balance)
452            .evm_spec(config.evm_spec_id())
453            .sender(evm_opts.sender)
454            .with_fork(evm_opts.get_fork(&config, env.clone()))
455            .enable_isolation(evm_opts.isolate)
456            .networks(evm_opts.networks)
457            .fail_fast(self.fail_fast)
458            .set_coverage(coverage)
459            .build::<MultiCompiler>(strategy, output.clone(), resolc_output, env, evm_opts)?;
460
461        let libraries = runner.libraries.clone();
462        let mut outcome = self.run_tests_inner(runner, config, verbosity, filter, &output).await?;
463
464        if should_draw {
465            let (suite_name, test_name, mut test_result) =
466                outcome.remove_first().ok_or_eyre("no tests were executed")?;
467
468            let (_, arena) = test_result
469                .traces
470                .iter_mut()
471                .find(|(kind, _)| *kind == TraceKind::Execution)
472                .unwrap();
473
474            // Decode traces.
475            let decoder = outcome.last_run_decoder.as_ref().unwrap();
476            decode_trace_arena(arena, decoder).await;
477            let mut fst = folded_stack_trace::build(arena);
478
479            let label = if self.flamegraph { "flamegraph" } else { "flamechart" };
480            let contract = suite_name.split(':').next_back().unwrap();
481            let test_name = test_name.trim_end_matches("()");
482            let file_name = format!("cache/{label}_{contract}_{test_name}.svg");
483            let file = std::fs::File::create(&file_name).wrap_err("failed to create file")?;
484            let file = std::io::BufWriter::new(file);
485
486            let mut options = inferno::flamegraph::Options::default();
487            options.title = format!("{label} {contract}::{test_name}");
488            options.count_name = "gas".to_string();
489            if self.flamechart {
490                options.flame_chart = true;
491                fst.reverse();
492            }
493
494            // Generate SVG.
495            inferno::flamegraph::from_lines(&mut options, fst.iter().map(String::as_str), file)
496                .wrap_err("failed to write svg")?;
497            sh_println!("Saved to {file_name}")?;
498
499            // Open SVG in default program.
500            if let Err(e) = opener::open(&file_name) {
501                sh_err!("Failed to open {file_name}; please open it manually: {e}")?;
502            }
503        }
504
505        if should_debug {
506            // Get first non-empty suite result. We will have only one such entry.
507            let (_, _, test_result) =
508                outcome.remove_first().ok_or_eyre("no tests were executed")?;
509
510            let sources =
511                ContractSources::from_project_output(&output, project_root, Some(&libraries))?;
512
513            // Run the debugger.
514            let mut builder = Debugger::builder()
515                .traces(
516                    test_result.traces.iter().filter(|(t, _)| t.is_execution()).cloned().collect(),
517                )
518                .sources(sources)
519                .breakpoints(test_result.breakpoints.clone());
520
521            if let Some(decoder) = &outcome.last_run_decoder {
522                builder = builder.decoder(decoder);
523            }
524
525            let mut debugger = builder.build();
526            if let Some(dump_path) = &self.dump {
527                debugger.dump_to_file(dump_path)?;
528            } else {
529                debugger.try_run_tui()?;
530            }
531        }
532
533        Ok(outcome)
534    }

536    /// Run all tests that matches the filter predicate from a test runner
537    async fn run_tests_inner(
538        &self,
539        mut runner: MultiContractRunner,
540        config: Arc<Config>,
541        verbosity: u8,
542        filter: &ProjectPathsAwareFilter,
543        output: &ProjectCompileOutput,
544    ) -> eyre::Result<TestOutcome> {
545        if self.list {
546            return list(runner, filter);
547        }
548
549        trace!(target: "forge::test", "running all tests");
550
551        // If we need to render to a serialized format, we should not print anything else to stdout.
552        let silent = self.gas_report && shell::is_json() || self.summary && shell::is_json();
553
554        let num_filtered = runner.matching_test_functions(filter).count();
555
556        if num_filtered == 0 {
557            let mut total_tests = num_filtered;
558            if !filter.is_empty() {
559                total_tests = runner.matching_test_functions(&EmptyTestFilter::default()).count();
560            }
561            if total_tests == 0 {
562                sh_println!(
563                    "No tests found in project! Forge looks for functions that start with `test`"
564                )?;
565            } else {
566                let mut msg = format!("no tests match the provided pattern:\n{filter}");
567                // Try to suggest a test when there's no match.
568                if let Some(test_pattern) = &filter.args().test_pattern {
569                    let test_name = test_pattern.as_str();
570                    // Filter contracts but not test functions.
571                    let candidates = runner.all_test_functions(filter).map(|f| &f.name);
572                    if let Some(suggestion) = utils::did_you_mean(test_name, candidates).pop() {
573                        write!(msg, "\nDid you mean `{suggestion}`?")?;
574                    }
575                }
576                sh_warn!("{msg}")?;
577            }
578            return Ok(TestOutcome::empty(Some(runner), false));
579        }
580
581        if num_filtered != 1 && (self.debug || self.flamegraph || self.flamechart) {
582            let action = if self.flamegraph {
583                "generate a flamegraph"
584            } else if self.flamechart {
585                "generate a flamechart"
586            } else {
587                "run the debugger"
588            };
589            let filter = if filter.is_empty() {
590                String::new()
591            } else {
592                format!("\n\nFilter used:\n{filter}")
593            };
594            eyre::bail!(
595                "{num_filtered} tests matched your criteria, but exactly 1 test must match in order to {action}.\n\n\
596                 Use --match-contract and --match-path to further limit the search.{filter}",
597            );
598        }
599
600        // If exactly one test matched, we enable full tracing.
601        if num_filtered == 1 && self.decode_internal {
602            runner.decode_internal = InternalTraceMode::Full;
603        }
604
605        // Run tests in a non-streaming fashion and collect results for serialization.
606        if !self.gas_report && !self.summary && shell::is_json() {
607            let mut results = runner.test_collect(filter)?;
608            results.values_mut().for_each(|suite_result| {
609                for test_result in suite_result.test_results.values_mut() {
610                    if verbosity >= 2 {
611                        // Decode logs at level 2 and above.
612                        test_result.decoded_logs = decode_console_logs(&test_result.logs);
613                    } else {
614                        // Empty logs for non verbose runs.
615                        test_result.logs = vec![];
616                    }
617                }
618            });
619            sh_println!("{}", serde_json::to_string(&results)?)?;
620            return Ok(TestOutcome::new(Some(runner), results, self.allow_failure));
621        }
622
623        if self.junit {
624            let results = runner.test_collect(filter)?;
625            sh_println!("{}", junit_xml_report(&results, verbosity).to_string()?)?;
626            return Ok(TestOutcome::new(Some(runner), results, self.allow_failure));
627        }
628
629        let remote_chain =
630            if runner.fork.is_some() { runner.env.tx.chain_id.map(Into::into) } else { None };
631        let known_contracts = runner.known_contracts.clone();
632
633        let libraries = runner.libraries.clone();
634
635        // Run tests in a streaming fashion.
636        let (tx, rx) = channel::<(String, SuiteResult)>();
637        let timer = Instant::now();
638        let show_progress = config.show_progress;
639        let handle = tokio::task::spawn_blocking({
640            let filter = filter.clone();
641            move || runner.test(&filter, tx, show_progress).map(|()| runner)
642        });
643
644        // Set up trace identifiers.
645        let mut identifier = TraceIdentifiers::new().with_local(&known_contracts);
646
647        // Avoid using external identifiers for gas report as we decode more traces and this will be
648        // expensive.
649        if !self.gas_report {
650            identifier = identifier.with_external(&config, remote_chain)?;
651        }
652
653        // Build the trace decoder.
654        let mut builder = CallTraceDecoderBuilder::new()
655            .with_known_contracts(&known_contracts)
656            .with_label_disabled(self.disable_labels)
657            .with_verbosity(verbosity);
658        // Signatures are of no value for gas reports.
659        if !self.gas_report {
660            builder =
661                builder.with_signature_identifier(SignaturesIdentifier::from_config(&config)?);
662        }
663
664        if self.decode_internal {
665            let sources =
666                ContractSources::from_project_output(output, &config.root, Some(&libraries))?;
667            builder = builder.with_debug_identifier(DebugTraceIdentifier::new(sources));
668        }
669        let mut decoder = builder.build();
670
671        let mut gas_report = self.gas_report.then(|| {
672            GasReport::new(
673                config.gas_reports.clone(),
674                config.gas_reports_ignore.clone(),
675                config.gas_reports_include_tests,
676            )
677        });
678
679        let mut gas_snapshots = BTreeMap::<String, BTreeMap<String, String>>::new();
680
681        let mut outcome = TestOutcome::empty(None, self.allow_failure);
682
683        let mut any_test_failed = false;
684        let mut backtrace_builder = None;
685        for (contract_name, mut suite_result) in rx {
686            let tests = &mut suite_result.test_results;
687            let has_tests = !tests.is_empty();
688
689            // Clear the addresses and labels from previous test.
690            decoder.clear_addresses();
691
692            // We identify addresses if we're going to print *any* trace or gas report.
693            let identify_addresses = verbosity >= 3
694                || self.gas_report
695                || self.debug
696                || self.flamegraph
697                || self.flamechart;
698
699            // Print suite header.
700            if !silent {
701                sh_println!()?;
702                for warning in &suite_result.warnings {
703                    sh_warn!("{warning}")?;
704                }
705                if has_tests {
706                    let len = tests.len();
707                    let tests = if len > 1 { "tests" } else { "test" };
708                    sh_println!("Ran {len} {tests} for {contract_name}")?;
709                }
710            }
711
712            // Process individual test results, printing logs and traces when necessary.
713            for (name, result) in tests {
714                let show_traces =
715                    !self.suppress_successful_traces || result.status == TestStatus::Failure;
716                if !silent {
717                    sh_println!("{}", result.short_result(name))?;
718
719                    // Display invariant metrics if invariant kind.
720                    if let TestKind::Invariant { metrics, .. } = &result.kind
721                        && !metrics.is_empty()
722                    {
723                        let _ = sh_println!("\n{}\n", format_invariant_metrics_table(metrics));
724                    }
725
726                    // We only display logs at level 2 and above
727                    if verbosity >= 2 && show_traces {
728                        // We only decode logs from Hardhat and DS-style console events
729                        let console_logs = decode_console_logs(&result.logs);
730                        if !console_logs.is_empty() {
731                            sh_println!("Logs:")?;
732                            for log in console_logs {
733                                sh_println!("  {log}")?;
734                            }
735                            sh_println!()?;
736                        }
737                    }
738                }
739
740                // We shouldn't break out of the outer loop directly here so that we finish
741                // processing the remaining tests and print the suite summary.
742                any_test_failed |= result.status == TestStatus::Failure;
743
744                // Clear the addresses and labels from previous runs.
745                decoder.clear_addresses();
746                decoder.labels.extend(result.labels.iter().map(|(k, v)| (*k, v.clone())));
747
748                // Identify addresses and decode traces.
749                let mut decoded_traces = Vec::with_capacity(result.traces.len());
750                for (kind, arena) in &mut result.traces {
751                    if identify_addresses {
752                        decoder.identify(arena, &mut identifier);
753                    }
754
755                    // verbosity:
756                    // - 0..3: nothing
757                    // - 3: only display traces for failed tests
758                    // - 4: also display the setup trace for failed tests
759                    // - 5..: display all traces for all tests, including storage changes
760                    let should_include = match kind {
761                        TraceKind::Execution => {
762                            (verbosity == 3 && result.status.is_failure()) || verbosity >= 4
763                        }
764                        TraceKind::Setup => {
765                            (verbosity == 4 && result.status.is_failure()) || verbosity >= 5
766                        }
767                        TraceKind::Deployment => false,
768                    };
769
770                    if should_include {
771                        decode_trace_arena(arena, &decoder).await;
772                        decoded_traces.push(render_trace_arena_inner(arena, false, verbosity > 4));
773                    }
774                }
775
776                if !silent && show_traces && !decoded_traces.is_empty() {
777                    sh_println!("Traces:")?;
778                    for trace in &decoded_traces {
779                        sh_println!("{trace}")?;
780                    }
781                }
782
783                // Extract and display backtrace for failed tests when verbosity >= 3
784                if !silent
785                    && result.status.is_failure()
786                    && verbosity >= 3
787                    && !result.traces.is_empty()
788                    && let Some((_, arena)) =
789                        result.traces.iter().find(|(kind, _)| matches!(kind, TraceKind::Execution))
790                {
791                    // Lazily initialize the backtrace builder on first failure
792                    let builder = backtrace_builder.get_or_insert_with(|| {
793                        BacktraceBuilder::new(
794                            output,
795                            config.root.clone(),
796                            config.parsed_libraries().ok(),
797                            config.via_ir,
798                        )
799                    });
800
801                    let backtrace = builder.from_traces(arena);
802
803                    if !backtrace.is_empty() {
804                        sh_println!("{}", backtrace)?;
805                    }
806                }
807
808                if let Some(gas_report) = &mut gas_report {
809                    gas_report.analyze(result.traces.iter().map(|(_, a)| &a.arena), &decoder).await;
810
811                    for trace in &result.gas_report_traces {
812                        decoder.clear_addresses();
813
814                        // Re-execute setup and deployment traces to collect identities created in
815                        // setUp and constructor.
816                        for (kind, arena) in &result.traces {
817                            if !matches!(kind, TraceKind::Execution) {
818                                decoder.identify(arena, &mut identifier);
819                            }
820                        }
821
822                        for arena in trace {
823                            decoder.identify(arena, &mut identifier);
824                            gas_report.analyze([arena], &decoder).await;
825                        }
826                    }
827                }
828                // Clear memory.
829                result.gas_report_traces = Default::default();
830
831                // Collect and merge gas snapshots.
832                for (group, new_snapshots) in &result.gas_snapshots {
833                    gas_snapshots.entry(group.clone()).or_default().extend(new_snapshots.clone());
834                }
835            }
836
837            // Write gas snapshots to disk if any were collected.
838            if !gas_snapshots.is_empty() {
839                // By default `gas_snapshot_check` is set to `false` in the config.
840                //
841                // The user can either:
842                // - Set `FORGE_SNAPSHOT_CHECK=true` in the environment.
843                // - Pass `--gas-snapshot-check=true` as a CLI argument.
844                // - Set `gas_snapshot_check = true` in the config.
845                //
846                // If the user passes `--gas-snapshot-check=<bool>` then it will override the config
847                // and the environment variable, disabling the check if `false` is passed.
848                //
849                // Exiting early with code 1 if differences are found.
850                if self.gas_snapshot_check.unwrap_or(config.gas_snapshot_check) {
851                    let differences_found = gas_snapshots.clone().into_iter().fold(
852                        false,
853                        |mut found, (group, snapshots)| {
854                            // If the snapshot file doesn't exist, we can't compare so we skip.
855                            if !&config.snapshots.join(format!("{group}.json")).exists() {
856                                return false;
857                            }
858
859                            let previous_snapshots: BTreeMap<String, String> =
860                                fs::read_json_file(&config.snapshots.join(format!("{group}.json")))
861                                    .expect("Failed to read snapshots from disk");
862
863                            let diff: BTreeMap<_, _> = snapshots
864                                .iter()
865                                .filter_map(|(k, v)| {
866                                    previous_snapshots.get(k).and_then(|previous_snapshot| {
867                                        if previous_snapshot != v {
868                                            Some((
869                                                k.clone(),
870                                                (previous_snapshot.clone(), v.clone()),
871                                            ))
872                                        } else {
873                                            None
874                                        }
875                                    })
876                                })
877                                .collect();
878
879                            if !diff.is_empty() {
880                                let _ = sh_eprintln!(
881                                    "{}",
882                                    format!("\n[{group}] Failed to match snapshots:").red().bold()
883                                );
884
885                                for (key, (previous_snapshot, snapshot)) in &diff {
886                                    let _ = sh_eprintln!(
887                                        "{}",
888                                        format!("- [{key}] {previous_snapshot} → {snapshot}").red()
889                                    );
890                                }
891
892                                found = true;
893                            }
894
895                            found
896                        },
897                    );
898
899                    if differences_found {
900                        sh_eprintln!()?;
901                        eyre::bail!("Snapshots differ from previous run");
902                    }
903                }
904
905                // By default `gas_snapshot_emit` is set to `true` in the config.
906                //
907                // The user can either:
908                // - Set `FORGE_SNAPSHOT_EMIT=false` in the environment.
909                // - Pass `--gas-snapshot-emit=false` as a CLI argument.
910                // - Set `gas_snapshot_emit = false` in the config.
911                //
912                // If the user passes `--gas-snapshot-emit=<bool>` then it will override the config
913                // and the environment variable, enabling the check if `true` is passed.
914                if self.gas_snapshot_emit.unwrap_or(config.gas_snapshot_emit) {
915                    // Create `snapshots` directory if it doesn't exist.
916                    fs::create_dir_all(&config.snapshots)?;
917
918                    // Write gas snapshots to disk per group.
919                    gas_snapshots.clone().into_iter().for_each(|(group, snapshots)| {
920                        fs::write_pretty_json_file(
921                            &config.snapshots.join(format!("{group}.json")),
922                            &snapshots,
923                        )
924                        .expect("Failed to write gas snapshots to disk");
925                    });
926                }
927            }
928
929            // Print suite summary.
930            if !silent && has_tests {
931                sh_println!("{}", suite_result.summary())?;
932            }
933
934            // Add the suite result to the outcome.
935            outcome.results.insert(contract_name, suite_result);
936
937            // Stop processing the remaining suites if any test failed and `fail_fast` is set.
938            if self.fail_fast && any_test_failed {
939                break;
940            }
941        }
942        outcome.last_run_decoder = Some(decoder);
943        let duration = timer.elapsed();
944
945        trace!(target: "forge::test", len=outcome.results.len(), %any_test_failed, "done with results");
946
947        if let Some(gas_report) = gas_report {
948            let finalized = gas_report.finalize();
949            sh_println!("{}", &finalized)?;
950            outcome.gas_report = Some(finalized);
951        }
952
953        if !self.summary && !shell::is_json() {
954            sh_println!("{}", outcome.summary(duration))?;
955        }
956
957        if self.summary && !outcome.results.is_empty() {
958            let summary_report = TestSummaryReport::new(self.detailed, outcome.clone());
959            sh_println!("{}", &summary_report)?;
960        }
961
962        // Reattach the task.
963        match handle.await {
964            Ok(result) => outcome.runner = Some(result?),
965            Err(e) => match e.try_into_panic() {
966                Ok(payload) => std::panic::resume_unwind(payload),
967                Err(e) => return Err(e.into()),
968            },
969        }
970
971        // Persist test run failures to enable replaying.
972        persist_run_failures(&config, &outcome);
973
974        Ok(outcome)
975    }

977    /// Returns the flattened [`FilterArgs`] arguments merged with [`Config`].
978    /// Loads and applies filter from file if only last test run failures performed.
979    pub fn filter(&self, config: &Config) -> Result<ProjectPathsAwareFilter> {
980        let mut filter = self.filter.clone();
981        if self.rerun {
982            filter.test_pattern = last_run_failures(config);
983        }
984        if filter.path_pattern.is_some() {
985            if self.path.is_some() {
986                bail!("Can not supply both --match-path and |path|");
987            }
988        } else {
989            filter.path_pattern = self.path.clone();
990        }
991        Ok(filter.merge_with_config(config))
992    }

995    pub fn is_watch(&self) -> bool {
996        self.watch.watch.is_some()
997    }

999    /// Returns the [`watchexec::Config`] necessary to bootstrap a new watch loop.
1000    pub(crate) fn watchexec_config(&self) -> Result<watchexec::Config> {
1001        self.watch.watchexec_config(|| {
1002            let config = self.load_config()?;
1003            let foundry_toml: PathBuf = config.root.join(Config::FILE_NAME);
1004            Ok([config.src, config.test, config.script, foundry_toml])
1005        })
1006    }

1010    fn metadata(&self) -> Metadata {
1011        Metadata::named("Core Build Args Provider")
1012    }
1013
1014    fn data(&self) -> Result<Map<Profile, Dict>, figment::Error> {
1015        let mut dict = Dict::default();
1016
1017        let mut fuzz_dict = Dict::default();
1018        if let Some(fuzz_seed) = self.fuzz_seed {
1019            fuzz_dict.insert("seed".to_string(), fuzz_seed.to_string().into());
1020        }
1021        if let Some(fuzz_runs) = self.fuzz_runs {
1022            fuzz_dict.insert("runs".to_string(), fuzz_runs.into());
1023        }
1024        if let Some(fuzz_timeout) = self.fuzz_timeout {
1025            fuzz_dict.insert("timeout".to_string(), fuzz_timeout.into());
1026        }
1027        if let Some(fuzz_input_file) = self.fuzz_input_file.clone() {
1028            fuzz_dict.insert("failure_persist_file".to_string(), fuzz_input_file.into());
1029        }
1030        if let Some(ref fuzz_int_max) = self.fuzz_int_max
1031            && let Ok(max_val) = parse_fuzz_int_max(fuzz_int_max)
1032        {
1033            fuzz_dict.insert("max_fuzz_int".to_string(), max_val.to_string().into());
1034        }
1035        dict.insert("fuzz".to_string(), fuzz_dict.into());
1036
1037        if let Some(etherscan_api_key) =
1038            self.etherscan_api_key.as_ref().filter(|s| !s.trim().is_empty())
1039        {
1040            dict.insert("etherscan_api_key".to_string(), etherscan_api_key.to_string().into());
1041        }
1042
1043        if self.show_progress {
1044            dict.insert("show_progress".to_string(), true.into());
1045        }
1046
1047        Ok(Map::from([(Config::selected_profile(), dict)]))
1048    }

1069/// Load persisted filter (with last test run failures) from file.
1070fn last_run_failures(config: &Config) -> Option<regex::Regex> {
1071    match fs::read_to_string(&config.test_failures_file) {
1072        Ok(filter) => Regex::new(&filter)
1073            .inspect_err(|e| {
1074                _ = sh_warn!(
1075                    "failed to parse test filter from {:?}: {e}",
1076                    config.test_failures_file
1077                )
1078            })
1079            .ok(),
1080        Err(_) => None,
1081    }
1082}

1084/// Persist filter with last test run failures (only if there's any failure).
1085fn persist_run_failures(config: &Config, outcome: &TestOutcome) {
1086    if outcome.failed() > 0 && fs::create_file(&config.test_failures_file).is_ok() {
1087        let mut filter = String::new();
1088        let mut failures = outcome.failures().peekable();
1089        while let Some((test_name, _)) = failures.next() {
1090            if test_name.is_any_test()
1091                && let Some(test_match) = test_name.split("(").next()
1092            {
1093                filter.push_str(test_match);
1094                if failures.peek().is_some() {
1095                    filter.push('|');
1096                }
1097            }
1098        }
1099        let _ = fs::write(&config.test_failures_file, filter);
1100    }

1103/// Generate test report in JUnit XML report format.
1104fn junit_xml_report(results: &BTreeMap<String, SuiteResult>, verbosity: u8) -> Report {
1105    let mut total_duration = Duration::default();
1106    let mut junit_report = Report::new("Test run");
1107    junit_report.set_timestamp(Utc::now());
1108    for (suite_name, suite_result) in results {
1109        let mut test_suite = TestSuite::new(suite_name);
1110        total_duration += suite_result.duration;
1111        test_suite.set_time(suite_result.duration);
1112        test_suite.set_system_out(suite_result.summary());
1113        for (test_name, test_result) in &suite_result.test_results {
1114            let mut test_status = match test_result.status {
1115                TestStatus::Success => TestCaseStatus::success(),
1116                TestStatus::Failure => TestCaseStatus::non_success(NonSuccessKind::Failure),
1117                TestStatus::Skipped => TestCaseStatus::skipped(),
1118            };
1119            if let Some(reason) = &test_result.reason {
1120                test_status.set_message(reason);
1121            }
1122
1123            let mut test_case = TestCase::new(test_name, test_status);
1124            test_case.set_time(test_result.duration);
1125
1126            let mut sys_out = String::new();
1127            let result_report = test_result.kind.report();
1128            write!(sys_out, "{test_result} {test_name} {result_report}").unwrap();
1129            if verbosity >= 2 && !test_result.logs.is_empty() {
1130                write!(sys_out, "\\nLogs:\\n").unwrap();
1131                let console_logs = decode_console_logs(&test_result.logs);
1132                for log in console_logs {
1133                    write!(sys_out, "  {log}\\n").unwrap();
1134                }
1135            }
1136
1137            test_case.set_system_out(sys_out);
1138            test_suite.add_test_case(test_case);
1139        }
1140        junit_report.add_test_suite(test_suite);
1141    }
1142    junit_report.set_time(total_duration);
1143    junit_report
1144}

1146/// Parses the fuzz-int-max value from string to U256.
1147/// Supports:
1148/// - Decimal: "340282366920938463463374607431768211455"
1149/// - Hex: "0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFF"
1150/// - Keywords: "u128", "u128_max", "u64", "u64_max"
1151fn parse_fuzz_int_max(value: &str) -> Result<U256> {
1152    let value = value.trim().to_lowercase();
1153    match value.as_str() {
1154        "u128" | "u128_max" => Ok(U256::from(u128::MAX)),
1155        "u64" | "u64_max" => Ok(U256::from(u64::MAX)),
1156        _ if value.starts_with("0x") => U256::from_str_radix(&value[2..], 16)
1157            .map_err(|e| eyre::eyre!("Invalid hex value for --fuzz-int-max: {e}")),
1158        _ => {
1159            value.parse::<U256>().map_err(|e| eyre::eyre!("Invalid value for --fuzz-int-max: {e}"))
1160        }
1161    }
1162}
