27/// CLI arguments for `forge coverage`.
28#[derive(Parser)]
29pub struct CoverageArgs {
30    /// The report type to use for coverage.
31    ///
32    /// This flag can be used multiple times.
33    #[arg(long, value_enum, default_value = "summary")]
34    report: Vec<CoverageReportKind>,
35
36    /// The version of the LCOV "tracefile" format to use.
37    ///
38    /// Format: `MAJOR[.MINOR]`.
39    ///
40    /// Main differences:
41    /// - `1.x`: The original v1 format.
42    /// - `2.0`: Adds support for "line end" numbers for functions.
43    /// - `2.2`: Changes the format of functions.
44    #[arg(long, default_value = "1", value_parser = parse_lcov_version)]
45    lcov_version: Version,
46
47    /// Enable viaIR with minimum optimization
48    ///
49    /// This can fix most of the "stack too deep" errors while resulting a
50    /// relatively accurate source map.
51    #[arg(long)]
52    ir_minimum: bool,
53
54    /// The path to output the report.
55    ///
56    /// If not specified, the report will be stored in the root of the project.
57    #[arg(
58        long,
59        short,
60        value_hint = ValueHint::FilePath,
61        value_name = "PATH"
62    )]
63    report_file: Option<PathBuf>,
64
65    /// Whether to include libraries in the coverage report.
66    #[arg(long)]
67    include_libs: bool,
68
69    /// Whether to exclude tests from the coverage report.
70    #[arg(long)]
71    exclude_tests: bool,
72
73    /// The coverage reporters to use. Constructed from the other fields.
74    #[arg(skip)]
75    reporters: Vec<Box<dyn CoverageReporter>>,
76
77    #[command(flatten)]
78    test: TestArgs,
79}

81impl CoverageArgs {
82    pub async fn run(mut self) -> Result<()> {
83        let (mut config, evm_opts) = self.load_config_and_evm_opts()?;
84
85        // install missing dependencies
86        if install::install_missing_dependencies(&mut config).await && config.auto_detect_remappings
87        {
88            // need to re-configure here to also catch additional remappings
89            config = self.load_config()?;
90        }
91
92        // Set fuzz seed so coverage reports are deterministic
93        config.fuzz.seed = Some(U256::from_be_bytes(STATIC_FUZZ_SEED));
94
95        let (paths, mut output) = {
96            let (project, output) = self.build(&config)?;
97            (project.paths, output)
98        };
99
100        self.populate_reporters(&paths.root);
101
102        sh_println!("Analysing contracts...")?;
103        let report = self.prepare(&paths, &mut output)?;
104
105        sh_println!("Running tests...")?;
106        self.collect(&paths.root, std::sync::Arc::new(output), report, config, evm_opts, None).await
107    }
108
109    fn populate_reporters(&mut self, root: &Path) {
110        self.reporters = self
111            .report
112            .iter()
113            .map(|report_kind| match report_kind {
114                CoverageReportKind::Summary => {
115                    Box::<CoverageSummaryReporter>::default() as Box<dyn CoverageReporter>
116                }
117                CoverageReportKind::Lcov => {
118                    let path =
119                        root.join(self.report_file.as_deref().unwrap_or("lcov.info".as_ref()));
120                    Box::new(LcovReporter::new(path, self.lcov_version.clone()))
121                }
122                CoverageReportKind::Bytecode => Box::new(BytecodeReporter::new(
123                    root.to_path_buf(),
124                    root.join("bytecode-coverage"),
125                )),
126                CoverageReportKind::Debug => Box::new(DebugReporter),
127            })
128            .collect::<Vec<_>>();
129    }

131    /// Builds the project.
132    fn build(&self, config: &Config) -> Result<(Project, ProjectCompileOutput)> {
133        let mut project = config.ephemeral_project()?;
134
135        if self.ir_minimum {
136            sh_warn!(
137                "`--ir-minimum` enables `viaIR` with minimum optimization, \
138                 which can result in inaccurate source mappings.\n\
139                 Only use this flag as a workaround if you are experiencing \"stack too deep\" errors.\n\
140                 Note that `viaIR` is production ready since Solidity 0.8.13 and above.\n\
141                 See more: https://github.com/foundry-rs/foundry/issues/3357"
142            )?;
143        } else {
144            sh_warn!(
145                "optimizer settings and `viaIR` have been disabled for accurate coverage reports.\n\
146                 If you encounter \"stack too deep\" errors, consider using `--ir-minimum` which \
147                 enables `viaIR` with minimum optimization resolving most of the errors"
148            )?;
149        }
150
151        config.disable_optimizations(&mut project, self.ir_minimum);
152
153        let output = ProjectCompiler::default()
154            .compile(&project)?
155            .with_stripped_file_prefixes(project.root());
156
157        Ok((project, output))
158    }

160    /// Builds the coverage report.
161    #[instrument(name = "Coverage::prepare", skip_all)]
162    fn prepare(
163        &self,
164        project_paths: &ProjectPathsConfig,
165        output: &mut ProjectCompileOutput,
166    ) -> Result<CoverageReport> {
167        let mut report = CoverageReport::default();
168
169        output.parser_mut().solc_mut().compiler_mut().enter_mut(|compiler| {
170            if compiler.gcx().stage() < Some(solar::config::CompilerStage::Lowering) {
171                let _ = compiler.lower_asts();
172            }
173            convert_solar_errors(compiler.dcx())
174        })?;
175        let output = &*output;
176
177        // Collect source files.
178        let mut versioned_sources = HashMap::<Version, SourceFiles>::default();
179        for (path, source_file, version) in output.output().sources.sources_with_version() {
180            // Filter out vyper sources.
181            if path
182                .extension()
183                .and_then(|s| s.to_str())
184                .is_some_and(|ext| VYPER_EXTENSIONS.contains(&ext))
185            {
186                continue;
187            }
188
189            report.add_source(version.clone(), source_file.id as usize, path.clone());
190
191            // Filter out libs dependencies and tests.
192            if (!self.include_libs && project_paths.has_library_ancestor(path))
193                || (self.exclude_tests && project_paths.is_test(path))
194            {
195                continue;
196            }
197
198            let path = project_paths.root.join(path);
199            versioned_sources
200                .entry(version.clone())
201                .or_default()
202                .sources
203                .insert(source_file.id, path);
204        }
205
206        // Get source maps and bytecodes.
207        let artifacts: Vec<ArtifactData> = output
208            .artifact_ids()
209            .par_bridge() // This parses source maps, so we want to run it in parallel.
210            .filter_map(|(id, artifact)| {
211                let source_id = report.get_source_id(id.version.clone(), id.source.clone())?;
212                ArtifactData::new(&id, source_id, artifact)
213            })
214            .collect();
215
216        // Add coverage items.
217        for (version, sources) in &versioned_sources {
218            let source_analysis = SourceAnalysis::new(sources, output)?;
219            let anchors = artifacts
220                .par_iter()
221                .filter(|artifact| artifact.contract_id.version == *version)
222                .map(|artifact| {
223                    let creation_code_anchors = artifact.creation.find_anchors(&source_analysis);
224                    let deployed_code_anchors = artifact.deployed.find_anchors(&source_analysis);
225                    (artifact.contract_id.clone(), (creation_code_anchors, deployed_code_anchors))
226                })
227                .collect_vec_list();
228            report.add_anchors(anchors.into_iter().flatten());
229            report.add_analysis(version.clone(), source_analysis);
230        }
231
232        if self.reporters.iter().any(|reporter| reporter.needs_source_maps()) {
233            report.add_source_maps(artifacts.into_iter().map(|artifact| {
234                (artifact.contract_id, (artifact.creation.source_map, artifact.deployed.source_map))
235            }));
236        }
237
238        Ok(report)
239    }

241    /// Runs tests, collects coverage data and generates the final report.
242    #[instrument(name = "Coverage::collect", skip_all)]
243    async fn collect(
244        mut self,
245        project_root: &Path,
246        output: std::sync::Arc<ProjectCompileOutput>,
247        mut report: CoverageReport,
248        config: Config,
249        evm_opts: EvmOpts,
250        dual_compiled_contracts: Option<DualCompiledContracts>,
251    ) -> Result<()> {
252        let filter = self.test.filter(&config)?;
253        let outcome = self
254            .test
255            .run_tests(
256                project_root,
257                config,
258                evm_opts,
259                output,
260                None,
261                &filter,
262                true,
263                dual_compiled_contracts,
264            )
265            .await?;
266        outcome.ensure_ok(false)?;
267
268        let known_contracts = outcome.runner.as_ref().unwrap().known_contracts.clone();
269
270        // Add hit data to the coverage report
271        let data = outcome.results.values().flat_map(|suite| {
272            let mut hits = Vec::new();
273            for result in suite.test_results.values() {
274                let Some(hit_maps) = result.line_coverage.as_ref() else { continue };
275                for map in hit_maps.0.values() {
276                    if let Some((id, _)) = known_contracts.find_by_deployed_code(map.bytecode()) {
277                        hits.push((id, map, true));
278                    } else if let Some((id, _)) =
279                        known_contracts.find_by_creation_code(map.bytecode())
280                    {
281                        hits.push((id, map, false));
282                    }
283                }
284            }
285            hits
286        });
287
288        for (artifact_id, map, is_deployed_code) in data {
289            if let Some(source_id) =
290                report.get_source_id(artifact_id.version.clone(), artifact_id.source.clone())
291            {
292                report.add_hit_map(
293                    &ContractId {
294                        version: artifact_id.version.clone(),
295                        source_id,
296                        contract_name: artifact_id.name.as_str().into(),
297                    },
298                    map,
299                    is_deployed_code,
300                )?;
301            }
302        }
303
304        // Filter out ignored sources from the report.
305        if let Some(not_re) = &filter.args().coverage_pattern_inverse {
306            let file_root = filter.paths().root.as_path();
307            report.retain_sources(|path: &Path| {
308                let path = path.strip_prefix(file_root).unwrap_or(path);
309                !not_re.is_match(&path.to_string_lossy())
310            });
311        }
312
313        // Output final reports.
314        self.report(&report)?;
315
316        Ok(())
317    }

319    #[instrument(name = "Coverage::report", skip_all)]
320    fn report(&mut self, report: &CoverageReport) -> Result<()> {
321        for reporter in &mut self.reporters {
322            let _guard = debug_span!("reporter.report", kind=%reporter.name()).entered();
323            reporter.report(report)?;
324        }
325        Ok(())
326    }

347/// Helper function that will link references in unlinked bytecode to the 0 address.
348///
349/// This is needed in order to analyze the bytecode for contracts that use libraries.
350fn dummy_link_bytecode(mut obj: CompactBytecode) -> Option<Bytes> {
351    let link_references = obj.link_references.clone();
352    for (file, libraries) in link_references {
353        for library in libraries.keys() {
354            obj.link(&file, library, Address::ZERO);
355        }
356    }
357
358    obj.object.resolve();
359    obj.object.into_bytes()
360}
361
362/// Helper function that will link references in unlinked bytecode to the 0 address.
363///
364/// This is needed in order to analyze the bytecode for contracts that use libraries.
365fn dummy_link_deployed_bytecode(obj: CompactDeployedBytecode) -> Option<Bytes> {
366    obj.bytecode.and_then(dummy_link_bytecode)
367}

369pub struct ArtifactData {
370    pub contract_id: ContractId,
371    pub creation: BytecodeData,
372    pub deployed: BytecodeData,
373}
374
375impl ArtifactData {
376    pub fn new(id: &ArtifactId, source_id: usize, artifact: &impl Artifact) -> Option<Self> {
377        Some(Self {
378            contract_id: ContractId {
379                version: id.version.clone(),
380                source_id,
381                contract_name: id.name.as_str().into(),
382            },
383            creation: BytecodeData::new(
384                artifact.get_source_map()?.ok()?,
385                artifact
386                    .get_bytecode()
387                    .and_then(|bytecode| dummy_link_bytecode(bytecode.into_owned()))?,
388            ),
389            deployed: BytecodeData::new(
390                artifact.get_source_map_deployed()?.ok()?,
391                artifact
392                    .get_deployed_bytecode()
393                    .and_then(|bytecode| dummy_link_deployed_bytecode(bytecode.into_owned()))?,
394            ),
395        })
396    }
397}

399pub struct BytecodeData {
400    source_map: SourceMap,
401    bytecode: Bytes,
402    /// The instruction counter to program counter mapping.
403    ///
404    /// The source maps are indexed by *instruction counters*, which are the indexes of
405    /// instructions in the bytecode *minus any push bytes*.
406    ///
407    /// Since our line coverage inspector collects hit data using program counters, the anchors
408    /// also need to be based on program counters.
409    ic_pc_map: IcPcMap,
410}
411
412impl BytecodeData {
413    fn new(source_map: SourceMap, bytecode: Bytes) -> Self {
414        let ic_pc_map = IcPcMap::new(&bytecode);
415        Self { source_map, bytecode, ic_pc_map }
416    }
417
418    pub fn find_anchors(&self, source_analysis: &SourceAnalysis) -> Vec<ItemAnchor> {
419        find_anchors(&self.bytecode, &self.source_map, &self.ic_pc_map, source_analysis)
420    }
421}

423fn parse_lcov_version(s: &str) -> Result<Version, String> {
424    let vr = VersionReq::parse(&format!("={s}")).map_err(|e| e.to_string())?;
425    let [c] = &vr.comparators[..] else {
426        return Err("invalid version".to_string());
427    };
428    if c.op != semver::Op::Exact {
429        return Err("invalid version".to_string());
430    }
431    if !c.pre.is_empty() {
432        return Err("pre-releases are not supported".to_string());
433    }
434    Ok(Version::new(c.major, c.minor.unwrap_or(0), c.patch.unwrap_or(0)))
435}
