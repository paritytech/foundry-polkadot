42/// Initializes tracing for tests.
43pub fn init_tracing() {
44    use std::sync::Once;
45    static ONCE: Once = Once::new();
46    ONCE.call_once(|| {
47        if std::env::var_os("RUST_BACKTRACE").is_none() {
48            unsafe { std::env::set_var("RUST_BACKTRACE", "1") };
49        }
50        let _ = tracing_subscriber::FmtSubscriber::builder()
51            .with_env_filter(env_filter())
52            .with_test_writer()
53            .try_init();
54        let _ = ui_test::color_eyre::install();
55    });
56}
57
58fn env_filter() -> tracing_subscriber::EnvFilter {
59    const DEFAULT_DIRECTIVES: &[&str] = &include!("../../cli/src/utils/default_directives.txt");
60    let mut filter = tracing_subscriber::EnvFilter::builder()
61        .with_default_directive("foundry_test_utils=debug".parse().unwrap())
62        .from_env_lossy();
63    for &directive in DEFAULT_DIRECTIVES {
64        filter = filter.add_directive(directive.parse().unwrap());
65    }
66    filter
67}
